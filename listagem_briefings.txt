"""
Briefings — conversa estruturada entre arquiteto e cliente.

============================================================================
REGRA ARQUITETURAL
============================================================================

`briefings` responde por:

    - modelos reutilizáveis de briefing (template)
    - execução de um briefing para um cliente específico
    - histórico imutável de eventos da conversa
    - projeção do estado atual de cada pergunta
    - transcrição e análise (IA) rastreáveis

`briefings` NÃO responde por:

    - identidade do usuário           → pertence a `accounts`
    - contexto do cliente             → pertence a `clients`
    - armazenamento de arquivos       → pertence a `documents`
    - projetos                        → pertence a `projects` (futuro)
    - timeline transversal            → pertence a `timeline` (futuro)

============================================================================
PRINCÍPIOS
============================================================================

1. Separação modelo / execução
   BriefingTemplate é reutilizável. Briefing é uma execução imutável
   para um cliente. Alterar o template NÃO altera execuções existentes
   (estrutura é copiada para um snapshot no momento da criação).

2. Tipo da pergunta ≠ formato da resposta
   `question_type` descreve a intenção do arquiteto.
   `allowed_response_formats` define o que o cliente pode enviar.

3. Histórico append-only
   BriefingEvent nunca é editado nem apagado. Cada resposta, dúvida,
   pedido de revisão ou intervenção é um novo evento.

4. Estado atual é projeção
   BriefingAnswer é a projeção do estado atual de uma pergunta.
   É derivada dos eventos, mas mantida explicitamente para consulta
   rápida. Respostas anteriores permanecem no histórico de eventos.

5. Transcrição e análise têm estado próprio
   Uma falha na transcrição NÃO invalida a resposta original.
   Uma análise antiga nunca parece atual (campo `is_stale`).

6. Rastreabilidade
   Cada item da análise aponta para a pergunta e o evento de origem.

============================================================================
MODELOS (V1)
============================================================================

BriefingTemplate        → modelo reutilizável criado pelo arquiteto
BriefingSection         → agrupamento de perguntas dentro do template
BriefingQuestion        → pergunta dentro de uma sessão do template

Briefing                → execução de um template para um cliente
BriefingAnswer          → projeção do estado atual de uma pergunta
BriefingEvent           → histórico imutável de eventos da conversa

============================================================================
FRONTEIRA CONGELADA (V1)
============================================================================

    accounts      → identidade
    workspaces    → tenancy
    console       → governança, capabilities, planos
    clients       → cliente e portal
    documents     → armazenamento de arquivos (imagens, áudio)
    briefings     → este app
    projects      → operação (futuro)
    timeline      → memória factual (futuro)
    intelligence  → interpretação (futuro)

"""

default_app_config = 'apps.briefings.apps.BriefingsConfig'

    state_display = serializers.CharField(
        source='get_state_display', read_only=True,
    )
    transcription_status_display = serializers.CharField(
        source='get_transcription_status_display', read_only=True,
    )
    latest_audio_url = serializers.SerializerMethodField()

    class Meta:
        model = BriefingAnswer
        fields = [
            'id',
            'question_id',
            'state',
            'state_display',
            'latest_content',
            'latest_audio',
            'latest_audio_url',
            'latest_transcription',
            'transcription_status',
            'transcription_status_display',
            'latest_event',
            'attachment_ids',
            'created_at',
            'updated_at',
        ]
        read_only_fields = fields

    def get_latest_audio_url(self, obj):
        if not obj.latest_audio or not obj.latest_audio.file:
            return ''
        try:
            return obj.latest_audio.file.url
        except Exception:
            return ''

    def get_attachment_ids(self, obj):
        return [str(d.id) for d in obj.attachments.all()]

"""
Serializers do portal do cliente.

Estes serializers são consumidos por views autenticadas via
`ClientPortalSession`. A estrutura do briefing é derivada do
`structure_snapshot` + os estados atuais das perguntas.
"""

from rest_framework import serializers

from apps.briefings.models import Briefing, BriefingAnswer


# =========================================================================
# LEITURA — o briefing visto pelo cliente
# =========================================================================

class PortalBriefingSerializer(serializers.Serializer):
    """
    Envelope do briefing para o cliente.

    Estrutura:
        {
            "id", "title", "status", "expires_at",
            "progress",                       # vindo de BriefingService
            "next_question_id",               # próxima elegível (ou null)
            "sections": [
                {
                    "id", "title", "description",
                    "questions": [
                        {
                            "id", "text", "instructions",
                            "allowed_response_formats",
                            "is_required",
                            "reference_document_ids",
                            "state",              # da projeção
                            "latest_content",
                            "latest_audio_url",
                            "latest_transcription",
                            "transcription_status",
                            "attachment_ids",
                        }
                    ]
                }
            ]
        }

    Nunca expõe dados internos do workspace.
    """

    def to_representation(self, briefing: Briefing):
        from apps.briefings.services.briefing_service import BriefingService

        answers = {
            str(a.question_id): a
            for a in briefing.answers.select_related(
                'latest_audio',
            ).prefetch_related('attachments')
        }

        sections = []
        ordered_sections = sorted(
            briefing.structure_snapshot.get('sections', []),
            key=lambda s: s.get('position', 0),
        )
        for section in ordered_sections:
            questions_payload = []
            ordered_questions = sorted(
                section.get('questions', []),
                key=lambda q: q.get('position', 0),
            )
            for q in ordered_questions:
                qid = str(q['id'])
                answer = answers.get(qid)
                latest_audio_url = ''
                if answer and answer.latest_audio and answer.latest_audio.file:
                    try:
                        latest_audio_url = answer.latest_audio.file.url
                    except Exception:
                        latest_audio_url = ''

                questions_payload.append({
                    'id': qid,
                    'text': q.get('text', ''),
                    'instructions': q.get('instructions', ''),
                    'question_type': q.get('question_type', 'text'),
                    'allowed_response_formats': list(
                        q.get('allowed_response_formats', [])
                    ),
                    'is_required': bool(q.get('is_required', True)),
                    'position': q.get('position', 0),
                    'reference_document_ids': list(
                        q.get('reference_document_ids', [])
                    ),
                    'state': answer.state if answer else BriefingAnswer.STATE_PENDING,
                    'latest_content': answer.latest_content if answer else '',
                    'latest_audio_url': latest_audio_url,
                    'latest_transcription': (
                        answer.latest_transcription if answer else ''
                    ),
                    'transcription_status': (
                        answer.transcription_status if answer
                        else BriefingAnswer.TRANSCRIPTION_NOT_APPLICABLE
                    ),
                    'attachment_ids': (
                        [str(d.id) for d in answer.attachments.all()]
                        if answer else []
                    ),
                })
            sections.append({
                'id': section.get('id'),
                'title': section.get('title', ''),
                'description': section.get('description', ''),
                'position': section.get('position', 0),
                'questions': questions_payload,
            })

        next_q = BriefingService.get_next_eligible_question(briefing)

        return {
            'id': str(briefing.id),
            'title': briefing.title,
            'status': briefing.status,
            'expires_at': (
                briefing.expires_at.isoformat()
                if briefing.expires_at else None
            ),
            'progress': BriefingService.compute_progress(briefing),
            'next_question_id': str(next_q['id']) if next_q else None,
            'sections': sections,
        }

import logging

    BriefingNoteNotFoundError,
)

logger = logging.getLogger('briefings')


class BriefingViewSet(viewsets.GenericViewSet):

    permission_classes = [IsAuthenticated]
    lookup_field = 'id'

    # ------------------------------------------------------------------
    # CONTEXTO
    # ------------------------------------------------------------------

    def _current_workspace(self):
        workspace_id = self.request.session.get('current_workspace_id')
        if not workspace_id:
            return None
        return WorkspaceQueries.get_for_user(workspace_id, self.request.user)

    def _require_capability(self, workspace, capability_code):
        membership = WorkspaceMembership.objects.filter(
            workspace=workspace,
            user=self.request.user,
        ).select_related('governance').first()

        if membership is None:
            return Response(
                {'error': 'Você não é membro deste Workspace'},
                status=status.HTTP_403_FORBIDDEN,
            )
        if membership.governance.key == 'proprietario':
            return None
        has = ConsoleGovernanceCapability.objects.filter(
            governance=membership.governance,
            capability__code=capability_code,
            capability__is_active=True,
        ).exists()
        if not has:
            return Response(
                {'error': f'Sua governança não permite esta ação ({capability_code}).'},
                status=status.HTTP_403_FORBIDDEN,
            )
        return None

    @staticmethod
    def _handle_domain_error(exc):
        if isinstance(exc, BriefingNotFoundError):
            return Response({'error': str(exc)}, status=404)
        if isinstance(exc, BriefingReopenError):
            return Response({'error': str(exc)}, status=409)
        if isinstance(exc, BriefingClosedError):
            return Response({'error': str(exc)}, status=409)
        if isinstance(exc, BriefingExpiredError):
            return Response({'error': str(exc)}, status=409)
        if isinstance(exc, BriefingPausedError):
            return Response({'error': str(exc)}, status=409)
        if isinstance(exc, BriefingNoteNotFoundError):
            return Response({'error': str(exc)}, status=404)
        if isinstance(exc, BriefingError):
            return Response({'error': str(exc)}, status=400)
        return None

    # ------------------------------------------------------------------
    # CRUD
    # ------------------------------------------------------------------

    def list(self, request):
        workspace = self._current_workspace()
        if workspace is None:
            return Response({'error': 'Nenhum workspace ativo'}, status=400)
        err = self._require_capability(workspace, 'briefings.briefing.view')
        if err is not None:
            return err

        status_filter = request.query_params.get('status')
        client_id = request.query_params.get('client')

        qs = workspace.briefings.select_related(
            'client', 'template',
        ).order_by('-created_at')
        if status_filter:
            qs = qs.filter(status=status_filter)
        if client_id:
            qs = qs.filter(client_id=client_id)

        return Response(BriefingListSerializer(qs, many=True).data)

    def create(self, request):
        workspace = self._current_workspace()
        if workspace is None:
            return Response({'error': 'Nenhum workspace ativo'}, status=400)
        err = self._require_capability(workspace, 'briefings.briefing.start')
        if err is not None:
            return err

        serializer = BriefingCreateSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        try:
            template = BriefingTemplateService.get_or_raise(
                serializer.validated_data['template_id'], workspace,
            )
        except Exception as exc:
            handled = self._handle_domain_error(exc)
            if handled is not None:
                return handled
            return Response({'error': 'Template não encontrado'}, status=404)

        client_id = serializer.validated_data['client_id']
        client = Client.objects.filter(
            id=client_id, workspace=workspace,
        ).first()
        if client is None:
            return Response({'error': 'Cliente não encontrado'}, status=404)

        try:
            briefing = BriefingService.create_briefing(
                template=template,
                client=client,
                workspace=workspace,
                actor=request.user,
                expires_at=serializer.validated_data.get('expires_at'),
                title=serializer.validated_data.get('title') or None,
            )
        except Exception as exc:
            handled = self._handle_domain_error(exc)
            if handled is not None:
                return handled
            logger.exception('Erro ao criar briefing')
            return Response({'error': 'Erro inesperado.'}, status=500)

        return Response(
            BriefingDetailSerializer(briefing).data,
            status=status.HTTP_201_CREATED,
        )

    def retrieve(self, request, id=None):
        workspace = self._current_workspace()
        if workspace is None:
            return Response({'error': 'Nenhum workspace ativo'}, status=400)
        err = self._require_capability(workspace, 'briefings.briefing.view')
        if err is not None:
            return err

        try:
            briefing = BriefingService.get_or_raise(id, workspace)
        except Exception as exc:
            handled = self._handle_domain_error(exc)
            if handled is not None:
                return handled
            return Response({'error': 'Erro'}, status=500)

        return Response(BriefingDetailSerializer(briefing).data)

    def destroy(self, request, id=None):
        workspace = self._current_workspace()
        if workspace is None:
            return Response({'error': 'Nenhum workspace ativo'}, status=400)
        err = self._require_capability(workspace, 'briefings.briefing.delete')
        if err is not None:
            return err

        try:
            briefing = BriefingService.get_or_raise(id, workspace)
        except Exception as exc:
            handled = self._handle_domain_error(exc)
            if handled is not None:
                return handled
            return Response({'error': 'Erro'}, status=500)

        briefing.delete()
        return Response(status=status.HTTP_204_NO_CONTENT)

class PortalBriefingAnswerView(APIView):

    permission_classes = [IsPortalSessionAuthenticated]

    def post(self, request, briefing_id=None, question_id=None):
        briefing = _resolve_briefing_or_404(briefing_id, request)
        if briefing is None:
            return Response({'error': 'Briefing não encontrado'}, status=404)

        serializer = PortalAnswerSubmitSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        # Resolve áudio e anexos como Documents do workspace
        audio_document = None
        if serializer.validated_data.get('audio_document_id'):
            audio_document = Document.objects.filter(
                id=serializer.validated_data['audio_document_id'],
                workspace=briefing.workspace,
            ).first()
            if audio_document is None:
                return Response({'error': 'Áudio não encontrado'}, status=400)

        attachments = []
        attachment_ids = serializer.validated_data.get('attachment_ids') or []
        if attachment_ids:
            attachments = list(Document.objects.filter(
                id__in=attachment_ids,
                workspace=briefing.workspace,
            ))
            if len(attachments) != len(set(str(a) for a in attachment_ids)):
                return Response(
                    {'error': 'Um ou mais anexos não foram encontrados'},
                    status=400,
                )

        try:
            answer = BriefingService.submit_answer(
                briefing_id=briefing.id,
                portal_session=request.portal_session,
                question_id=question_id,
                content_text=serializer.validated_data.get('text', ''),
                audio_document=audio_document,
                attachment_documents=attachments,
            )
        except Exception as exc:
            handled = _handle_error(exc)
            if handled is not None:
                return handled
            logger.exception('Erro inesperado ao submeter resposta')
            return Response({'error': 'Erro inesperado.'}, status=500)

        # Recarrega o briefing para refletir progresso
        briefing.refresh_from_db()
        return Response(PortalBriefingSerializer().to_representation(briefing))

    def _require_capability(self, workspace, capability_code):
        membership = WorkspaceMembership.objects.filter(
            workspace=workspace,
            user=self.request.user,
        ).select_related('governance').first()

        if membership is None:
            return Response(
                {'error': 'Você não é membro deste Workspace'},
                status=status.HTTP_403_FORBIDDEN,
            )
        if membership.governance.key == 'proprietario':
            return None
        has = ConsoleGovernanceCapability.objects.filter(
            governance=membership.governance,
            capability__code=capability_code,
            capability__is_active=True,
        ).exists()
        if not has:
            return Response(
                {'error': f'Sua governança não permite esta ação ({capability_code}).'},
                status=status.HTTP_403_FORBIDDEN,

"""
Modelo BriefingAnswer — projeção do estado atual de uma pergunta.

REGRA ARQUITETURAL
------------------
BriefingAnswer NÃO substitui o histórico. Ele é uma VIEW do estado
atual, mantida em sincronia com os eventos.

Toda pergunta respondida tem UM e apenas UM BriefingAnswer. Novas
respostas atualizam o registro (apontando para o novo evento), mas
o histórico de respostas anteriores permanece em BriefingEvent.

`state` reflete o estado DA PERGUNTA:

    pending               → ainda não respondida
    answered              → respondida e sem pendências
    pending_clarification → cliente sinalizou dúvida
    review_requested      → cliente pediu nova oportunidade de responder

`transcription_status` é INDEPENDENTE do estado da resposta.

PRESERVAÇÃO DO ÁUDIO
--------------------
`latest_audio` usa PROTECT. Um Document que representa um áudio
de resposta de briefing NÃO pode ser excluído enquanto a projeção
existir. Isso garante o requisito "preserve o arquivo de áudio
original". Se for necessário remover o áudio, o caminho é excluir
o Briefing inteiro — DELETE real em cascata.

"""
Modelo Briefing — execução de um template para um cliente.

------------------

Estados do briefing (`status`):

    in_progress  → briefing em andamento, aceita respostas

    expired      → passou do prazo e não foi concluído

`expires_at` é opcional. Se definido, o briefing expira quando o
prazo passa e o status é `in_progress`.

        help_text=_('Quando o arquiteto encerrou o briefing'),

        ]

    # ------------------------------------------------------------------
    # ESTADO
    # ------------------------------------------------------------------

    def is_answerable(self) -> bool:
        """Se o cliente pode enviar respostas agora."""
        if self.status != self.STATUS_IN_PROGRESS:
            return False
        if self.expires_at and timezone.now() >= self.expires_at:
            return False
        return True

    def has_expired(self) -> bool:
        if not self.expires_at:
            return False
        return timezone.now() >= self.expires_at

    def mark_expired_if_needed(self) -> bool:
        """Marca como expirado se o prazo passou e status ainda permite."""
        if (
            self.status == self.STATUS_IN_PROGRESS
            and self.has_expired()

            self.status = self.STATUS_EXPIRED
            self.save(update_fields=['status', 'updated_at'])
            return True
        return False

    # SNAPSHOT — helpers
    # ------------------------------------------------------------------

    def snapshot_questions(self) -> list:
        """Lista plana de perguntas na ordem (sessão, posição)."""
        result = []
        sections = sorted(
            self.structure_snapshot.get('sections', []),
            key=lambda s: s.get('position', 0),
        )
        for section in sections:
            questions = sorted(
                section.get('questions', []),
                key=lambda q: q.get('position', 0),
            )
            for q in questions:
                result.append({
                    **q,
                    'section_id': section.get('id'),
                    'section_title': section.get('title'),
                })
        return result

    def snapshot_question_by_id(self, question_id: str) -> dict:
        for q in self.snapshot_questions():
            if str(q.get('id')) == str(question_id):
                return q
        return None

        from django.core.exceptions import ValidationError

            })

        valid = {
            self.RESPONSE_FORMAT_TEXT,
            self.RESPONSE_FORMAT_AUDIO,
            self.RESPONSE_FORMAT_IMAGE,
        }
        invalid = [f for f in formats if f not in valid]
        if invalid:
            raise ValidationError({
                'allowed_response_formats': _(
                    f'Formatos inválidos: {invalid}. '
                    f'Válidos: {sorted(valid)}.'
                )
            })

    def save(self, *args, **kwargs):
        if not self.allowed_response_formats:
            self.allowed_response_formats = list(self.DEFAULT_ALLOWED_FORMATS)
        self.full_clean()
        super().save(*args, **kwargs)

            .order_by('-created_at')

    def get_template_in_workspace(template_id, workspace):

                .first()

            return None

    # BRIEFINGS

    def list_for_workspace(workspace, status=None):
        qs = (
            Briefing.objects
            .filter(workspace=workspace)
            .select_related('client', 'template')
            .order_by('-created_at')
        )
        if status:
            qs = qs.filter(status=status)
        return qs

    def get_in_workspace(briefing_id, workspace):
        try:
            return (
                Briefing.objects
                .filter(id=briefing_id, workspace=workspace)
                .select_related('client', 'template', 'workspace')
                .first()
            )
        except (ValueError, TypeError):
            return None

    @staticmethod
    def get_for_client(client, briefing_id):
        """
        Resolve um briefing no contexto do portal do cliente.

        NÃO confiar em UUIDs enviados pelo navegador. A checagem é
        sempre (client, id).
        """
        try:
            return (
                Briefing.objects
                .filter(id=briefing_id, client=client)
                .select_related('workspace', 'template', 'client')
                .first()
            )
        except (ValueError, TypeError):
            return None

"""
BriefingService — ciclo de vida da execução.

REGRA ARQUITETURAL
------------------
Toda mutação passa por este service. Nenhuma view escreve
diretamente em `Briefing`, `BriefingAnswer` ou `BriefingEvent`.

Concorrência:
    `submit_answer` roda dentro de `transaction.atomic` e usa
    `select_for_update` no Briefing para serializar envios
    concorrentes do mesmo cliente.

Sequência:
    O cliente só pode responder a próxima pergunta elegível.
    Validação é feita aqui, com base no snapshot.

ANÁLISE
-------
Este service NÃO gera análise. Análise pertence ao app `intelligence`.

NOTIFICAÇÃO
-----------
Ao concluir, agenda (via `on_commit`) o envio de e-mail através
do `NotificationService`.
"""

import logging
from typing import Optional

from django.db import transaction
from django.utils import timezone

from apps.briefings.models import (
    Briefing,
    BriefingAnswer,
    BriefingEvent,
    BriefingTemplate,
)
from apps.briefings.exceptions import (
    BriefingError,
    BriefingNotFoundError,
    BriefingClosedError,
    BriefingExpiredError,
    BriefingPausedError,
    InvalidQuestionSequenceError,
    InvalidResponseFormatError,
    BriefingReopenError,
)
from apps.briefings.services.event_service import BriefingEventService

logger = logging.getLogger('briefings')


class BriefingService:

    # ==================================================================
    # CRIAÇÃO
    # ==================================================================

    @staticmethod
    @transaction.atomic
    def create_briefing(
        *,
        template: BriefingTemplate,
        client,
        workspace,
        actor=None,
        expires_at=None,
        title: Optional[str] = None,
    ) -> Briefing:
        """
        Cria uma execução a partir de um template publicado.

        Copia a estrutura do template para `structure_snapshot`.
        """
        if template.workspace_id != workspace.id:
            raise BriefingError('Template não pertence a este workspace.')

        if client.workspace_id != workspace.id:
            raise BriefingError('Cliente não pertence a este workspace.')

        if template.status != BriefingTemplate.STATUS_PUBLISHED:
            raise BriefingError(
                'Só templates publicados podem originar execuções.'
            )

        if not template.is_active:
            raise BriefingError(
                'O template está inativo e não pode originar novas '
                'execuções.'
            )

        if not template.sections.exists():
            raise BriefingError(
                'O template não possui sessões para executar.'
            )

        snapshot = BriefingService._build_structure_snapshot(template)

        briefing = Briefing.objects.create(
            workspace=workspace,
            client=client,
            template=template,
            title=(title or template.name).strip(),
            structure_snapshot=snapshot,
            status=Briefing.STATUS_IN_PROGRESS,
            expires_at=expires_at,
            created_by=actor,
        )

        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_REOPENED,
            user=actor,
            content={
                'action': 'created',
                'template_id': str(template.id),
                'expires_at': (
                    expires_at.isoformat() if expires_at else None
                ),
            },
        )

        return briefing

    @staticmethod
    def _build_structure_snapshot(template: BriefingTemplate) -> dict:
        """
        Serializa a estrutura do template num dict imutável.
        """
        sections = list(
            template.sections
            .prefetch_related('questions__reference_documents')
            .order_by('position', 'created_at')
        )

        payload_sections = []
        for section in sections:
            payload_questions = []
            for question in section.questions.order_by(
                'position', 'created_at',
            ):
                payload_questions.append({
                    'id': str(question.id),
                    'text': question.text,
                    'instructions': question.instructions,
                    'question_type': question.question_type,
                    'allowed_response_formats': list(
                        question.allowed_response_formats or []
                    ),
                    'position': question.position,
                    'is_required': question.is_required,
                    'reference_document_ids': [
                        str(d.id)
                        for d in question.reference_documents.all()
                    ],
                })
            payload_sections.append({
                'id': str(section.id),
                'title': section.title,
                'description': section.description,
                'position': section.position,
                'questions': payload_questions,
            })

        return {
            'snapshot_version': 1,
            'template_name': template.name,
            'sections': payload_sections,
        }

    # ==================================================================
    # LEITURA / PROGRESSO
    # ==================================================================

    @staticmethod
    def get_or_raise(briefing_id, workspace) -> Briefing:
        try:
            briefing = (
                Briefing.objects
                .filter(id=briefing_id, workspace=workspace)
                .first()
            )
        except (ValueError, TypeError):
            briefing = None
        if briefing is None:
            raise BriefingNotFoundError('Briefing não encontrado.')
        return briefing

    @staticmethod
    def compute_progress(briefing: Briefing) -> dict:
        """
        Cálculo canônico de progresso.
        """
        questions = briefing.snapshot_questions()
        required_ids = [
            str(q['id']) for q in questions if q.get('is_required')
        ]
        total_required = len(required_ids)

        answers = {
            str(a.question_id): a
            for a in BriefingAnswer.objects.filter(briefing=briefing)
        }

        answered = 0
        pending_review = 0
        never_answered = 0

        for qid in required_ids:
            answer = answers.get(qid)
            if answer is None or answer.state == BriefingAnswer.STATE_PENDING:
                never_answered += 1
            elif answer.state == BriefingAnswer.STATE_ANSWERED:
                answered += 1
            elif answer.state in (
                BriefingAnswer.STATE_PENDING_CLARIFICATION,
                BriefingAnswer.STATE_REVIEW_REQUESTED,
            ):
                pending_review += 1

        percent = (
            round((answered / total_required) * 100, 2)
            if total_required > 0 else 0.0
        )

        return {
            'total_required': total_required,
            'answered': answered,
            'pending_review': pending_review,
            'never_answered': never_answered,
            'percent': percent,
            'needs_attention': pending_review > 0,
        }

    # RESPOSTA (fluxo do portal)
    # ==================================================================

    @staticmethod
    @transaction.atomic
    def submit_answer(
        *,
        briefing_id,
        portal_session,
        question_id,
        content_text: str = '',
        audio_document=None,
        attachment_documents=None,
    ) -> BriefingAnswer:
        """
        Registra uma resposta do cliente.
        """
        briefing = (
            Briefing.objects
            .select_for_update()
            .filter(id=briefing_id)
            .select_related('client', 'workspace')
            .first()
        )
        if briefing is None:
            raise BriefingNotFoundError('Briefing não encontrado.')

        if (
            portal_session is None
            or portal_session.client_id != briefing.client_id
        ):
            raise BriefingNotFoundError('Briefing não encontrado.')

        if briefing.status == Briefing.STATUS_PAUSED:
            raise BriefingPausedError('O briefing está pausado.')
        if briefing.status == Briefing.STATUS_CLOSED:
            raise BriefingClosedError('O briefing foi encerrado.')
        if briefing.status == Briefing.STATUS_EXPIRED:
            raise BriefingExpiredError('O briefing expirou.')
        if briefing.has_expired():
            briefing.status = Briefing.STATUS_EXPIRED
            briefing.save(update_fields=['status', 'updated_at'])
            raise BriefingExpiredError('O briefing expirou.')
        if briefing.status == Briefing.STATUS_COMPLETED:
            raise BriefingError('O briefing já foi concluído.')

        next_question = BriefingService.get_next_eligible_question(briefing)
        if next_question is None:
            raise BriefingError('Não há perguntas pendentes.')
        if str(next_question['id']) != str(question_id):
            raise InvalidQuestionSequenceError(
                'Você só pode responder à próxima pergunta da sequência.'
            )

        allowed = set(next_question.get('allowed_response_formats') or [])
        has_text = bool((content_text or '').strip())
        has_audio = audio_document is not None
        has_attachments = bool(attachment_documents)

        if has_text and 'text' not in allowed:
            raise InvalidResponseFormatError(
                'Esta pergunta não aceita resposta por texto.'
            )
        if has_audio and 'audio' not in allowed:
            raise InvalidResponseFormatError(
                'Esta pergunta não aceita resposta por áudio.'
            )
        if has_attachments and 'image' not in allowed:
            raise InvalidResponseFormatError(
                'Esta pergunta não aceita anexos de imagem.'
            )
        if not (has_text or has_audio):
            raise InvalidResponseFormatError(
                'Envie um texto ou um áudio.'
            )

        content_payload = {
            'text': (content_text or '').strip() or None,
            'audio_document_id': (
                str(audio_document.id) if audio_document else None
            ),
            'attachment_document_ids': [
                str(d.id) for d in (attachment_documents or [])
            ],
        }

        event = BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_ANSWERED,
            portal_session=portal_session,
            question_id=next_question['id'],
            content=content_payload,
        )

        answer, _ = BriefingAnswer.objects.select_for_update().get_or_create(
            briefing=briefing,
            question_id=next_question['id'],
            defaults={'state': BriefingAnswer.STATE_ANSWERED},
        )
        answer.state = BriefingAnswer.STATE_ANSWERED
        answer.latest_content = (content_text or '').strip()
        answer.latest_event = event

        if audio_document is not None:
            answer.latest_audio = audio_document
            answer.transcription_status = BriefingAnswer.TRANSCRIPTION_PENDING
            answer.latest_transcription = ''
        else:
            answer.latest_audio = None
            answer.transcription_status = (
                BriefingAnswer.TRANSCRIPTION_NOT_APPLICABLE
            )
            answer.latest_transcription = ''

        answer.save()

        if attachment_documents:
            answer.attachments.set(attachment_documents)

        # 8. Conclusão automática?
        if BriefingService._all_required_answered(briefing):
            briefing.status = Briefing.STATUS_COMPLETED
            briefing.completed_at = timezone.now()
            briefing.save(update_fields=[
                'status', 'completed_at', 'updated_at',
            ])
            BriefingEventService.log(
                briefing=briefing,
                kind=BriefingEvent.KIND_COMPLETED,
                portal_session=portal_session,
                content={'auto': True},
            )

            # Notificação por e-mail — FORA da transação
            from django.db import transaction as _tx
            _briefing_id = briefing.id
            _tx.on_commit(
                lambda: BriefingService._schedule_completion_notification(
                    _briefing_id,
                )
            )

        # 9. Agenda transcrição (best-effort)
        if audio_document is not None:
            try:
                from django.db import transaction as _tx
                _tx.on_commit(
                    lambda: BriefingService._schedule_transcription(
                        answer.id,
                    )
                )
            except Exception:
                logger.exception('Falha ao agendar transcrição')

        return answer

    # ==================================================================

    @staticmethod
    def get_next_eligible_question(briefing: Briefing) -> Optional[dict]:
        questions = briefing.snapshot_questions()
        answers = {
            str(a.question_id): a
            for a in BriefingAnswer.objects.filter(briefing=briefing)
        }

        for q in questions:
            if not q.get('is_required'):
                continue
            answer = answers.get(str(q['id']))
            if answer is None:
                return q
            if answer.state in (
                BriefingAnswer.STATE_PENDING,
                BriefingAnswer.STATE_REVIEW_REQUESTED,
                BriefingAnswer.STATE_PENDING_CLARIFICATION,
            ):
                return q
        return None

    @staticmethod
    def _all_required_answered(briefing: Briefing) -> bool:
        questions = briefing.snapshot_questions()
        required_ids = {
            str(q['id']) for q in questions if q.get('is_required')
        }
        if not required_ids:
            return True
        answered_ids = set(
            str(a.question_id)
            for a in BriefingAnswer.objects.filter(
                briefing=briefing,
                state=BriefingAnswer.STATE_ANSWERED,
            )
        )
        return required_ids.issubset(answered_ids)

    @staticmethod
    @transaction.atomic
    def pause(*, briefing: Briefing, actor) -> Briefing:
        if briefing.status != Briefing.STATUS_IN_PROGRESS:
            raise BriefingError(
                'Só é possível pausar briefings em andamento.'
            )
        briefing.status = Briefing.STATUS_PAUSED
        briefing.save(update_fields=['status', 'updated_at'])
        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_PAUSED,
            user=actor,
        )
        return briefing

    @staticmethod
    @transaction.atomic
    def resume(*, briefing: Briefing, actor) -> Briefing:
        if briefing.status != Briefing.STATUS_PAUSED:
            raise BriefingError(
                'Só é possível retomar briefings pausados.'
            )
        if briefing.has_expired():
            raise BriefingExpiredError(
                'O briefing expirou. Reabra com um novo prazo.'
            )
        briefing.status = Briefing.STATUS_IN_PROGRESS
        briefing.save(update_fields=['status', 'updated_at'])
        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_RESUMED,
            user=actor,
        )
        return briefing

    @staticmethod
    @transaction.atomic
    def close(*, briefing: Briefing, actor) -> Briefing:
        if briefing.status == Briefing.STATUS_CLOSED:
            return briefing
        briefing.status = Briefing.STATUS_CLOSED
        briefing.closed_at = timezone.now()
        briefing.save(update_fields=[
            'status', 'closed_at', 'updated_at',
        ])
        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_CLOSED,
            user=actor,
        )
        return briefing

    @staticmethod
    @transaction.atomic
    def reopen(
        *,
        briefing: Briefing,
        actor,
        new_expires_at=None,
        reason: str = '',
    ) -> Briefing:
        allowed_states = {
            Briefing.STATUS_PAUSED,
            Briefing.STATUS_CLOSED,
            Briefing.STATUS_EXPIRED,
            Briefing.STATUS_COMPLETED,
        }
        if briefing.status not in allowed_states:
            raise BriefingReopenError(
                f'Não é possível reabrir um briefing no estado '
                f'"{briefing.status}".'
            )

        needs_new_deadline = (
            briefing.status == Briefing.STATUS_EXPIRED
            or briefing.has_expired()
        )
        if needs_new_deadline and new_expires_at is None:
            raise BriefingReopenError(
                'O briefing expirou. Informe um novo prazo para reabri-lo.'
            )

        if new_expires_at is not None:
            if new_expires_at <= timezone.now():
                raise BriefingReopenError(
                    'O novo prazo precisa ser no futuro.'
                )
            briefing.expires_at = new_expires_at

        briefing.status = Briefing.STATUS_IN_PROGRESS
        briefing.closed_at = None
        briefing.completed_at = None
        briefing.save(update_fields=[
            'status', 'closed_at', 'completed_at',
            'expires_at', 'updated_at',
        ])

        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_REOPENED,
            user=actor,
            content={
                'reason': reason or '',
                'new_expires_at': (
                    new_expires_at.isoformat() if new_expires_at else None
                ),
            },
        )
        return briefing

    @staticmethod
    @transaction.atomic
    def request_clarification(
        *,
        briefing: Briefing,
        portal_session,
        question_id,
        message: str,
    ) -> BriefingAnswer:
        message = (message or '').strip()
        if not message:
            raise BriefingError('Escreva sua dúvida.')

        answer = BriefingAnswer.objects.select_for_update().filter(
            briefing=briefing, question_id=question_id,
        ).first()
        if answer is None:
            raise BriefingError(
                'Só é possível tirar dúvida sobre perguntas já respondidas.'
            )

        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_CLARIFICATION_REQUESTED,
            portal_session=portal_session,
            question_id=question_id,
            content={'message': message},
        )
        answer.state = BriefingAnswer.STATE_PENDING_CLARIFICATION
        answer.save(update_fields=['state', 'updated_at'])
        return answer

    @staticmethod
    @transaction.atomic
    def request_review(
        *,
        briefing: Briefing,
        portal_session,
        question_id,
        message: str = '',
    ) -> BriefingAnswer:
        answer = BriefingAnswer.objects.select_for_update().filter(
            briefing=briefing, question_id=question_id,
        ).first()
        if answer is None:
            raise BriefingError(
                'Só é possível solicitar revisão de perguntas respondidas.'
            )

        BriefingEventService.log(
            briefing=briefing,
            kind=BriefingEvent.KIND_REVIEW_REQUESTED,
            portal_session=portal_session,
            question_id=question_id,
            content={'message': (message or '').strip()},
        )
        answer.state = BriefingAnswer.STATE_REVIEW_REQUESTED
        answer.save(update_fields=['state', 'updated_at'])
        return answer

            )

"""
BriefingEventService — porta única de escrita no histórico.

REGRA ARQUITETURAL
------------------
Nenhum ponto do código escreve em `BriefingEvent` diretamente.
Só este service escreve.

Isso garante:
    - consistência do campo `content`
    - autoria explícita (user XOR portal_session)
    - eventos de sistema (user=None, portal_session=None) permitidos
      para ações internas (envio de e-mail, notificações), mas
      marcados como tal
"""

from typing import Optional

from apps.briefings.models import Briefing, BriefingEvent
from apps.briefings.exceptions import BriefingError


class BriefingEventService:

    @staticmethod
    def log(
        *,
        briefing: Briefing,
        kind: str,
        user=None,
        portal_session=None,
        question_id=None,
        content: Optional[dict] = None,
    ) -> BriefingEvent:
        """
        Registra um evento no histórico.

        Autoridade:
            - Se `user` preenchido → autor é o usuário do escritório.
            - Se `portal_session` preenchido → autor é o cliente.
            - Se ambos vazios → evento de sistema (ex.: envio
              automático de e-mail pelo NotificationService).
            - Se ambos preenchidos → erro (autor ambíguo).

        Validação:
            - kind precisa estar em KIND_CHOICES
            - user, se passado, precisa estar autenticado
            - portal_session, se passada, precisa estar válida
        """
        if user is not None and portal_session is not None:
            raise BriefingError(
                'Evento não pode ter dois autores '
                '(user e portal_session).'
            )

        valid_kinds = {c[0] for c in BriefingEvent.KIND_CHOICES}
        if kind not in valid_kinds:
            raise BriefingError(f'kind inválido: {kind}')

        if user is not None and not getattr(user, 'is_authenticated', False):
            user = None

        if portal_session is not None and not getattr(
            portal_session, 'is_valid', lambda: True
        )():
            raise BriefingError('Sessão do portal inválida')

        return BriefingEvent.objects.create(
            briefing=briefing,
            kind=kind,
            question_id=question_id,
            content=content or {},
            user=user,
            portal_session=portal_session,
        )

    # ------------------------------------------------------------------
    # CONSULTAS
    # ------------------------------------------------------------------

    @staticmethod
    def list_for_briefing(briefing: Briefing):
        return (
            BriefingEvent.objects
            .filter(briefing=briefing)
            .select_related('user', 'portal_session')
            .order_by('-created_at')
        )

    @staticmethod
    def latest_answer_event(briefing: Briefing, question_id):
        return (
            BriefingEvent.objects
            .filter(
                briefing=briefing,
                question_id=question_id,
                kind=BriefingEvent.KIND_ANSWERED,
            )
            .order_by('-created_at')
            .first()
        )

"""
BriefingExportService — gera PDF do briefing.

Dois modos:

    mode='architect' (padrão)
        PDF completo com dados brutos, sem notas privadas mas
        incluindo transcrições, anexos, referências e estado.

    mode='client'
        PDF resumido em formato moodboard, SEM notas internas,
        SEM info de progresso administrativo. Foco em perguntas
        e respostas, com imagens visíveis. Acompanha o e-mail de
        conclusão.

REGRA ARQUITETURAL
------------------
O PDF mostra os dados BRUTOS. Não substitui por resumos.
Não usa IA. Funciona mesmo se o provedor de IA estiver offline.
"""

import logging

from django.template.loader import render_to_string
from django.utils import timezone

from apps.briefings.services.briefing_service import BriefingService

logger = logging.getLogger('briefings')


class BriefingExportService:

    DEFAULT_TEMPLATE = 'briefings/export.html'
    CLIENT_TEMPLATE = 'briefings/export_client.html'

    @staticmethod
    def render_pdf(
        briefing,
        request=None,
        template_name: str = None,
        mode: str = 'architect',
    ) -> tuple:
        """
        Retorna (bytes, filename).

        `mode` decide o template padrão:
            'architect' → briefings/export.html
            'client'    → briefings/export_client.html

        `template_name` sobrescreve o padrão.
        """
        if template_name is None:
            template_name = (
                BriefingExportService.CLIENT_TEMPLATE
                if mode == 'client'
                else BriefingExportService.DEFAULT_TEMPLATE
            )

        context = BriefingExportService._build_context(
            briefing, request=request, mode=mode,
        )
        html = render_to_string(
            template_name,
            context,
            request=request,
        )

        try:
            from weasyprint import HTML
        except ImportError as exc:
            raise RuntimeError(
                'WeasyPrint não instalado. Rode `pip install weasyprint` '
                'e certifique-se de que as bibliotecas de sistema '
                '(pango, cairo) estão disponíveis.'
            ) from exc

        base_url = (
            request.build_absolute_uri('/') if request else None
        )
        pdf_bytes = HTML(string=html, base_url=base_url).write_pdf()

        return pdf_bytes, context['filename']

    # ------------------------------------------------------------------
    # CONTEXTO
    # ------------------------------------------------------------------

    @staticmethod
    def _build_context(briefing, request=None, mode='architect') -> dict:
        answers_by_qid = {
            str(a.question_id): a
            for a in briefing.answers.select_related(
                'latest_audio', 'latest_event',
            ).prefetch_related('attachments')
        }

        sections_payload = []
        ordered_sections = sorted(
            briefing.structure_snapshot.get('sections', []),
            key=lambda s: s.get('position', 0),
        )
        for section in ordered_sections:
            questions_payload = []
            ordered_questions = sorted(
                section.get('questions', []),
                key=lambda q: q.get('position', 0),
            )
            for q in ordered_questions:
                qid = str(q['id'])
                answer = answers_by_qid.get(qid)

                refs = []
                for doc_id in (q.get('reference_document_ids') or []):
                    refs.append({'id': doc_id, 'url': ''})

                latest_content = ''
                latest_transcription = ''
                transcription_status = ''
                attachments = []
                audio_url = ''
                updated_at = None
                if answer is not None:
                    latest_content = answer.latest_content or ''
                    latest_transcription = answer.latest_transcription or ''
                    transcription_status = (
                        answer.get_transcription_status_display()
                    )
                    updated_at = answer.updated_at
                    for doc in answer.attachments.all():
                        attachments.append({
                            'id': str(doc.id),
                            'name': doc.full_name,
                            'url': BriefingExportService._doc_url(
                                doc, request,
                            ),
                        })
                    if answer.latest_audio:
                        audio_url = BriefingExportService._doc_url(
                            answer.latest_audio, request,
                        )

                questions_payload.append({
                    'id': qid,
                    'text': q.get('text', ''),
                    'instructions': q.get('instructions', ''),
                    'is_required': bool(q.get('is_required', True)),
                    'reference_documents': refs,
                    'state': answer.state if answer else 'pending',
                    'latest_content': latest_content,
                    'latest_transcription': latest_transcription,
                    'transcription_status': transcription_status,
                    'audio_url': audio_url,
                    'attachments': attachments,
                    'updated_at': updated_at,
                })
            sections_payload.append({
                'id': section.get('id'),
                'title': section.get('title', ''),
                'description': section.get('description', ''),
                'questions': questions_payload,
            })

        progress = BriefingService.compute_progress(briefing)

        now = timezone.now()
        slug = ''.join(
            c if c.isalnum() else '-'
            for c in (briefing.title or 'briefing').lower()
        ).strip('-') or 'briefing'
        filename = f"briefing-{slug}-{now:%Y%m%d}.pdf"

        return {
            'briefing': briefing,
            'client_name': briefing.client.name if briefing.client else '',
            'workspace_name': (
                briefing.workspace.name if briefing.workspace else ''
            ),
            'sections': sections_payload,
            'progress': progress,
            'generated_at': now,
            'filename': filename,
            'mode': mode,
        }

    @staticmethod
    def _doc_url(doc, request=None) -> str:
        try:
            if doc.file:
                return doc.file.url
        except Exception:
            pass
        if request is not None:
            return request.build_absolute_uri(
                f'/api/documents/{doc.id}/preview/'
            )
        return f'/api/documents/{doc.id}/preview/'

"""
NotificationService — envio automático de e-mail ao concluir briefing.

REGRA ARQUITETURAL
------------------
Duas garantias:

1. IDEMPOTÊNCIA: o envio de conclusão é disparado uma e apenas
   uma vez por briefing. Um `BriefingEvent` com
   `kind='completion_email_sent'` marca que o envio já ocorreu.
   Se a função rodar de novo, ela verifica e sai.

2. NÃO BLOQUEIA: o envio é agendado via `transaction.on_commit`,
   nunca dentro da transação de conclusão do briefing. Se o
   provedor de e-mail estiver fora, o briefing continua concluído.

Falha de envio NÃO grava o evento de idempotência. Isso permite
retentativa (manual ou por cron futuro).
"""

import logging

from apps.briefings.models import Briefing, BriefingEvent
from apps.briefings.services.event_service import BriefingEventService

logger = logging.getLogger('briefings')


class NotificationService:

    KIND_MARKER = BriefingEvent.KIND_COMPLETION_EMAIL_SENT

    # ==================================================================
    # API
    # ==================================================================

    @staticmethod
    def on_briefing_completed(briefing_id) -> bool:
        """
        Chamado quando um briefing vira `completed`.

        Retorna `True` se um e-mail foi disparado nesta chamada,
        `False` se já havia sido disparado antes ou se a situação
        não permite (ex.: briefing não existe).

        Nunca levanta.
        """
        try:
            briefing = Briefing.objects.select_related(
                'client', 'workspace', 'template', 'created_by',
            ).get(id=briefing_id)
        except Briefing.DoesNotExist:
            logger.warning(
                'Briefing %s não existe em on_briefing_completed',
                briefing_id,
            )
            return False

        if briefing.status != Briefing.STATUS_COMPLETED:
            logger.info(
                'Briefing %s não está completed, ignorando notificação',
                briefing_id,
            )
            return False

        if NotificationService._already_notified(briefing):
            logger.info(
                'Briefing %s já teve e-mail de conclusão enviado, '
                'ignorando', briefing_id,
            )
            return False

        try:
            NotificationService._send(briefing)
        except Exception:
            logger.exception(
                'Falha ao enviar e-mail de conclusão '
                '(briefing=%s). Evento de idempotência NÃO foi '
                'gravado — retentativa é possível.',
                briefing_id,
            )
            return False

        # Só marca como enviado após sucesso
        NotificationService._mark_notified(briefing)
        logger.info(
            'E-mail de conclusão enviado (briefing=%s)', briefing_id,
        )
        return True

    # ==================================================================
    # INTERNO
    # ==================================================================

    @staticmethod
    def _already_notified(briefing: Briefing) -> bool:
        return BriefingEvent.objects.filter(
            briefing=briefing,
            kind=NotificationService.KIND_MARKER,
        ).exists()

    @staticmethod
    def _mark_notified(briefing: Briefing) -> None:
        BriefingEventService.log(
            briefing=briefing,
            kind=NotificationService.KIND_MARKER,
            user=None,
            portal_session=None,
            content={
                'architect_email': briefing.created_by.email
                    if briefing.created_by else None,
                'client_email': briefing.client.email or None,
            },
        )

    @staticmethod
    def _send(briefing: Briefing) -> None:
        from apps.accounts.services.email import EmailService

        architect_email = (
            briefing.created_by.email
            if briefing.created_by else None
        )

        # Gera PDF moodboard do cliente (sem notas internas)
        client_pdf_bytes = None
        client_pdf_filename = None
        try:
            from apps.briefings.services.export_service import (
                BriefingExportService,
            )
            client_pdf_bytes, client_pdf_filename = (
                BriefingExportService.render_pdf(
                    briefing,
                    template_name='briefings/export_client.html',
                    mode='client',
                )
            )
        except Exception:
            logger.exception(
                'Falha ao gerar PDF do cliente para envio de '
                'conclusão (briefing=%s). Vai enviar sem anexo.',
                briefing.id,
            )

        client_portal_url = NotificationService._portal_url(briefing)

        if architect_email:
            EmailService.send_briefing_completed(
                briefing=briefing,
                client_portal_url=client_portal_url,
                architect_email=architect_email,
                client_pdf_bytes=client_pdf_bytes,
                client_pdf_filename=client_pdf_filename,
            )
        else:
            # Sem e-mail do arquiteto: ainda envia para o cliente
            if briefing.client.email:
                from django.conf import settings
                site_name = getattr(settings, 'SITE_NAME', 'Clivo')
                EmailService.send(
                    to=briefing.client.email,
                    subject=(
                        f'Seu briefing foi registrado — {briefing.title}'
                    ),
                    text=(
                        f'Olá,\n\n'
                        f'Recebemos suas respostas do briefing '
                        f'"{briefing.title}".\n\n'
                        f'— {site_name}'
                    ),
                )

    @staticmethod
    def _portal_url(briefing: Briefing) -> str:
        """
        Link autenticado do portal do cliente para este briefing.
        """
        from django.conf import settings
        frontend_url = getattr(
            settings, 'FRONTEND_URL', ''
        ).rstrip('/')
        if not frontend_url:
            return ''
        return f'{frontend_url}/'

"""
SendService — envio manual do briefing pelo arquiteto.

Diferente do NotificationService (automático, apenas em conclusão),
este é acionado pelo arquiteto a qualquer momento:

    - briefing concluído ou não
    - reenvio após ajuste
    - envio para e-mail alternativo

Registra SEMPRE um BriefingEvent `manual_send` com quem enviou,
para quem e quando. Isso torna o reenvio auditável.

REGRA ARQUITETURAL
------------------
- Não gera PDF em modo arquiteto. Sempre em modo cliente,
  sem notas internas.
- Se o envio falhar, o EmailService levanta e o evento NÃO é
  gravado. Isso evita estado inconsistente (marcar como enviado
  um e-mail que nunca saiu).
- Não bloqueia a conclusão do briefing. É uma ação explícita do
  arquiteto, disparada por ele.
"""

import logging

from apps.briefings.models import Briefing
from apps.briefings.services.event_service import BriefingEventService

logger = logging.getLogger('briefings')


class BriefingSendService:

    KIND_MANUAL_SEND = 'manual_send'

    @staticmethod
    def send_to_client(
        *,
        briefing: Briefing,
        actor,
        to_email: str = None,
        message: str = '',
        request=None,
    ) -> dict:
        """
        Envia (ou reenvia) o briefing para o cliente.

        Args:
            briefing: execução
            actor: usuário que está enviando (arquiteto/equipe)
            to_email: destinatário. Se None, usa `briefing.client.email`.
            message: texto opcional a ser incluído no corpo
            request: opcional, para correlation_id e detecção de host

        Returns:
            {
                'sent': bool,
                'to': str,
                'event_id': str,
            }

        Levanta BriefingError se não houver e-mail de destino.

        Levanta outras exceções propagadas do EmailService quando o
        envio falhar (smtp down, credencial inválida, etc.).
        """
        from apps.accounts.services.email import EmailService
        from apps.briefings.services.export_service import (
            BriefingExportService,
        )
        from apps.briefings.exceptions import BriefingError

        # ------------------------------------------------------------------
        # 1. Resolve o destinatário
        # ------------------------------------------------------------------

        target_email = (to_email or '').strip() or (
            briefing.client.email or ''
        )
        if not target_email:
            raise BriefingError(
                'Nenhum e-mail de destino. '
                'Informe um endereço ou cadastre um no cliente.'
            )

        # ------------------------------------------------------------------
        # 2. Gera PDF em modo cliente (sem notas internas)
        #
        # Falha na geração NÃO bloqueia o envio. O e-mail ainda é
        # enviado — apenas sem anexo. O evento registra has_pdf=False.
        # ------------------------------------------------------------------

        pdf_bytes = None
        pdf_filename = None
        try:
            pdf_bytes, pdf_filename = BriefingExportService.render_pdf(
                briefing,
                template_name='briefings/export_client.html',
                mode='client',
            )
        except Exception:
            logger.exception(
                'Falha ao gerar PDF para envio manual (briefing=%s). '
                'O e-mail será enviado sem anexo.',
                briefing.id,
            )

        # ------------------------------------------------------------------
        # 3. Resolve URL do portal do cliente
        # ------------------------------------------------------------------

        portal_url = ''
        try:
            from django.conf import settings
            base = getattr(settings, 'FRONTEND_URL', '').rstrip('/')
            if base:
                portal_url = f'{base}/'
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 4. Envia o e-mail
        #
        # Se o envio falhar, a exceção sobe. O evento NÃO é gravado.
        # Isso evita marcar como enviado algo que não saiu.
        # ------------------------------------------------------------------

        EmailService.send_briefing_to_client(
            briefing=briefing,
            to_email=target_email,
            pdf_bytes=pdf_bytes,
            pdf_filename=pdf_filename,
            message=message,
            client_portal_url=portal_url,
        )

        # ------------------------------------------------------------------
        # 5. Registra o evento (append-only) — só após sucesso do envio
        # ------------------------------------------------------------------

        event = BriefingEventService.log(
            briefing=briefing,
            kind=BriefingSendService.KIND_MANUAL_SEND,
            user=actor,
            content={
                'to': target_email,
                'by': actor.email if actor else None,
                'has_pdf': pdf_bytes is not None,
                'message': message or None,
            },
        )

        logger.info(
            'Briefing %s enviado manualmente para %s por %s',
            briefing.id,
            target_email,
            actor.email if actor else '?',
        )

        return {
            'sent': True,
            'to': target_email,
            'event_id': str(event.id),
        }

"""
BriefingTemplateService — ciclo de vida do template reutilizável.

Emite `BriefingTemplateEvent` para toda alteração estrutural,
garantindo auditoria de "quem alterou o quê, quando".
"""

import logging

from django.db import transaction

from apps.briefings.models import (
    BriefingTemplate,
    BriefingSection,
    BriefingQuestion,
    BriefingTemplateEvent,
)
from apps.briefings.exceptions import (
    BriefingTemplateNotFoundError,
    BriefingTemplateNotDraftError,
    BriefingTemplateInactiveError,
    BriefingSectionNotFoundError,
    BriefingQuestionNotFoundError,
    BriefingError,
)

logger = logging.getLogger('briefings')


class BriefingTemplateService:

    # ==================================================================
    # EVENTOS
    # ==================================================================

    @staticmethod
    def _log_event(
        *, template, kind, user=None,
        section_id=None, question_id=None, content=None,
    ):
        BriefingTemplateEvent.objects.create(
            template=template,
            kind=kind,
            section_id=section_id,
            question_id=question_id,
            content=content or {},
            user=user if (user and getattr(user, 'is_authenticated', False)) else None,
        )

    # ==================================================================
    # TEMPLATE
    # ==================================================================

    @staticmethod
    @transaction.atomic
    def create_template(
        *, workspace, name: str, description: str = '',
        category: str = '', actor=None,
    ) -> BriefingTemplate:
        name = (name or '').strip()
        if not name:
            raise BriefingError('O nome do template é obrigatório.')

        template = BriefingTemplate.objects.create(
            workspace=workspace,
            name=name,
            description=(description or '').strip(),
            category=(category or '').strip(),
            status=BriefingTemplate.STATUS_DRAFT,
            is_active=True,
            created_by=actor,
        )
        BriefingTemplateService._log_event(
            template=template,
            kind=BriefingTemplateEvent.KIND_TEMPLATE_CREATED,
            user=actor,
            content={'name': template.name},
        )
        return template

    @staticmethod
    @transaction.atomic
    def update_template(
        *, template: BriefingTemplate,
        name: str = None, description: str = None, category: str = None,
        actor=None,
    ) -> BriefingTemplate:
        fields = []
        changes = {}

        if name is not None:
            name = name.strip()
            if not name:
                raise BriefingError('O nome do template não pode ser vazio.')
            if template.name != name:
                changes['name'] = {'old': template.name, 'new': name}
                template.name = name
                fields.append('name')

        if description is not None:
            new_desc = description.strip()
            if template.description != new_desc:
                changes['description'] = True
                template.description = new_desc
                fields.append('description')

        if category is not None:
            new_cat = category.strip()
            if template.category != new_cat:
                changes['category'] = {
                    'old': template.category, 'new': new_cat,
                }
                template.category = new_cat
                fields.append('category')

        if fields:
            template.save(update_fields=fields + ['updated_at'])
            BriefingTemplateService._log_event(
                template=template,
                kind=BriefingTemplateEvent.KIND_TEMPLATE_UPDATED,
                user=actor,
                content={'changes': changes},
            )
        return template

    @staticmethod
    @transaction.atomic
    def set_cover(*, template: BriefingTemplate, uploaded_file, actor=None):
        if uploaded_file is None:
            raise BriefingError('Nenhuma imagem enviada.')

        name = getattr(uploaded_file, 'name', '') or ''
        ext = name.rsplit('.', 1)[-1].lower() if '.' in name else ''
        if ext not in {'jpg', 'jpeg', 'jfif', 'png', 'webp', 'gif'}:
            raise BriefingError(
                'Formato inválido. Envie JPG, PNG, WEBP ou GIF.'
            )

        old_name = None
        old_storage = None
        if template.cover:
            try:
                old_name = template.cover.name
                old_storage = template.cover.storage
            except Exception:
                pass

        try:
            template.cover.save(name, uploaded_file, save=False)
        except Exception as exc:
            logger.exception('Falha ao salvar capa do template')
            raise BriefingError('Falha ao salvar a capa no storage.') from exc

        template.save(update_fields=['cover', 'updated_at'])

        if old_name and old_storage:
            try:
                old_storage.delete(old_name)
            except Exception:
                logger.warning(
                    'Não foi possível remover a capa antiga: %s', old_name,
                )

        BriefingTemplateService._log_event(
            template=template,
            kind=BriefingTemplateEvent.KIND_TEMPLATE_UPDATED,
            user=actor,
            content={'action': 'cover_changed'},
        )
        return template

    @staticmethod
    @transaction.atomic
    def clear_cover(*, template: BriefingTemplate, actor=None):
        if not template.cover:
            return template
        old_name = template.cover.name
        old_storage = template.cover.storage
        template.cover = None
        template.save(update_fields=['cover', 'updated_at'])
        try:
            old_storage.delete(old_name)
        except Exception:
            logger.warning('Não foi possível remover a capa: %s', old_name)
        BriefingTemplateService._log_event(
            template=template,
            kind=BriefingTemplateEvent.KIND_TEMPLATE_UPDATED,
            user=actor,
            content={'action': 'cover_removed'},
        )
        return template

    @staticmethod
    @transaction.atomic
    def delete_template(*, template: BriefingTemplate, actor=None) -> None:
        if template.executions.exists():
            raise BriefingError(
                'Não é possível excluir um template que já possui '
                'execuções. Arquive-o em vez disso.'
            )
        # DELETE real. Eventos de template caem em cascata (FK CASCADE).
        if template.cover:
            try:
                template.cover.delete(save=False)
            except Exception:
                logger.warning('Falha ao remover capa ao excluir template')
        template.delete()

    @staticmethod
    @transaction.atomic
    def publish_template(
        *, template: BriefingTemplate, actor=None,
    ) -> BriefingTemplate:
        if template.status != BriefingTemplate.STATUS_DRAFT:
            raise BriefingTemplateNotDraftError(
                'Só templates em rascunho podem ser publicados.'
            )
        if not template.sections.exists():
            raise BriefingError('Não é possível publicar sem sessões.')
        template.status = BriefingTemplate.STATUS_PUBLISHED
        template.save(update_fields=['status', 'updated_at'])
        BriefingTemplateService._log_event(
            template=template,
            kind=BriefingTemplateEvent.KIND_TEMPLATE_PUBLISHED,
            user=actor,
        )
        return template

    @staticmethod
    @transaction.atomic
    def archive_template(
        *, template: BriefingTemplate, actor=None,
    ) -> BriefingTemplate:
        template.status = BriefingTemplate.STATUS_ARCHIVED
        template.save(update_fields=['status', 'updated_at'])
        BriefingTemplateService._log_event(
            template=template,

"""
TranscriptionService — transcrição de áudio.

REGRA ARQUITETURAL
------------------
Transcrição é OPCIONAL. Se o provedor não estiver configurado, o
status fica `failed` com mensagem clara. NUNCA simulamos uma
transcrição.

A transcrição NÃO substitui o áudio original. O áudio permanece
acessível em `BriefingAnswer.latest_audio`.
"""

import logging

from django.conf import settings
from django.utils import timezone

from apps.briefings.models import BriefingAnswer, BriefingEvent
from apps.briefings.exceptions import TranscriptionError
from apps.briefings.services.event_service import BriefingEventService

logger = logging.getLogger('briefings')


class TranscriptionService:

    # ==================================================================
    # DISPONIBILIDADE
    # ==================================================================

    @staticmethod
    def is_available() -> bool:
        """Verifica se o provedor está configurado no ambiente."""
        api_key = getattr(settings, 'OPENAI_API_KEY', None)
        return bool(api_key)

    @staticmethod
    def is_required() -> bool:
        """Se `True`, a ausência do provedor é erro."""
        return bool(getattr(settings, 'BRIEFINGS_REQUIRE_AI', False))

    # ==================================================================
    # SINCRONO
    # ==================================================================

    @staticmethod
    def transcribe_answer(answer_id) -> None:
        """
        Transcreve o áudio vinculado a uma resposta.

        Idempotente: se já houver transcrição concluída, não faz nada.
        """
        try:
            answer = BriefingAnswer.objects.select_related(
                'latest_audio', 'briefing',
            ).get(id=answer_id)
        except BriefingAnswer.DoesNotExist:
            logger.warning('Answer %s não encontrado', answer_id)
            return

        if answer.latest_audio is None:
            answer.transcription_status = (
                BriefingAnswer.TRANSCRIPTION_NOT_APPLICABLE
            )
            answer.save(update_fields=['transcription_status', 'updated_at'])
            return

        if answer.transcription_status == BriefingAnswer.TRANSCRIPTION_COMPLETED:
            return

        if not TranscriptionService.is_available():
            answer.transcription_status = BriefingAnswer.TRANSCRIPTION_FAILED
            answer.latest_transcription = ''
            answer.save(update_fields=[
                'transcription_status', 'latest_transcription', 'updated_at',
            ])
            logger.warning(
                'Transcrição indisponível (answer=%s). '
                'Configure OPENAI_API_KEY para habilitar.', answer_id,
            )
            if TranscriptionService.is_required():
                raise TranscriptionError(
                    'Provedor de transcrição não configurado.'
                )
            return

        answer.transcription_status = BriefingAnswer.TRANSCRIPTION_PROCESSING
        answer.save(update_fields=['transcription_status', 'updated_at'])

        try:
            text = TranscriptionService._call_provider(answer.latest_audio)
        except Exception as exc:
            logger.exception('Falha na transcrição (answer=%s)', answer_id)
            answer.transcription_status = BriefingAnswer.TRANSCRIPTION_FAILED
            answer.latest_transcription = ''
            answer.save(update_fields=[
                'transcription_status', 'latest_transcription', 'updated_at',
            ])
            raise TranscriptionError(str(exc)) from exc

        answer.latest_transcription = text
        answer.transcription_status = BriefingAnswer.TRANSCRIPTION_COMPLETED
        answer.save(update_fields=[
            'latest_transcription', 'transcription_status', 'updated_at',
        ])

        BriefingEventService.log(
            briefing=answer.briefing,
            kind=BriefingEvent.KIND_TRANSCRIPTION_GENERATED,
            user=None,
            portal_session=None,
            question_id=answer.question_id,
            content={'status': 'completed'},
        )

    # ==================================================================
    # PROVIDER
    # ==================================================================

    @staticmethod
    def _call_provider(audio_document) -> str:
        """
        Chama o provedor real. Não simula.

        Usa a biblioteca `openai` (Whisper). Se a lib não estiver
        instalada, levanta TranscriptionError.
        """
        try:
            from openai import OpenAI
        except ImportError as exc:
            raise TranscriptionError(
                'Biblioteca openai não instalada.'
            ) from exc

        api_key = getattr(settings, 'OPENAI_API_KEY', None)
        model = getattr(settings, 'AI_TRANSCRIPTION_MODEL', 'whisper-1')

        client = OpenAI(api_key=api_key)

        file_handle = audio_document.file.open('rb')
        try:
            result = client.audio.transcriptions.create(
                model=model,
                file=file_handle,
                response_format='text',
            )
        finally:
            try:
                file_handle.close()
            except Exception:
                pass

        if isinstance(result, str):
            return result.strip()
        text = getattr(result, 'text', None)
        return (text or '').strip()

    # ==================================================================
    # ASSÍNCRONO
    # ==================================================================

    @staticmethod
    def run_async(answer_id) -> None:
        """
        Publica a tarefa para execução assíncrona.

        Se Celery não estiver configurado, executa inline como
        fallback — não perde o evento.
        """
        try:
            from config.celery import app as celery_app  # noqa: F401
            from apps.briefings.tasks import transcribe_answer_task

            transcribe_answer_task.delay(str(answer_id))
            return
        except Exception:
            logger.info(
                'Celery indisponível. Executando transcrição inline '
                '(answer=%s).', answer_id,
            )

        try:
            TranscriptionService.transcribe_answer(answer_id)
        except Exception:
            logger.exception(
                'Falha na transcrição inline (answer=%s)', answer_id,
            )

from apps.documents.models import Document
from apps.briefings.models import BriefingAnswer
from apps.briefings.services import BriefingService


@pytest.mark.django_db
class TestAudioPreservation:

    def _make_audio_doc(self, workspace, user, idx=1):
        # Cria um Document "fake" — não precisa de arquivo real
        # para o teste de PROTECT.
        return Document.objects.create(
            workspace=workspace,
            name=f'audio-{idx}',
            file=f'documents/audio-{idx}.webm',
            extension='webm',
            mime_type='audio/webm',
            size=1024,
            created_by=user,
        )

    def test_document_with_briefing_answer_cannot_be_deleted(
        self, briefing_a, portal_session_a, workspace_a, user_owner,
    ):
        audio = self._make_audio_doc(workspace_a, user_owner)
        q = briefing_a.snapshot_questions()[0]

        answer = BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=q['id'],
            content_text='',
            audio_document=audio,
        )
        assert answer.latest_audio_id == audio.id

        with pytest.raises(ProtectedError):
            audio.delete()

    def test_transcription_status_independent_of_state(
        self, briefing_a, portal_session_a, workspace_a, user_owner,
    ):
        audio = self._make_audio_doc(workspace_a, user_owner, idx=2)
        q = briefing_a.snapshot_questions()[0]

        answer = BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=q['id'],
            content_text='',
            audio_document=audio,
        )
        assert answer.state == BriefingAnswer.STATE_ANSWERED
        assert answer.transcription_status in (
            BriefingAnswer.TRANSCRIPTION_PENDING,
            BriefingAnswer.TRANSCRIPTION_FAILED,
            BriefingAnswer.TRANSCRIPTION_COMPLETED,
        )
        # O estado da resposta NÃO é alterado por falha de transcrição
        assert answer.state == BriefingAnswer.STATE_ANSWERED

    def test_deleting_briefing_cascades(
        self, briefing_a, portal_session_a, workspace_a, user_owner,
    ):
        """
        Excluir o briefing inteiro remove answers/events/notes em
        cascata. Os Documents de áudio permanecem (PROTECT só
        bloqueia exclusão DIRETA do Document, não a cascata do
        Briefing).
        """
        audio = self._make_audio_doc(workspace_a, user_owner, idx=3)
        q = briefing_a.snapshot_questions()[0]
        BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=q['id'],
            content_text='',
            audio_document=audio,
        )

        briefing_a.delete()

        # Document permanece — a exclusão do briefing só removeu
        # a FK que apontava para ele.
        audio.refresh_from_db()
        assert audio.pk is not None

"""
Testes de concorrência no submit_answer.

Estratégia em duas camadas:

1. Teste de CONTRATO (`test_submit_answer_uses_select_for_update`)
   Verifica que o serviço usa `select_for_update` no Briefing,
   garantindo que o lock existe. Roda em qualquer banco — inclusive
   SQLite.

2. Teste REAL de concorrência (`test_simultaneous_answers_create_single_event`)
   Duas threads chamam `submit_answer` ao mesmo tempo. Só roda em
   PostgreSQL (SQLite em memória não compartilha o banco entre
   conexões de threads). Em CI com Postgres, roda. Localmente em
   SQLite, é pulado.
"""

import threading

import pytest
from django.db import connection
from django.test import TransactionTestCase

from apps.briefings.models import BriefingEvent
from apps.briefings.services import BriefingService


# =========================================================================

@pytest.mark.django_db
def test_submit_answer_uses_select_for_update(
    briefing_a, portal_session_a,
):
    """
    Verifica que `submit_answer` usa `select_for_update` no Briefing.

    Não substitui um teste real de concorrência, mas documenta o
    contrato: o lock existe e é chamado no caminho crítico. Se
    alguém remover o `select_for_update` no futuro, este teste
    quebra e a intenção fica registrada.
    """
    from unittest.mock import patch
    from django.db.models import QuerySet

    real_select = QuerySet.select_for_update
    called = {'count': 0}

    def spy_select_for_update(self_qs, *args, **kwargs):
        called['count'] += 1
        return real_select(self_qs, *args, **kwargs)

    questions = briefing_a.snapshot_questions()
    first = questions[0]

    with patch.object(
        QuerySet, 'select_for_update', spy_select_for_update,
    ):
        BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=first['id'],
            content_text='Resposta.',
        )

    # Pelo menos uma chamada — a do Briefing.
    assert called['count'] >= 1, (
        'submit_answer não chamou select_for_update. '
        'O lock de concorrência pode ter sido removido.'
    )

def _running_on_postgres() -> bool:
    """Detecta se o banco atual é Postgres."""
    try:
        return connection.vendor == 'postgresql'
    except Exception:
        return False


@pytest.mark.skipif(
    not _running_on_postgres(),
    reason=(
        'Teste real de concorrência requer Postgres. '
        'SQLite em memória não compartilha o banco entre threads. '
        'Rode no CI com Postgres habilitado.'
    ),
)
class TestConcurrencyReal(TransactionTestCase):
    """
    Duas threads chamam `submit_answer` na mesma pergunta.

    Com `select_for_update` correto, uma vence, a outra recebe
    erro de sequência. Apenas UM `BriefingEvent` do tipo `answered`
    é persistido.
    """

    reset_sequences = False

    def setUp(self):
        from decimal import Decimal
        from django.contrib.auth import get_user_model
        from django.utils import timezone
        from apps.console.models import (
            ConsoleGovernance, Plan,
        )
        from apps.workspaces.models import (
            Workspace, WorkspaceMembership, WorkspaceSubscription,
        )
        from apps.clients.models import Client, ClientPortalSession
        from apps.briefings.models import (
            BriefingTemplate, BriefingSection, BriefingQuestion,
        )
        from apps.briefings.services import BriefingService

        User = get_user_model()

        self.user = User.objects.create_user(
            email='conc@briefings.test', password='x',
        )
        self.gov = ConsoleGovernance.objects.create(
            key='proprietario', name='Proprietário', description='',
            scope='workspace', is_system=True, is_protected=True,
        )
        self.plan = Plan.objects.create(
            key='basic', name='Basic', price=Decimal('0'),
            is_active=True, is_default=True,
        )
        self.ws = Workspace.objects.create(name='Conc', slug='conc')
        WorkspaceSubscription.objects.create(
            workspace=self.ws, plan=self.plan, is_active=True,
            started_at=timezone.now(),
        )
        WorkspaceMembership.objects.create(
            workspace=self.ws, user=self.user, governance=self.gov,
        )
        self.client = Client.objects.create(
            workspace=self.ws, name='Cliente Conc',
            email='cli@conc.test', public_code='CLI-CONC01',
        )
        self.portal_session, _ = ClientPortalSession.issue(
            client=self.client,
        )

        tpl = BriefingTemplate.objects.create(
            workspace=self.ws, name='T', status='published',
            is_active=True, created_by=self.user,
        )
        s = BriefingSection.objects.create(
            template=tpl, title='S', position=1,
        )
        BriefingQuestion.objects.create(
            section=s, text='Q1', position=1, is_required=True,
            allowed_response_formats=['text'],
        )
        self.briefing = BriefingService.create_briefing(
            template=tpl, client=self.client,
            workspace=self.ws, actor=self.user,
        )
        self.question_id = self.briefing.snapshot_questions()[0]['id']

    def tearDown(self):
        connection.close()

    def test_simultaneous_answers_create_single_event(self):
        errors = []
        successes = []
        lock = threading.Lock()

        def worker():
            try:
                BriefingService.submit_answer(
                    briefing_id=self.briefing.id,
                    portal_session=self.portal_session,
                    question_id=self.question_id,
                    content_text='Resposta concorrente.',
                )
                with lock:
                    successes.append(True)
            except Exception as exc:
                with lock:
                    errors.append(exc)
            finally:
                connection.close()

        t1 = threading.Thread(target=worker)
        t2 = threading.Thread(target=worker)
        t1.start()
        t2.start()
        t1.join()
        t2.join()

        count = BriefingEvent.objects.filter(
            briefing=self.briefing,
            kind=BriefingEvent.KIND_ANSWERED,
            question_id=self.question_id,
        ).count()
        assert count == 1, (
            f'Esperado 1 evento answered, obtido {count}. '
            f'Sucessos={len(successes)} erros={len(errors)}.'
        )

        assert len(successes) >= 1, (
            'Nenhuma das duas requisições foi bem-sucedida.'
        )

"""
Testes do NotificationService — envio automático em conclusão.

Cobrem:
    - Envio único por briefing (idempotência)
    - Segundo chamado NÃO dispara novo envio
    - Falha de e-mail NÃO marca como enviado
    - Status diferente de `completed` não dispara
    - Briefing inexistente não levanta
    - submit_answer AGENDA a notificação via on_commit

Estratégia de teste:
    Chamamos `NotificationService.on_briefing_completed()` DIRETAMENTE
    em vez de depender do on_commit do `submit_answer`. Motivo: dentro
    do `@pytest.mark.django_db` cada teste roda numa transação envolvida
    que faz rollback — e o `on_commit` do Django não dispara nesse
    contexto. O contrato "on_commit agenda" é verificado separadamente
    (test_completion_schedules_on_commit).
"""

import pytest
from unittest.mock import patch
from uuid import uuid4

from apps.briefings.models import Briefing, BriefingEvent
from apps.briefings.services import (
    BriefingService,
    NotificationService,
)
from apps.accounts.services.email import EmailService


def _complete_briefing(briefing, portal_session):
    """Leva um briefing a `completed` respondendo todas as perguntas."""
    questions = briefing.snapshot_questions()
    for q in questions:
        BriefingService.submit_answer(
            briefing_id=briefing.id,
            portal_session=portal_session,
            question_id=q['id'],
            content_text='ok',
        )


@pytest.mark.django_db
class TestCompletionNotification:

    # ==================================================================
    # IDEMPOTÊNCIA
    # ==================================================================

    def test_notification_runs_once_on_completed_briefing(
        self, briefing_a, portal_session_a,
    ):
        """
        Primeira chamada envia; segunda NÃO envia.

        Checa o marcador `completion_email_sent` e a contagem de
        chamadas ao EmailService.
        """
        # Leva ao estado completed
        _complete_briefing(briefing_a, portal_session_a)
        briefing_a.refresh_from_db()
        assert briefing_a.status == Briefing.STATUS_COMPLETED

        with patch.object(
            EmailService, 'send_briefing_completed', return_value=1,
        ) as mock_send:
            first = NotificationService.on_briefing_completed(
                briefing_a.id,
            )
            second = NotificationService.on_briefing_completed(
                briefing_a.id,
            )

        # Primeira faz o envio
        assert first is True
        # Segunda é idempotente e não envia
        assert second is False

        # Uma única chamada real ao EmailService
        assert mock_send.call_count == 1

        # Um único evento marcador
        markers = BriefingEvent.objects.filter(
            briefing=briefing_a,
            kind=BriefingEvent.KIND_COMPLETION_EMAIL_SENT,
        )
        assert markers.count() == 1

    def test_notification_not_sent_when_not_completed(self, briefing_a):
        """
        Briefing ainda `in_progress` NÃO dispara notificação.
        """
        with patch.object(
            EmailService, 'send_briefing_completed', return_value=1,
        ) as mock_send:
            result = NotificationService.on_briefing_completed(
                briefing_a.id,
            )

        assert result is False
        mock_send.assert_not_called()

    def test_notification_no_exception_for_missing_briefing(self):
        """
        Briefing inexistente retorna False sem levantar.
        """
        result = NotificationService.on_briefing_completed(uuid4())
        assert result is False

    # ==================================================================
    # FALHA DE ENVIO
    # ==================================================================

    def test_failed_send_does_not_mark_as_sent(
        self, briefing_a, portal_session_a,
    ):
        """
        Se o envio falha, o evento marcador NÃO é gravado — retentativa
        é possível.
        """
        _complete_briefing(briefing_a, portal_session_a)
        briefing_a.refresh_from_db()

        # 1ª tentativa — falha
        with patch.object(
            EmailService, 'send_briefing_completed',
            side_effect=Exception('smtp down'),
        ):
            result = NotificationService.on_briefing_completed(
                briefing_a.id,
            )
        assert result is False

        # Nenhum marcador
        assert BriefingEvent.objects.filter(
            briefing=briefing_a,
            kind=BriefingEvent.KIND_COMPLETION_EMAIL_SENT,
        ).count() == 0

        # 2ª tentativa — sucesso
        with patch.object(
            EmailService, 'send_briefing_completed', return_value=1,
        ):
            result = NotificationService.on_briefing_completed(
                briefing_a.id,
            )
        assert result is True

        assert BriefingEvent.objects.filter(
            briefing=briefing_a,
            kind=BriefingEvent.KIND_COMPLETION_EMAIL_SENT,
        ).count() == 1

    # ==================================================================
    # AGENDAMENTO VIA on_commit
    # ==================================================================

    def test_completion_schedules_on_commit(
        self, briefing_a, portal_session_a,
    ):
        """
        Verifica que `submit_answer` AGENDA um on_commit quando o
        briefing passa a `completed`. Não executamos o callback —
        apenas checamos que foi agendado.

        Usa `django.test.TestCase.captureOnCommitCallbacks` via
        `django.test.utils`, disponível no pytest-django.
        """
        from django.test import TestCase

        # Envolvemos a chamada num capture de callbacks do Django.
        # `captureOnCommitCallbacks` exige uma instância de TestCase.
        class _Capture(TestCase):
            def runTest(self):
                pass

        capture = _Capture()

        with capture.captureOnCommitCallbacks(execute=False) as callbacks:
            _complete_briefing(briefing_a, portal_session_a)

        # Pelo menos um callback de on_commit foi agendado
        # (idealmente o de notificação; pode haver também o de
        # transcrição, mas não há áudio neste teste).
        assert len(callbacks) >= 1

"""
Testes de prazo e reabertura.
"""

from datetime import timedelta

import pytest
from django.utils import timezone

from apps.briefings.models import Briefing, BriefingEvent
from apps.briefings.services import BriefingService
from apps.briefings.exceptions import BriefingReopenError


@pytest.mark.django_db
class TestReopen:

    def test_reopen_expired_without_new_deadline_fails(
        self, briefing_a, user_owner,
    ):
        # Força estado expirado
        briefing_a.status = Briefing.STATUS_EXPIRED
        briefing_a.save(update_fields=['status'])

        with pytest.raises(BriefingReopenError):
            BriefingService.reopen(
                briefing=briefing_a, actor=user_owner,
            )

    def test_reopen_expired_with_past_deadline_fails(
        self, briefing_a, user_owner,
    ):
        briefing_a.status = Briefing.STATUS_EXPIRED
        briefing_a.save(update_fields=['status'])

        past = timezone.now() - timedelta(days=1)
        with pytest.raises(BriefingReopenError):
            BriefingService.reopen(
                briefing=briefing_a, actor=user_owner,
                new_expires_at=past,
            )

    def test_reopen_expired_with_future_deadline_succeeds(
        self, briefing_a, user_owner,
    ):
        briefing_a.status = Briefing.STATUS_EXPIRED
        briefing_a.save(update_fields=['status'])

        future = timezone.now() + timedelta(days=7)
        b = BriefingService.reopen(
            briefing=briefing_a, actor=user_owner,
            new_expires_at=future, reason='Prorrogado por pedido do cliente.',
        )

        assert b.status == Briefing.STATUS_IN_PROGRESS
        assert b.expires_at is not None
        assert b.expires_at >= future - timedelta(seconds=2)

        # Evento registrado
        ev = BriefingEvent.objects.filter(
            briefing=b, kind=BriefingEvent.KIND_REOPENED,
        ).order_by('-created_at').first()
        assert ev is not None
        assert ev.content.get('reason') == 'Prorrogado por pedido do cliente.'
        assert ev.content.get('new_expires_at') is not None

    def test_reopen_closed_without_deadline_succeeds(
        self, briefing_a, user_owner,
    ):
        # Fecha o briefing
        BriefingService.close(briefing=briefing_a, actor=user_owner)
        briefing_a.refresh_from_db()
        assert briefing_a.status == Briefing.STATUS_CLOSED

        b = BriefingService.reopen(
            briefing=briefing_a, actor=user_owner,
            reason='Reaberto manualmente.',
        )
        assert b.status == Briefing.STATUS_IN_PROGRESS

import pytest

from apps.briefings.models import (
    Briefing,
    BriefingAnswer,
    BriefingEvent,
)

@pytest.mark.django_db
class TestSequence:

    def test_answer_first_question(
        self, briefing_a, portal_session_a,
    ):
        questions = briefing_a.snapshot_questions()
        first = questions[0]

        answer = BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=first['id'],
            content_text='Minha rotina é tranquila.',
        )

        assert answer.state == BriefingAnswer.STATE_ANSWERED
        assert answer.latest_content == 'Minha rotina é tranquila.'

    def test_cannot_answer_question_out_of_order(
        self, briefing_a, portal_session_a,
    ):
        questions = briefing_a.snapshot_questions()
        third = questions[2]  # nunca respondida, e não é a próxima

        with pytest.raises(InvalidQuestionSequenceError):
            BriefingService.submit_answer(
                briefing_id=briefing_a.id,
                portal_session=portal_session_a,
                question_id=third['id'],
                content_text='Pulando fila.',
            )

    def test_flow_completes_when_all_answered(
        self, briefing_a, portal_session_a,
    ):
        questions = briefing_a.snapshot_questions()

        for q in questions:
            BriefingService.submit_answer(
                briefing_id=briefing_a.id,
                portal_session=portal_session_a,
                question_id=q['id'],
                content_text=f'Resposta para {q["id"]}',
            )

        briefing_a.refresh_from_db()
        assert briefing_a.status == Briefing.STATUS_COMPLETED
        assert briefing_a.completed_at is not None

    def test_review_requested_remains_next_eligible(
        self, briefing_a, portal_session_a,
    ):
        questions = briefing_a.snapshot_questions()
        first = questions[0]

        # Responde
        BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=first['id'],
            content_text='Minha rotina...',
        )

        # Cliente pede revisão na mesma pergunta
        BriefingService.request_review(
            briefing=briefing_a,
            portal_session=portal_session_a,
            question_id=first['id'],
            message='Quero complementar.',
        )

        # Próxima elegível é de novo a pergunta 1
        nxt = BriefingService.get_next_eligible_question(briefing_a)
        assert str(nxt['id']) == str(first['id'])

    def test_answer_event_created_per_submission(
        self, briefing_a, portal_session_a,
    ):
        """
        Reenvio de resposta exige `request_review` antes.

        O fluxo é:
            1. Cliente responde.
            2. Cliente pede revisão (volta a ser elegível).
            3. Cliente responde de novo.
            4. Dois eventos `answered` ficam registrados no histórico.
        """
        questions = briefing_a.snapshot_questions()
        first = questions[0]

        # 1ª resposta
        BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=first['id'],
            content_text='Primeira resposta.',
        )

        # Solicita revisão — reabre a pergunta
        BriefingService.request_review(
            briefing=briefing_a,
            portal_session=portal_session_a,
            question_id=first['id'],
            message='Quero reformular.',
        )

        # 2ª resposta — agora permitida, pois a pergunta voltou
        # a ser a próxima elegível
        BriefingService.submit_answer(
            briefing_id=briefing_a.id,
            portal_session=portal_session_a,
            question_id=first['id'],
            content_text='Segunda resposta (após review).',
        )

        answered_events = BriefingEvent.objects.filter(
            briefing=briefing_a,
            kind=BriefingEvent.KIND_ANSWERED,
            question_id=first['id'],
        )
        assert answered_events.count() == 2

    def test_content_invalid_format_rejected(
        self, briefing_a, portal_session_a,
    ):
        from apps.briefings.exceptions import InvalidResponseFormatError

        questions = briefing_a.snapshot_questions()
        third = questions[2]  # allowed_response_formats=['text']

        # Responder as duas primeiras para liberar a terceira
        first, second = questions[0], questions[1]
        for q in (first, second):
            BriefingService.submit_answer(
                briefing_id=briefing_a.id,
                portal_session=portal_session_a,
                question_id=q['id'],
                content_text='ok',
            )

        # Agora `third` é a próxima elegível. Ela só aceita texto;
        # enviamos nada para forçar erro de formato.
        with pytest.raises(InvalidResponseFormatError):
            BriefingService.submit_answer(
                briefing_id=briefing_a.id,
                portal_session=portal_session_a,
                question_id=third['id'],
                content_text='',
            )

"""
Views HTML do módulo briefings.

    /fluxo/                                       → painel do arquiteto
    /w/<public_id>/briefings/<briefing_id>/       → detalhe autenticado

Ambas as páginas consomem a API do próprio módulo. O shell
(sidebar.html) resolve o resto no client-side.
"""

from django.shortcuts import redirect, render
from django.views import View

from apps.workspaces.helpers import resolve_workspace_or_redirect
from apps.workspaces.models import WorkspaceMembership
from apps.briefings.models import Briefing


class FluxoView(View):
    """Página de Fluxo (Briefings) — painel do arquiteto."""

    def get(self, request):
        if not request.user.is_authenticated:
            return redirect('/login/')
        return render(request, 'fluxo.html')


class BriefingDetailView(View):
    """
    Detalhe HTML autenticado de um briefing.

    Rota: /w/<public_id>/briefings/<briefing_id>/

    Resolve o workspace a partir do public_id + user. Depois
    valida que o briefing pertence a esse workspace. Só então
    renderiza.
    """

    def get(self, request, public_id, briefing_id):
        if not request.user.is_authenticated:
            return redirect('/login/')

        workspace, error = resolve_workspace_or_redirect(request, public_id)
        if error is not None:
            return error

        is_member = WorkspaceMembership.objects.filter(
            workspace=workspace, user=request.user,
        ).exists()
        if not is_member:
            return redirect('/workspaces/entrar/')

        briefing = Briefing.objects.filter(
            id=briefing_id, workspace=workspace,
        ).first()
        if briefing is None:
            return redirect('/fluxo/')

        return render(request, 'briefing_detail.html', {
            'workspace': workspace,
            'briefing_id': str(briefing.id),
        })
