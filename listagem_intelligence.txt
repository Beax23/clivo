"""
Views da API de Inteligência.

Endpoints:
    GET  /api/intelligence/mesa/
        Containers da Mesa para o workspace atual (opcional ?client=).

    GET  /api/intelligence/clientes/<client_id>/
        Containers do Cliente.

    GET  /api/intelligence/briefings/<briefing_id>/relatorio/
        Relatório do briefing (8 containers).

    GET  /api/intelligence/tags/
        Lista tags do workspace (?client=&state=).

    GET  /api/intelligence/tags/<id>/
        Detalhe da tag + histórico.

    PATCH /api/intelligence/tags/<id>/
        Transição de estado (confirmada/descartada) com histórico.

Autorização:
    - Usuário autenticado.
    - Membro do workspace (via `user_has_capability` do módulo
      `workspaces`, que é o único ponto de autorização cross-app).
    - Capability necessária: `intelligence.insight.view` para leitura
      e `intelligence.tag.manage` para transição de tag.
"""

import logging

from rest_framework import status

def _resolve_workspace(request):
    """
    Resolve o workspace ativo da sessão. Retorna None se não houver.
    Reusa o contrato de `workspaces` — sem duplicar resolução.
    """
    ws_id = request.session.get('current_workspace_id')
    if not ws_id:
        return None
    return WorkspaceQueries.get_for_user(ws_id, request.user)


def _require(workspace, user, capability: str):
    """
    Retorna (True, None) ou (False, Response 403).
    """
    if workspace is None:
        return False, Response(
            {'error': 'Nenhum workspace ativo'},
            status=status.HTTP_400_BAD_REQUEST,
        )
    if not user_has_capability(workspace, user, capability):
        return False, Response(
            {'error': f'Capability necessária: {capability}'},
            status=status.HTTP_403_FORBIDDEN,
        )
    return True, None


class MesaView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.insight.view',
        )
        if not ok:
            return err

        client = None
        client_id = request.query_params.get('client')
        if client_id:
            from apps.clients.models import Client
            client = Client.objects.filter(
                id=client_id, workspace=workspace,
            ).first()
            if client is None:
                return Response(
                    {'error': 'Cliente não encontrado'},
                    status=status.HTTP_404_NOT_FOUND,
                )

        include_stale = request.query_params.get('include_stale') == '1'
        payload = ContainerService.build_mesa(
            workspace=workspace,
            client=client,
            include_stale=include_stale,
        )
        serializer = ContainerPayloadSerializer(payload)
        return Response(serializer.data)

class TagDetailView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request, tag_id):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.insight.view',
        )
        if not ok:
            return err

        tag = InsightItem.objects.filter(
            id=tag_id, workspace=workspace, kind=InsightItem.KIND_TAG,
        ).first()
        if tag is None:
            return Response(
                {'error': 'Tag não encontrada'},
                status=status.HTTP_404_NOT_FOUND,
            )

        history = TagService.history_for(tag)
        return Response({
            'tag': InsightItemSerializer(tag).data,
            'history': InsightTagHistorySerializer(history, many=True).data,
        })

    def patch(self, request, tag_id):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.tag.manage',
        )
        if not ok:
            return err

        tag = InsightItem.objects.filter(
            id=tag_id, workspace=workspace, kind=InsightItem.KIND_TAG,
        ).first()
        if tag is None:
            return Response(
                {'error': 'Tag não encontrada'},
                status=status.HTTP_404_NOT_FOUND,
            )

        serializer = TagTransitionSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        try:
            tag = TagService.transition(
                insight=tag,
                new_state=serializer.validated_data['new_state'],
                changed_by=request.user,
                reason=serializer.validated_data.get('reason', ''),
            )
        except InvalidTagTransitionError as exc:
            return Response(
                {'error': str(exc)},
                status=status.HTTP_400_BAD_REQUEST,
            )

        return Response(InsightItemSerializer(tag).data)

"""
Comando de gerenciamento para processar jobs de inteligência.

CONCORRÊNCIA
------------
Cada job é reivindicado via `JobService.claim_next_job()`, que usa
`select_for_update(skip_locked=True)` E marca o job como `processing`
dentro da mesma transação. Dois workers concorrentes nunca processam
o mesmo job.
"""

import logging
import time

from django.core.management.base import BaseCommand

from apps.intelligence.services import JobService, AnalysisService
from apps.intelligence.models import AnalysisJob, AnalysisRun
from apps.intelligence.exceptions import BudgetExceededError, StaleAnalysisError

logger = logging.getLogger('intelligence')


class Command(BaseCommand):
    help = 'Processa os jobs de análise de inteligência pendentes.'

    def add_arguments(self, parser):
        parser.add_argument('--interval', type=int, default=5)
        parser.add_argument('--run-once', action='store_true')

    def handle(self, *args, **options):
        run_once = options['run_once']
        interval = options['interval']

        self.stdout.write(self.style.SUCCESS('Worker de Inteligência iniciado.'))

        while True:
            try:
                job = JobService.claim_next_job()

                if job is None:
                    if run_once:
                        break
                    time.sleep(interval)
                    continue

                self.process_job(job)

            except KeyboardInterrupt:
                self.stdout.write(self.style.WARNING('\nWorker interrompido.'))
                break
            except Exception:
                logger.exception('Erro no loop principal do worker.')
                if run_once:
                    raise
                time.sleep(interval)

        self.stdout.write(self.style.SUCCESS('Worker finalizado.'))

    def process_job(self, job: AnalysisJob):
        job_id_str = str(job.id)
        self.stdout.write(f'Processando job {job_id_str}...')

        try:
            run = AnalysisService.execute(job)

            if run is None:
                self.stdout.write(self.style.WARNING(
                    f'Job {job_id_str} ignorado (duplicado).'
                ))
                JobService.mark_as_completed(job)
                return

            if run.status == AnalysisRun.STATUS_COMPLETED:
                self.stdout.write(self.style.SUCCESS(
                    f'Job {job_id_str} concluído (Run {run.id}).'
                ))
                JobService.mark_as_completed(job)
            else:
                JobService.mark_as_failed(job, f'Estado inesperado: {run.status}')

        except StaleAnalysisError as exc:
            self.stdout.write(self.style.WARNING(
                f'Job {job_id_str} ignorado (stale): {exc}'
            ))
            JobService.mark_as_completed(job)

        except BudgetExceededError as exc:
            self.stdout.write(self.style.ERROR(
                f'Job {job_id_str} falhou (orçamento): {exc}'
            ))
            JobService.mark_as_failed(job, f'BudgetExceeded: {exc}')

        except Exception as exc:
            self.stdout.write(self.style.ERROR(
                f'Job {job_id_str} falhou: {exc}'
            ))
            logger.exception('Falha ao processar job %s', job_id_str)
            JobService.mark_as_failed(job, str(exc))

"""
Alocação de custo de uma execução de IA aos containers/insights
que ela produziu.

REGRA ARQUITETURAL
------------------
`UsageRecord` (em `console`) registra o custo REAL da chamada ao
provedor. Este modelo registra a PARCELA desse custo atribuída a
cada `InsightItem` produzido, segundo uma regra de rateio explícita.

Separação de responsabilidades:
    - console.UsageRecord   → quanto foi cobrado pelo provedor.
    - InsightCostAllocation → como esse custo se distribui entre
                              os insights gerados.

A regra de rateio é determinística e documentada em
`CostAllocationService.allocate_for_run()`. Este modelo apenas
persiste o resultado.

Semântica dos campos:
    - `allocated_cost = None`   → custo desconhecido (não foi possível
                                  calcular, nem por rateio). O chamador
                                  deve tratar como "indisponível", nunca
                                  como zero.
    - `is_estimated = True`     → o valor é resultado de rateio, não
                                  de uma chamada individual. O campo
                                  `allocation_rule` documenta a regra.
"""

import uuid

from django.db import models
from django.utils.translation import gettext_lazy as _


class InsightCostAllocation(models.Model):

    RULE_SINGLE_INSIGHT = 'single_insight'
    RULE_EQUAL_SPLIT = 'equal_split'
    RULE_PROPORTIONAL_TOKENS = 'proportional_tokens'

    RULE_CHOICES = [
        (RULE_SINGLE_INSIGHT, 'Um único insight na execução'),
        (RULE_EQUAL_SPLIT, 'Divisão igual entre insights'),
        (RULE_PROPORTIONAL_TOKENS, 'Proporcional a tokens estimados'),
    ]

    id = models.UUIDField(
        primary_key=True, default=uuid.uuid4, editable=False,
    )

    insight = models.ForeignKey(
        'intelligence.InsightItem',
        on_delete=models.CASCADE,

        result = set()
        for kinds in mapping.values():
            result.update(kinds)

            matching = [
                slot_name for slot_name, kinds in slots.items()
                if kind in kinds
            ]
            if matching:
                result[container] = matching
        return result

REGRA ARQUITETURAL
------------------
`intelligence` NÃO chama OpenAI diretamente. Chama a ponte
`console.public.call_ai(...)`, que:
    - verifica quota,
    - seleciona provider/modelo,
    - executa a chamada real,
    - registra ProviderLog e UsageRecord,
    - devolve um payload normalizado.

`AnalysisService` é responsável por:
    - preparar input (via contratos públicos),
    - validar a saída da IA,
    - persistir insights, evidências e histórico,
    - alocar custo aos insights produzidos,
    - marcar insights anteriores como stale.

ANÁLISE PARCIAL
---------------
`AnalysisJob.target_kinds` define quais kinds devem ser produzidos.
Quando vazio, produz o conjunto completo padrão. Quando preenchido,
o prompt é instruído a produzir APENAS esses kinds, e a validação
rejeita kinds fora da lista.

MULTIMODAL
----------
Se `briefings.public.get_reference_document_payloads()` devolver
imagens acessíveis, o payload inclui entradas `images` que a ponte
`console.public.call_ai` encaminha ao provider. Se a interface não
estiver disponível, o input degrada para texto sem inventar imagens.
"""

class AnalysisService:

    @staticmethod
    def execute(job: AnalysisJob) -> Optional[AnalysisRun]:
        """
        Executa o fluxo completo.

        A chamada à IA acontece FORA de qualquer transação — não
        bloqueia o banco durante a latência do provider.
        """
        with transaction.atomic():
            run = AnalysisService._create_analysis_run(job)
            if run is None:
                return None

        try:
            from apps.console import public as console_public
            allowed, reason = console_public.is_ai_execution_allowed(job.workspace)
            if not allowed:
                raise BudgetExceededError(
                    f'Execução bloqueada pela política de quota: {reason}'
                )

            input_data = AnalysisService._prepare_input(run)
            if not input_data:
                raise AnalysisInputError('Falha ao preparar os dados de entrada.')

            ai_response = AnalysisService._call_ai_via_console(run, input_data)

            insights_data = AnalysisService._parse_and_validate_ai_response(
                ai_response,
                target_kinds=AnalysisService._resolve_target_kinds(run),
            )

            with transaction.atomic():
                AnalysisService._apply_ai_results(
                    run=run,
                    ai_response=ai_response,
                    insights_data=insights_data,
                    duration_ms=ai_response.get('duration_ms'),
                )

            # Alocação de custo e staleness — fora da transação principal
            # para não bloquear o commit dos insights.
            try:
                from apps.intelligence.services.cost_allocation_service import (
                    CostAllocationService,
                )
                CostAllocationService.allocate_for_run(run, ai_response)
            except Exception:
                logger.exception(
                    'Falha ao alocar custo (run=%s). Insights persistidos.',
                    run.id,
                )

            try:
                from apps.intelligence.services.staleness_service import (
                    StalenessService,
                )
                StalenessService.mark_previous_runs_stale(run)
            except Exception:
                logger.exception(
                    'Falha ao marcar runs anteriores como stale (run=%s).',
                    run.id,
                )

            return run

        except StaleAnalysisError:
            with transaction.atomic():
                run.status = AnalysisRun.STATUS_SKIPPED_STALE
                run.is_stale = True
                run.completed_at = timezone.now()
                run.save(update_fields=[
                    'status', 'is_stale', 'completed_at', 'updated_at',
                ])
            raise

        except Exception as exc:
            logger.exception('Falha na execução da análise (Run=%s)', run.id)
            with transaction.atomic():
                run.status = AnalysisRun.STATUS_FAILED
                run.error_message = str(exc)
                run.completed_at = timezone.now()
                run.save(update_fields=[
                    'status', 'error_message', 'completed_at', 'updated_at',
                ])
            raise

    # ------------------------------------------------------------------
    # TARGET KINDS
    # ------------------------------------------------------------------

    @staticmethod
    def _resolve_target_kinds(run: AnalysisRun) -> List[str]:
        """
        Resolve quais kinds este run deve produzir.

        Prioridade:
            1. `job.target_kinds` (definido pelo agendador, ex.: análise
               parcial de um container específico).
            2. Default por escopo (com ou sem briefing).
        """
        job = run.job
        requested = job.target_kinds or []
        valid = {c[0] for c in InsightItem.KIND_CHOICES}
        filtered = [k for k in requested if k in valid]
        if filtered:
            return filtered

        if run.briefing_id:
            return list(DEFAULT_KINDS_WITH_BRIEFING)
        return list(DEFAULT_KINDS_WITHOUT_BRIEFING)

    # ------------------------------------------------------------------
    # CRIAÇÃO DO RUN
    # ------------------------------------------------------------------

    @staticmethod
    def _create_analysis_run(job: AnalysisJob) -> Optional[AnalysisRun]:
        """
        Cria o run aplicando idempotência por `input_hash`.

        O `input_hash` inclui workspace + prompt + provider + model +
        payload + target_kinds. Isso diferencia análises parciais
        (kinds distintos) de re-execuções completas.
        """
        input_payload = AnalysisService._prepare_input(job)
        if not input_payload:
            raise AnalysisInputError('Falha ao preparar input para hashing.')

        model = getattr(settings, 'INTELLIGENCE_MODEL', 'gpt-4o-mini')
        prompt_version = getattr(settings, 'INTELLIGENCE_PROMPT_VERSION', 'v1')
        provider = getattr(settings, 'INTELLIGENCE_PROVIDER', 'openai')

        target_kinds = AnalysisService._resolve_target_kinds_from_job(job)
        payload_hash = hashlib.sha256(
            json.dumps(
                {'input': input_payload, 'kinds': sorted(target_kinds)},
                sort_keys=True, default=str,
            ).encode('utf-8')
        ).hexdigest()

        composite = '|'.join([
            str(job.workspace_id),
            prompt_version,
            provider,
            model,
            payload_hash,
        ])
        input_hash = hashlib.sha256(composite.encode('utf-8')).hexdigest()

        existing = AnalysisRun.objects.filter(
            input_hash=input_hash,
            status=AnalysisRun.STATUS_COMPLETED,
        ).first()
        if existing:
            logger.info(
                'Run idêntico já concluído (hash=%s). Ignorando.', input_hash,
            )
            return None

        return AnalysisRun.objects.create(
            job=job,
            workspace=job.workspace,
            client=job.client,
            briefing=job.briefing,
            input_hash=input_hash,
            data_version=job.data_version,
            provider=provider,
            model=model,
            prompt_version=prompt_version,
            status=AnalysisRun.STATUS_RUNNING,
            started_at=timezone.now(),
        )

    @staticmethod
    def _resolve_target_kinds_from_job(job: AnalysisJob) -> List[str]:
        requested = job.target_kinds or []
        valid = {c[0] for c in InsightItem.KIND_CHOICES}
        filtered = [k for k in requested if k in valid]
        if filtered:
            return filtered
        if job.briefing_id:
            return list(DEFAULT_KINDS_WITH_BRIEFING)
        return list(DEFAULT_KINDS_WITHOUT_BRIEFING)

    # ------------------------------------------------------------------
    # INPUT (via contratos públicos)
    # ------------------------------------------------------------------

    @staticmethod
    def _prepare_input(job_or_run) -> Optional[Dict[str, Any]]:
        """
        Monta o payload a partir dos contratos públicos.
        Nenhum import direto de models de `briefings`/`clients`.
        """
        from apps.briefings import public as briefings_public
        from apps.clients import public as clients_public

        client_snap = clients_public.get_client_snapshot(job_or_run.client_id)
        if not client_snap:
            raise AnalysisInputError('Cliente não encontrado.')

        payload = {'client': client_snap}

        if job_or_run.briefing_id:
            briefing_snap = briefings_public.get_briefing_snapshot(
                job_or_run.briefing_id
            )
            if not briefing_snap:
                raise AnalysisInputError('Briefing não encontrado.')
            payload['briefing'] = briefing_snap

            # Multimodal: tenta obter imagens via contrato público.
            # Se a interface não existir ou devolver vazio, o input
            # degrada para texto SEM inventar imagens.
            images = AnalysisService._collect_images(
                briefings_public, briefing_snap,
            )
            if images:
                payload['images'] = images

        return payload

    @staticmethod
    def _collect_images(briefings_public, briefing_snap: dict) -> List[Dict]:
        """
        Coleta imagens acessíveis do briefing via contrato público.

        Regra: se `get_reference_document_payloads` não existir ou
        devolver payloads sem URL acessível, retorna lista vazia.
        NUNCA inventa descrição de imagem a partir do ID.
        """
        getter = getattr(
            briefings_public, 'get_reference_document_payloads', None,
        )
        if getter is None:
            logger.info(
                'briefings.public.get_reference_document_payloads '
                'indisponível. Análise será apenas textual.'
            )
            return []

        doc_ids = set()
        for answer in briefing_snap.get('answers', []):
            doc_ids.update(answer.get('attachments') or [])
            if answer.get('audio_document_id'):
                doc_ids.add(answer['audio_document_id'])

        for section in (briefing_snap.get('structure_snapshot') or {}).get(
            'sections', []
        ):
            for q in section.get('questions', []):
                for rid in q.get('reference_document_ids') or []:
                    doc_ids.add(rid)

        if not doc_ids:
            return []

        try:
            payloads = getter(list(doc_ids)) or []
        except Exception:
            logger.exception(
                'Falha ao obter payloads de documentos. '
                'Análise será apenas textual.'
            )
            return []

        result = []
        for p in payloads:
            if not isinstance(p, dict):
                continue
            url = p.get('url')
            mime = p.get('mime_type') or ''
            if not url or not mime.startswith('image/'):
                continue
            result.append({
                'document_id': str(p.get('id') or ''),
                'url': url,
                'mime_type': mime,
                'source': p.get('source') or 'briefing',
            })
        return result

    # ------------------------------------------------------------------
    # IA — via ponte do console
    # ------------------------------------------------------------------

    @staticmethod
    def _call_ai_via_console(run: AnalysisRun, input_data: dict) -> dict:
        """
        Chama a ponte `console.public.call_ai`, que executa o provider
        real e registra uso/custo. Devolve payload normalizado.
        """
        from apps.console import public as console_public

        call_ai = getattr(console_public, 'call_ai', None)
        if call_ai is None:
            raise AIProviderError(
                'A ponte console.public.call_ai não está disponível. '
                'A inteligência não pode executar IA sem passar pelo '
                'Console (governança + custo).'
            )

        system_prompt = get_prompt()
        user_message = (
            "Analise os dados abaixo e produza insights no formato JSON:\n\n"
            f"{json.dumps(input_data, ensure_ascii=False, indent=2, default=str)}"
        )

        started = time.time()
        try:
            response = call_ai(
                workspace=run.workspace,
                operation='intelligence.analysis',
                provider_key=run.provider,
                model=run.model,
                prompt_version=run.prompt_version,
                messages=[
                    {'role': 'system', 'content': system_prompt},
                    {'role': 'user', 'content': user_message},
                ],
                images=input_data.get('images') or [],
                response_format='json_object',
                request_id=str(run.id),
            )
        except Exception as exc:
            duration_ms = int((time.time() - started) * 1000)
            logger.exception(
                'Falha na ponte call_ai (run=%s, duration=%sms)',
                run.id, duration_ms,
            )
            raise AIProviderError(f'Falha na chamada da IA: {exc}') from exc

        duration_ms = int((time.time() - started) * 1000)

        return {
            'raw_content': response.get('raw_content', ''),
            'input_tokens': int(response.get('input_tokens') or 0),
            'output_tokens': int(response.get('output_tokens') or 0),
            'duration_ms': response.get('duration_ms', duration_ms),
            'usage_record_id': response.get('usage_record_id'),
            'provider_request_id': response.get('provider_request_id') or '',
        }

    # ------------------------------------------------------------------
    # VALIDAÇÃO
    # ------------------------------------------------------------------

    @staticmethod
    def _parse_and_validate_ai_response(
        ai_response: dict,
        target_kinds: Optional[List[str]] = None,
    ) -> List[Dict]:
        """
        Valida o JSON. Exige 'insights' (lista) com 'kind' válido,
        'content' não vazio e 'confidence' válida.

        Se `target_kinds` for fornecido, rejeita kinds fora da lista
        (análise parcial não pode produzir o que não foi pedido).

        Rejeita `evidences=[]` para kinds que exigem evidência.
        """
        raw = ai_response.get('raw_content') or ''
        try:
            data = json.loads(raw)
        except json.JSONDecodeError as exc:
            raise AIResponseError(f'Resposta da IA não é JSON válido: {exc}')

        if not isinstance(data, dict) or 'insights' not in data:
            raise AIResponseError("Resposta da IA não contém a chave 'insights'.")

        insights = data['insights']
        if not isinstance(insights, list):
            raise AIResponseError("'insights' deve ser uma lista.")

        valid_kinds = {c[0] for c in InsightItem.KIND_CHOICES}
        valid_confidences = {c[0] for c in InsightItem.CONFIDENCE_CHOICES}
        allowed = set(target_kinds) if target_kinds else valid_kinds

        validated = []
        for i, item in enumerate(insights):
            if not isinstance(item, dict):
                raise AIResponseError(f'Insight #{i} não é um objeto.')
            kind = item.get('kind')
            content = (item.get('content') or '').strip()
            confidence = item.get(
                'confidence', InsightItem.CONFIDENCE_INTERPRETATION,
            )

            if kind not in valid_kinds:
                raise AIResponseError(f'Insight #{i}: kind inválido ({kind}).')
            if kind not in allowed:
                raise AIResponseError(
                    f'Insight #{i}: kind "{kind}" não solicitado nesta '
                    f'execução (target_kinds).'
                )
            if not content:
                raise AIResponseError(f'Insight #{i}: content vazio.')
            if confidence not in valid_confidences:
                raise AIResponseError(
                    f'Insight #{i}: confidence inválida ({confidence}).'
                )

            evidences = item.get('evidences') or []
            if not isinstance(evidences, list):
                raise AIResponseError(f'Insight #{i}: evidences deve ser lista.')

            if kind in KINDS_REQUIRING_EVIDENCE and not evidences:
                raise AIResponseError(
                    f'Insight #{i} ({kind}): evidência obrigatória ausente.'
                )

            validated.append({
                'kind': kind,
                'content': content,
                'confidence': confidence,
                'metadata': item.get('metadata') or {},
                'evidences': evidences,
            })

        return validated

    # ------------------------------------------------------------------
    # PERSISTÊNCIA
    # ------------------------------------------------------------------

    @staticmethod
    def _apply_ai_results(run, ai_response, insights_data, duration_ms):
        """
        Persiste os insights e evidências, e atualiza as métricas do run.
        Detecta stale comparando a data de criação do run com o
        `created_at` do último `InsightEvent` do alvo.
        """
        latest_event = (
            InsightEvent.objects
            .filter(
                workspace=run.workspace,
                client=run.client,
                briefing=run.briefing,
            )
            .order_by('-created_at')
            .first()
        )
        if latest_event and latest_event.created_at > run.created_at:
            raise StaleAnalysisError(
                'Existe um evento mais recente que o run. Não publicar.'
            )

        run.input_tokens = ai_response.get('input_tokens') or 0
        run.output_tokens = ai_response.get('output_tokens') or 0
        run.duration_ms = duration_ms
        run.status = AnalysisRun.STATUS_COMPLETED
        run.completed_at = timezone.now()
        run.save()

        for item in insights_data:
            insight = InsightItem.objects.create(
                run=run,
                workspace=run.workspace,
                client=run.client,
                briefing=run.briefing,
                kind=item['kind'],
                content=item['content'],
                confidence=item['confidence'],
                metadata=item['metadata'],
            )
            AnalysisService._persist_evidences(
                insight=insight,
                raw_evidences=item.get('evidences') or [],
                workspace=run.workspace,
            )

    @staticmethod
    def _persist_evidences(insight, raw_evidences, workspace):
        """
        Persiste as evidências, validando que cada fonte pertence ao
        mesmo workspace do insight. Fontes inválidas são descartadas
        com log, nunca silenciosamente aceitas.
        """
        from apps.briefings import public as briefings_public
        from apps.clients import public as clients_public

        for ev in raw_evidences:
            if not isinstance(ev, dict):
                continue
            source_type = ev.get('source_type')
            source_ref = (ev.get('source_ref') or '').strip()
            quote = (ev.get('quote') or '').strip()

            if source_type not in dict(InsightEvidence.SOURCE_TYPE_CHOICES):
                logger.warning(
                    'Evidência com source_type inválido ignorada: %s',
                    source_type,
                )
                continue
            if not source_ref:
                logger.warning('Evidência sem source_ref ignorada.')
                continue

            valid = False
            if source_type == InsightEvidence.SOURCE_TYPE_BRIEFING_ANSWER:
                snap = (
                    briefings_public.get_briefing_snapshot(insight.briefing_id)
                    if insight.briefing_id else None
                )
                if snap and any(
                    a['id'] == source_ref for a in snap['answers']
                ):
                    valid = True
            elif source_type == InsightEvidence.SOURCE_TYPE_BRIEFING_NOTE:
                snap = (
                    briefings_public.get_briefing_snapshot(insight.briefing_id)
                    if insight.briefing_id else None
                )
                if snap and any(
                    n['id'] == source_ref for n in snap['notes']
                ):
                    valid = True
            elif source_type == InsightEvidence.SOURCE_TYPE_CLIENT_CONTEXT:
                snap = clients_public.get_client_snapshot(insight.client_id)
                if snap:
                    valid = True

            if not valid:
                logger.warning(
                    'Evidência ignorada (fonte inexistente ou de outro '
                    'workspace): %s:%s',
                    source_type, source_ref,
                )
                continue

            InsightEvidence.objects.get_or_create(
                insight=insight,
                source_type=source_type,
                source_ref=source_ref,
                defaults={'quote': quote},
            )

"""
ContainerService — monta a resposta dos containers fixos.

REGRA ARQUITETURAL
------------------
Os containers são estruturas FIXAS (Mesa, Cliente, Relatório). Este
service é a ÚNICA fonte de verdade para montar o payload de cada um.
Views e serializers delegam aqui — nunca inventam um container novo.

Filtragem:
    - Por `kind` válido do container.
    - Por cliente/workspace/briefing quando aplicável.
    - Ignora insights `is_stale=True` por padrão (a interface pode
      pedir para incluir).
"""

import logging
from typing import Dict, List, Optional

from apps.intelligence.models import InsightItem

logger = logging.getLogger('intelligence')


class ContainerService:

    @staticmethod
    def build_mesa(
        *,
        workspace,
        client=None,
        briefing=None,
        include_stale: bool = False,
    ) -> Dict:
        """
        Monta os cinco containers da Mesa.

        Retorna:
            {
                'containers': {
                    '01_desejos_prioridades': [insight, ...],
                    '02_pontos_atencao': [...],
                    '03_mudancas': [...],
                    '04_novidades': [...],
                    '05_resumo': [...],
                },
                'counts': {slot: int, ...},
            }
        """
        return ContainerService._build(
            container=InsightItem.CONTAINER_MESA,
            workspace=workspace,
            client=client,
            briefing=briefing,
            include_stale=include_stale,
        )

    @staticmethod
    def build_cliente(
        *,
        workspace,
        client,
        include_stale: bool = False,
    ) -> Dict:
        """Monta os cinco containers do Cliente."""
        return ContainerService._build(
            container=InsightItem.CONTAINER_CLIENTE,
            workspace=workspace,
            client=client,
            briefing=None,  # conhecimento do cliente é transversal a briefings
            include_stale=include_stale,
        )

    @staticmethod
    def build_relatorio(
        *,
        workspace,
        briefing,
        include_stale: bool = False,
    ) -> Dict:
        """Monta o relatório do briefing (8 containers fixos)."""
        return ContainerService._build(
            container=InsightItem.CONTAINER_RELATORIO,
            workspace=workspace,
            client=briefing.client,
            briefing=briefing,
            include_stale=include_stale,
        )

    # ------------------------------------------------------------------
    # INTERNO
    # ------------------------------------------------------------------

    @staticmethod
    def _build(
        *,
        container: str,
        workspace,
        client,
        briefing,
        include_stale: bool,
    ) -> Dict:
        slots = InsightItem.CONTAINER_KINDS[container]
        all_kinds = InsightItem.kinds_for_container(container)

        qs = (
            InsightItem.objects
            .filter(workspace=workspace, kind__in=all_kinds)
            .select_related('client', 'briefing', 'run')
            .order_by('-created_at')
        )

"""
CostAllocationService — aloca custo real da execução aos insights
que ela produziu.

REGRA DE RATEIO (documentada e testada)
---------------------------------------
Para uma execução (AnalysisRun) que produziu N insights e teve
custo real C (lido de console.UsageRecord via usage_record_id):

    1. N == 0            → nenhuma alocação (não há a quem alocar).
    2. N == 1            → o único insight recebe C inteiro
                           (rule='single_insight').
    3. N > 1             → cada insight recebe C // N, e o ÚLTIMO
                           recebe C - (C // N) * (N - 1), absorvendo
                           o resto. A soma exata é C.
                           (rule='equal_split').

A absorção do resto no último insight é o que garante que a soma
das parcelas bate com o custo real. Sem isso, `C / 3` × 3 gera
diferença de 1 unidade na última casa decimal.

C é determinístico: vem do UsageRecord (custo real). Se C for
desconhecido (usage_record_id ausente ou UsageRecord sem preço),
a alocação é criada com `allocated_cost=None` e `is_estimated=True`
— sinalizando explicitamente "custo desconhecido", nunca zero.

IDEMPOTÊNCIA
------------
`allocate_for_run` pode ser chamado múltiplas vezes sem duplicar
linhas (constraint única em insight + usage_record_id). Re-execuções
com mesmo usage_record_id são no-op.
"""

import logging
from decimal import Decimal
from typing import Optional

from django.db import transaction

from apps.intelligence.models import (
    AnalysisRun, InsightItem, InsightCostAllocation,
)

logger = logging.getLogger('intelligence')


class CostAllocationService:

    @staticmethod
    def allocate_for_run(
        run: AnalysisRun,
        ai_response: dict,
    ) -> list:
        """
        Aloca o custo da execução aos insights produzidos.

        Deve ser chamado APÓS a persistência dos insights e APÓS o
        console ter criado o UsageRecord (via call_ai).

        Nunca levanta — falha de alocação não desfaz insights.
        """
        usage_record_id = ai_response.get('usage_record_id')
        if not usage_record_id:
            logger.info(
                'Run %s sem usage_record_id. Custo não alocável.', run.id,
            )
            return []

        total_cost, currency = CostAllocationService._fetch_usage_cost(
            usage_record_id,
        )

        insights = list(
            InsightItem.objects.filter(run=run).order_by('created_at')
        )
        if not insights:
            logger.info('Run %s não produziu insights. Nada a alocar.', run.id)
            return []

        allocations = []
        with transaction.atomic():
            if len(insights) == 1:
                allocations.append(
                    CostAllocationService._create(
                        insight=insights[0],
                        usage_record_id=usage_record_id,
                        cost=total_cost,
                        currency=currency,
                        rule=InsightCostAllocation.RULE_SINGLE_INSIGHT,
                        is_estimated=False if total_cost is not None else True,
                    )
                )
            else:
                n = len(insights)
                shares = []
                if total_cost is None:
                    shares = [None] * n
                else:
                    # Divisão com absorção do resto no último.
                    base = (total_cost / Decimal(n)).quantize(
                        Decimal('0.00000001'),
                    )
                    allocated_sum = base * (n - 1)
                    last = total_cost - allocated_sum
                    shares = [base] * (n - 1) + [last]

                for insight, share in zip(insights, shares):
                    allocations.append(
                        CostAllocationService._create(
                            insight=insight,
                            usage_record_id=usage_record_id,
                            cost=share,
                            currency=currency,
                            rule=InsightCostAllocation.RULE_EQUAL_SPLIT,
                            is_estimated=True,
                        )
                    )
        return [a for a in allocations if a is not None]

    # ------------------------------------------------------------------
    # HELPERS
    # ------------------------------------------------------------------

    @staticmethod
    def _fetch_usage_cost(usage_record_id) -> tuple:
        """
        Lê o custo real do UsageRecord.

        Retorna (Decimal | None, str). None significa desconhecido —
        NÃO significa zero.
        """
        try:
            from apps.console.models import UsageRecord
            record = UsageRecord.objects.filter(id=usage_record_id).only(
                'estimated_cost', 'currency',
            ).first()
        except Exception:
            logger.exception(
                'Falha ao consultar UsageRecord %s', usage_record_id,
            )
            return (None, '')

        if record is None:
            return (None, '')

        cost = record.estimated_cost
        currency = record.currency or ''
        if cost is None:
            return (None, currency)
        return (Decimal(cost), currency)

    @staticmethod
    def _create(
        *,
        insight,
        usage_record_id,
        cost: Optional[Decimal],
        currency: str,
        rule: str,
        is_estimated: bool,
    ):
        """
        Cria a alocação ou devolve a existente (idempotência).
        """
        try:
            obj, created = InsightCostAllocation.objects.get_or_create(
                insight=insight,
                usage_record_id=usage_record_id,
                defaults={
                    'allocated_cost': cost,
                    'currency': currency,
                    'allocation_rule': rule,
                    'is_estimated': is_estimated,
                },
            )
            return obj
        except Exception:
            logger.exception(
                'Falha ao criar alocação para insight=%s usage=%s',
                insight.id, usage_record_id,
            )
            return None

"""
IngestService — porta de entrada para novos eventos de inteligência.

FRONTEIRA
---------
Este serviço recebe apenas IDs e metadados — nunca instâncias de
models de outros módulos. A resolução dos objetos é feita pelo
JobService, que por sua vez usa contratos públicos.

IDEMPOTÊNCIA
------------
A `dedupe_key` inclui `occurrence_id`, que distingue eventos atômicos
legítimos (ex: duas edições na mesma nota) de reentrância do mesmo
evento. Quem chama é responsável por fornecer um `occurrence_id`
estável para o fato (ex: UUID do evento de origem, updated_at).
"""

import logging

from django.db import transaction

from apps.intelligence.models import InsightEvent
from apps.intelligence.services.job_service import JobService

logger = logging.getLogger('intelligence')


class IngestService:

    @staticmethod
    @transaction.atomic
    def ingest_change_event(
        *,
        workspace_id,
        client_id,
        source_kind: str,
        occurrence_id: str,
        briefing_id=None,
        source_ref: str = '',
        data_version: str,
    ):
        """
        Registra um evento de mudança e agenda uma análise.

        Retorna (event, created).
        """
        from apps.workspaces.models import Workspace
        from apps.clients.models import Client
        from apps.briefings.models import Briefing

        try:
            workspace = Workspace.objects.get(id=workspace_id)
            client = Client.objects.get(id=client_id)
            briefing = Briefing.objects.get(id=briefing_id) if briefing_id else None
        except Exception as exc:
            logger.exception('Falha ao resolver objetos do evento de ingestão')
            return (None, False)

        dedupe_key = ':'.join([
            str(workspace.id),
            str(client.id),
            str(briefing.id) if briefing else '',
            source_kind,
            source_ref or '',
            occurrence_id,
        ])

        event, created = InsightEvent.objects.get_or_create(
            dedupe_key=dedupe_key,
            defaults={
                'workspace': workspace,
                'client': client,
                'briefing': briefing,
                'source_kind': source_kind,
                'source_ref': source_ref,
                'occurrence_id': occurrence_id,
            },
        )

        if not created:
            logger.info("Evento duplicado ignorado (dedupe_key=%s)", dedupe_key)
            return (event, False)

        logger.info("Novo evento registrado: %s", source_kind)

        JobService.schedule_analysis_job(
            workspace_id=workspace.id,
            client_id=client.id,
            briefing_id=briefing.id if briefing else None,
            event=event,
            data_version=data_version,
        )

        return (event, True)

"""
JobService — agenda e reivindica jobs de análise de forma atômica.

CONCORRÊNCIA
------------
`claim_next_job()` abre uma transação, seleciona UM job pendente com
`select_for_update(skip_locked=True)` e o marca como `processing` ANTES
de liberar o lock. Dois workers concorrentes nunca reivindicam o mesmo
job.
"""

import hashlib
import logging
from typing import Optional

from django.db import transaction
from django.utils import timezone

from apps.intelligence.models import AnalysisJob, InsightEvent, InsightItem

logger = logging.getLogger('intelligence')


class JobService:

    @staticmethod
    @transaction.atomic
    def schedule_analysis_job(
        *,
        workspace_id,
        client_id,
        event: InsightEvent,
        data_version: str,
        briefing_id=None,
    ) -> AnalysisJob:
        from apps.workspaces.models import Workspace
        from apps.clients.models import Client
        from apps.briefings.models import Briefing

        workspace = Workspace.objects.get(id=workspace_id)
        client = Client.objects.get(id=client_id)
        briefing = Briefing.objects.get(id=briefing_id) if briefing_id else None

        if briefing:
            target_kinds = [
                InsightItem.KIND_SUMMARY_BRIEFING,
                InsightItem.KIND_DESIRE,
                InsightItem.KIND_ATTENTION,
                InsightItem.KIND_TAG,
                InsightItem.KIND_RECOMMENDED_QUESTION,
                InsightItem.KIND_GAP,
                InsightItem.KIND_INCONSISTENCY,
                InsightItem.KIND_CROSS_INSIGHT,
                InsightItem.KIND_SUCCESS_CRITERIA,
                InsightItem.KIND_REFERENCE_ANALYSIS,
                InsightItem.KIND_RESTRICTION,
                InsightItem.KIND_READINESS_ASSESSMENT,
            ]
        else:
            # Ingestão sem briefing: analisa o cliente como um todo.
            # `KIND_SUMMARY_CLIENT` não existe em InsightItem. O
            # equivalente semântico é `KIND_CLIENT_PROFILE`, que já
            # está na lista abaixo. Não duplicar.
            target_kinds = [
                InsightItem.KIND_KNOWLEDGE_ACCUMULATED,
                InsightItem.KIND_PREFERENCE_PROFILE,
                InsightItem.KIND_HOW_TO_CONDUCT,
                InsightItem.KIND_CLIENT_PROFILE,
            ]

        kinds_str = ','.join(sorted(target_kinds))
        kinds_hash = hashlib.sha256(kinds_str.encode('utf-8')).hexdigest()

        job_key = ':'.join([
            str(workspace.id),
            str(client.id),
            str(briefing.id) if briefing else '',
            kinds_hash,
            data_version,
        ])

        job, created = AnalysisJob.objects.get_or_create(
            job_key=job_key,
            defaults={
                'workspace': workspace,
                'client': client,
                'briefing': briefing,
                'target_kinds': target_kinds,
                'data_version': data_version,
            },
        )

        if created:
            logger.info("Novo job agendado (ID=%s)", job.id)
        else:
            logger.info("Job existente reutilizado (ID=%s)", job.id)
        return job

    @staticmethod
    def claim_next_job() -> Optional[AnalysisJob]:
        """
        Reivindica atomicamente um job pendente.

        O `select_for_update(skip_locked=True)` dentro da transação
        garante que dois workers não peguem o mesmo job. A atualização
        para `processing` acontece ANTES da transação fechar — o lock
        cobre a reivindicação.
        """
        with transaction.atomic():
            job = (
                AnalysisJob.objects
                .select_for_update(skip_locked=True)
                .filter(status=AnalysisJob.STATUS_PENDING)
                .order_by('created_at')
                .first()
            )
            if job is None:
                return None

            job.status = AnalysisJob.STATUS_PROCESSING
            job.attempt_count += 1
            job.processing_started_at = timezone.now()
            job.save(update_fields=[
                'status', 'attempt_count', 'processing_started_at', 'updated_at',
            ])
            return job

    @staticmethod
    def mark_as_completed(job: AnalysisJob):
        job.status = AnalysisJob.STATUS_COMPLETED
        job.completed_at = timezone.now()
        job.save(update_fields=['status', 'completed_at', 'updated_at'])

    @staticmethod
    def mark_as_failed(job: AnalysisJob, error_message: str):
        job.status = AnalysisJob.STATUS_FAILED
        job.failed_at = timezone.now()
        job.last_error_message = error_message
        job.save(update_fields=['status', 'failed_at', 'last_error_message', 'updated_at'])

"""
StalenessService — marca insights anteriores como stale quando uma
execução nova os substitui.

REGRA
-----
Quando um run novo conclui com sucesso, os insights de runs
anteriores do MESMO cliente E MESMO briefing que têm `kind` coberto
pelo run novo passam a `is_stale=True`.

NÃO apaga nada. O insight antigo continua consultável, mas sinalizado
como desatualizado. A interface pode exibi-lo com aviso ou ocultá-lo
por padrão.
"""

import logging

from django.db import transaction

from apps.intelligence.models import AnalysisRun, InsightItem

logger = logging.getLogger('intelligence')


class StalenessService:

    @staticmethod
    def mark_previous_runs_stale(new_run: AnalysisRun) -> int:
        """
        Marca como stale os insights de runs anteriores compatíveis.

        Compatibilidade: mesmo workspace, mesmo cliente, mesmo briefing
        (incluindo ambos NULL para análises de cliente sem briefing).

        Retorna a quantidade de insights marcados.
        """
        new_kinds = list(
            InsightItem.objects
            .filter(run=new_run)
            .values_list('kind', flat=True)
            .distinct()
        )
        if not new_kinds:
            return 0

        qs = (
            InsightItem.objects
            .filter(
                workspace=new_run.workspace,
                client=new_run.client,
                briefing=new_run.briefing,
                kind__in=new_kinds,
                is_stale=False,
            )
            .exclude(run=new_run)
        )
        count = qs.count()
        if count == 0:
            return 0

        with transaction.atomic():
            updated = qs.update(is_stale=True)

        logger.info(
            'StalenessService: %d insights marcados como stale '
            '(run=%s, kinds=%s)',
            updated, new_run.id, new_kinds,
        )
        return updated

"""
TagService — transição de estado de tags com histórico imutável.

REGRA ARQUITETURAL
------------------
Toda mudança em `InsightItem.review_state` para itens de kind='tag'
DEVE passar por este service. Ele grava uma linha em
`InsightTagHistory` a cada transição, preservando quem, quando e
qual era o estado anterior.

Insights de outros kinds podem usar `review_state` diretamente —
mas a diretriz de histórico só se aplica a tags.
"""

import logging
from typing import Optional

from django.db import transaction

from apps.intelligence.models import InsightItem, InsightTagHistory
from apps.intelligence.exceptions import IntelligenceError

logger = logging.getLogger('intelligence')


class InvalidTagTransitionError(IntelligenceError):
    """Transição de estado de tag inválida."""
    pass


class TagService:

    # Transições permitidas. A tag nasce em 'new'; pode ser confirmada
    # ou descartada; uma decisão pode ser revertida (revisão humana),
    # mas só com histórico explícito.
    ALLOWED = {
        (InsightTagHistory.STATE_NEW, InsightTagHistory.STATE_CONFIRMED),
        (InsightTagHistory.STATE_NEW, InsightTagHistory.STATE_DISCARDED),
        (InsightTagHistory.STATE_CONFIRMED, InsightTagHistory.STATE_DISCARDED),
        (InsightTagHistory.STATE_DISCARDED, InsightTagHistory.STATE_CONFIRMED),
    }

    @staticmethod
    @transaction.atomic
    def transition(
        *,
        insight: InsightItem,
        new_state: str,
        changed_by=None,
        reason: str = '',
    ) -> InsightItem:
        """
        Aplica uma transição de estado com histórico.

        Rejeita:
            - insights que não sejam kind='tag';
            - new_state fora de STATE_CHOICES;
            - transição não permitida (ex: confirmed → new);
            - no-op (mesmo estado).
        """
        if insight.kind != InsightItem.KIND_TAG:
            raise InvalidTagTransitionError(
                'Apenas itens de kind="tag" possuem histórico de estado.'
            )

        valid_states = {c[0] for c in InsightTagHistory.STATE_CHOICES}
        if new_state not in valid_states:
            raise InvalidTagTransitionError(
                f'Estado inválido: {new_state}.'
            )

        previous = insight.review_state
        if previous == new_state:
            raise InvalidTagTransitionError(
                f'A tag já está no estado "{new_state}".'
            )

        if (previous, new_state) not in TagService.ALLOWED:
            raise InvalidTagTransitionError(
                f'Transição não permitida: {previous} → {new_state}.'
            )

        InsightTagHistory.objects.create(
            insight=insight,
            previous_state=previous,
            new_state=new_state,
            changed_by=(
                changed_by
                if (changed_by and getattr(changed_by, 'is_authenticated', False))
                else None
            ),
            reason=reason or '',
        )

        insight.review_state = new_state
        insight.save(update_fields=['review_state', 'updated_at'])

        logger.info(
            'Tag transicionada: insight=%s %s→%s by=%s',
            insight.id, previous, new_state,
            getattr(changed_by, 'email', None) or 'system',
        )
        return insight

    # ------------------------------------------------------------------
    # CONSULTAS
    # ------------------------------------------------------------------

    @staticmethod
    def history_for(insight: InsightItem):
        """Histórico completo da tag, mais recente primeiro."""
        if insight.kind != InsightItem.KIND_TAG:
            return InsightTagHistory.objects.none()
        return (
            InsightTagHistory.objects
            .filter(insight=insight)
            .select_related('changed_by')
            .order_by('-created_at')
        )

    @staticmethod
    def list_tags(
        *,
        workspace,
        client=None,
        state: Optional[str] = None,
    ):
        """
        Lista tags do workspace (opcionalmente filtradas por cliente
        e/ou estado atual).
        """
        qs = (
            InsightItem.objects
            .filter(workspace=workspace, kind=InsightItem.KIND_TAG)
            .select_related('client', 'briefing')
            .order_by('-created_at')
        )
        if client is not None:
            qs = qs.filter(client=client)
        if state is not None:
            qs = qs.filter(review_state=state)
        return qs

def test_claim_next_job_is_exclusive(analysis_job):

    """

@pytest.mark.skipif(
    not _running_on_postgres(),
    reason=(
        'Teste real de concorrência requer Postgres. '
        'SQLite em memória não compartilha o banco entre threads. '
        'Rode no CI com Postgres habilitado.'
    ),
)
def test_claim_skips_locked_rows(analysis_job, django_db_blocker):
    """
    Simula um worker que segura o lock. Um segundo worker rodando
    em outra conexão (thread) deve usar `skip_locked` e retornar
    None em vez de bloquear.
    """
    import threading
    import time
    results = []

    def hold_lock():
        from django.db import transaction
        with django_db_blocker.unblock():
            with transaction.atomic():
                AnalysisJob.objects.select_for_update().filter(
                    id=analysis_job.id
                ).first()
                time.sleep(1.5)

    def try_claim():
        from django.db import connection
        with django_db_blocker.unblock():
            try:
                j = JobService.claim_next_job()
                results.append(j)
            finally:
                connection.close()

    t1 = threading.Thread(target=hold_lock)
    t1.start()
    time.sleep(0.3)
    t2 = threading.Thread(target=try_claim)
    t2.start()
    t1.join()
    t2.join()

    # O segundo worker deve ter retornado None (skip_locked)
    assert len(results) == 1
    assert results[0] is None
