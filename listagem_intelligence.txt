Endpoints:

def _resolve_workspace(request):
    ws_id = request.session.get('current_workspace_id')
    if not ws_id:
        return None
    return WorkspaceQueries.get_for_user(ws_id, request.user)


def _require(workspace, user, capability: str):
    if workspace is None:
        return False, Response(
            {'error': 'Nenhum workspace ativo'},
            status=status.HTTP_400_BAD_REQUEST,
        )
    if not user_has_capability(workspace, user, capability):
        return False, Response(
            {'error': f'Capability necessária: {capability}'},
            status=status.HTTP_403_FORBIDDEN,
        )
    return True, None


class MesaView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.insight.view',
        )
        if not ok:
            return err

        client = None
        client_id = request.query_params.get('client')
        if client_id:
            from apps.clients.models import Client
            client = Client.objects.filter(
                id=client_id, workspace=workspace,
            ).first()
            if client is None:
                return Response(
                    {'error': 'Cliente não encontrado'},
                    status=status.HTTP_404_NOT_FOUND,
                )

        include_stale = request.query_params.get('include_stale') == '1'
        payload = ContainerService.build_mesa(
            workspace=workspace,
            client=client,
            include_stale=include_stale,
        )
        serializer = ContainerPayloadSerializer(payload)
        return Response(serializer.data)

class TagDetailView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request, tag_id):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.insight.view',
        )
        if not ok:
            return err

        tag = InsightItem.objects.filter(
            id=tag_id, workspace=workspace, kind=InsightItem.KIND_TAG,
        ).first()
        if tag is None:
            return Response(
                {'error': 'Tag não encontrada'},
                status=status.HTTP_404_NOT_FOUND,
            )

        history = TagService.history_for(tag)
        return Response({
            'tag': InsightItemSerializer(tag).data,
            'history': InsightTagHistorySerializer(history, many=True).data,
        })

    def patch(self, request, tag_id):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.tag.manage',
        )
        if not ok:
            return err

        tag = InsightItem.objects.filter(
            id=tag_id, workspace=workspace, kind=InsightItem.KIND_TAG,
        ).first()
        if tag is None:
            return Response(
                {'error': 'Tag não encontrada'},
                status=status.HTTP_404_NOT_FOUND,
            )

        serializer = TagTransitionSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        try:
            tag = TagService.transition(
                insight=tag,
                new_state=serializer.validated_data['new_state'],
                changed_by=request.user,
                reason=serializer.validated_data.get('reason', ''),
            )
        except InvalidTagTransitionError as exc:
            return Response(
                {'error': str(exc)},
                status=status.HTTP_400_BAD_REQUEST,
            )

        return Response(InsightItemSerializer(tag).data)

class _ReanalyzeBase(APIView):
    """
    Base para as views de reanálise manual.

    Cada clique gera um InsightEvent novo com
    `source_kind=manual.reanalyze` e `occurrence_id` = timestamp
    monotônico. Isso evita que a idempotência por `dedupe_key`
    bloqueie cliques repetidos legítimos — o operador pode pedir
    reanálise quantas vezes quiser.

    Devolve `job_id` para o frontend acompanhar o andamento.
    """

    permission_classes = [IsAuthenticated]
    capability = 'intelligence.insight.view'

    def _ingest(self, request, workspace, client_id, briefing_id=None):
        occurrence_id = (
            f'manual-{timezone.now().timestamp()}-{request.user.id}'
        )
        data_version = f'manual-{timezone.now().isoformat()}'

        event, job, created = IngestService.ingest_change_event(
            workspace_id=workspace.id,
            client_id=client_id,
            briefing_id=briefing_id,
            source_kind=InsightEvent.SOURCE_KIND_MANUAL_REANALYZE,
            occurrence_id=occurrence_id,
            data_version=data_version,
        )

        if event is None or job is None:
            return Response(
                {'error': 'Falha ao agendar reanálise'},
                status=status.HTTP_500_INTERNAL_SERVER_ERROR,
            )

        return Response(
            {
                'job_id': str(job.id),
                'created': created,
                'event_id': str(event.id),
            },
            status=status.HTTP_200_OK,
        )

    """
    GET /api/intelligence/jobs/<job_id>/status/

    Devolve o estado atual do job + se o run associado concluiu.
    Usado pelo frontend para saber quando recarregar os containers.

    Retorna:
        {
            "job_id": str,
            "status": "pending"|"processing"|"completed"|"failed",
            "run_id": str | None,
            "run_status": str | None,
            "has_insights": bool,
            "error_message": str,
        }
    """

    def get(self, request, job_id):
        workspace = _resolve_workspace(request)
        ok, err = _require(
            workspace, request.user, 'intelligence.insight.view',
        )
        if not ok:
            return err

        job = AnalysisJob.objects.filter(
            id=job_id, workspace=workspace,
        ).first()
        if job is None:
            return Response(
                {'error': 'Job não encontrado'},
                status=status.HTTP_404_NOT_FOUND,
            )

        run = job.runs.order_by('-created_at').first()
        has_insights = bool(run and run.insights.exists())

        return Response({
            'job_id': str(job.id),
            'status': job.status,
            'run_id': str(run.id) if run else None,
            'run_status': run.status if run else None,
            'has_insights': has_insights,
            'error_message': job.last_error_message or '',
        })

"""
AppConfig do módulo intelligence.

INICIALIZAÇÃO AUTOMÁTICA DO WORKER
==================================
O worker de análise (`intelligence_worker`) precisa rodar continuamente
para processar os `AnalysisJob` pendentes. Este AppConfig inicia o
worker em uma thread daemon sempre que a aplicação Django sobe —
sem exigir terminal, sem exigir systemd, sem exigir NSSM.

Isso funciona em QUALQUER ambiente que execute Python:

    - DESENVOLVIMENTO (`python manage.py runserver`)
    - PRODUÇÃO LINUX (`gunicorn`, `uWSGI`)
    - PRODUÇÃO WINDOWS (`waitress`, `runserver` em prod)
    - PaaS (Render, Railway, Heroku, Fly.io)
    - Contêiner (Docker, se um dia você usar)

CONTROLE POR VARIÁVEIS DE AMBIENTE
==================================
O comportamento é controlado por 3 variáveis de ambiente. Nenhuma
delas é obrigatória — o default é "worker embutido ligado", que é o
que funciona em todos os ambientes sem configuração.

    INTELLIGENCE_WORKER_DISABLED
        "1"  → NÃO inicia o worker neste processo.
               Use quando o worker rodar em processo separado
               (systemd, NSSM, "Background Worker" de PaaS).
        ""   → inicia (default).

    INTELLIGENCE_WORKER_MODE
        "auto"     → default. Inicia o worker em TODOS os processos
                     que não forem comandos de management
                     (`migrate`, `shell`, `test`, etc.).
                     Em gunicorn com N workers, você terá N workers
                     embutidos. Seguro pela lógica de
                     `select_for_update(skip_locked=True)`, mas
                     consome CPU proporcional.
        "single"   → inicia o worker apenas em UM processo por
                     instância. Use a variável
                     INTELLIGENCE_WORKER_LEADER=1 para marcar QUAL
                     processo é o líder. Os demais não iniciam.
                     Ideal para gunicorn onde você quer 1 worker
                     embutido e não quer gerenciar processo separado.
        "off"      → equivalente a INTELLIGENCE_WORKER_DISABLED=1.

    INTELLIGENCE_WORKER_LEADER
        "1"  → este processo é o líder (só relevante em mode=single).
        ""   → não é líder.

SEGURANÇA
=========
    - A thread é `daemon=True`: morre junto com o processo principal.
    - Exceções dentro do worker são logadas, nunca propagadas.
    - `ready()` nunca bloqueia o startup do Django.
    - `_should_start_worker()` evita iniciar worker em comandos de
      management como `migrate`, `makemigrations`, `shell`, `test`,
      `collectstatic`, `createsuperuser`, `seed_*`, etc.

REAPROVEITAMENTO
================
O loop do worker é o `Command.handle()` do management command
`intelligence_worker`. Fonte única de verdade do processamento de
jobs. Não duplica lógica.
"""

import logging
import os
import sys
import threading
import time

logger = logging.getLogger('intelligence')


# Intervalo entre polls do worker (segundos).
_WORKER_INTERVAL_SECONDS = 5

# Delay inicial antes do primeiro poll. Dá tempo para o Django
# terminar de montar o app registry, as conexões de banco e o
# middleware. Sem isso, a primeira query pode falhar.
_WORKER_STARTUP_DELAY_SECONDS = 2

def _env_flag(name: str) -> bool:
    """Lê uma env var booleana ('1'/'true'/'yes' = True)."""
    raw = os.environ.get(name, '').strip().lower()
    return raw in ('1', 'true', 'yes', 'on')


def _worker_disabled() -> bool:
    """
    Worker desativado explicitamente neste processo.

    Duas formas equivalentes:
        - INTELLIGENCE_WORKER_DISABLED=1
        - INTELLIGENCE_WORKER_MODE=off
    """
    if _env_flag('INTELLIGENCE_WORKER_DISABLED'):
        return True
    mode = os.environ.get(
        'INTELLIGENCE_WORKER_MODE', 'auto',
    ).strip().lower()
    if mode == 'off':
        return True
    return False


def _worker_mode() -> str:
    """
    Modo do worker: 'auto', 'single' ou 'off'.

    Default: 'auto'.
    """
    raw = os.environ.get(
        'INTELLIGENCE_WORKER_MODE', 'auto',
    ).strip().lower()
    if raw in ('auto', 'single', 'off'):
        return raw
    logger.warning(
        'INTELLIGENCE_WORKER_MODE inválido: %r. Usando "auto".',
        raw,
    )
    return 'auto'


def _is_leader_process() -> bool:
    """
    True apenas quando este processo é o líder (mode=single).

    Em mode=auto, é sempre True (todos iniciam).
    Em mode=single, é True só no processo com
    INTELLIGENCE_WORKER_LEADER=1.
    """
    mode = _worker_mode()
    if mode == 'auto':
        return True
    if mode == 'single':
        return _env_flag('INTELLIGENCE_WORKER_LEADER')
    return False


def _is_runserver_autoreload_child() -> bool:
    """
    Detecta o processo filho do autoreload do `runserver`.

    Em desenvolvimento, `runserver` inicia DOIS processos. O
    `ready()` roda em ambos. Sem essa checagem, o worker subiria
    duas vezes no dev.
    """
    if 'runserver' not in sys.argv:
        return False
    return os.environ.get('RUN_MAIN') != 'true'


def _is_management_command_that_should_not_run_worker() -> bool:
    """
    True quando o processo atual é um `manage.py <comando>` que NÃO
    é `runserver` e NÃO deve iniciar o worker.

    Exemplos: migrate, makemigrations, shell, test, collectstatic,
    createsuperuser, seed_*, sync_*, bootstrap_console, etc.

    Exceção: `runserver` — inicia worker.
    """
    argv = sys.argv or []
    if not any('manage.py' in arg for arg in argv):
        return False
    if 'runserver' in argv:
        return False
    return True

class IntelligenceConfig(AppConfig):
    default_auto_field = 'django.db.models.BigAutoField'
    name = 'apps.intelligence'
    verbose_name = _('Inteligência')

    def ready(self):
        """
        Inicializa o módulo intelligence.

        - Não registra signals (ingestão é explícita).
        - Inicia o worker embutido quando aplicável.
        """
        if _is_runserver_autoreload_child():
            return

        if _is_management_command_that_should_not_run_worker():
            return

        if _worker_disabled():
            logger.info(
                'Worker de Inteligência desativado neste processo '
                '(INTELLIGENCE_WORKER_DISABLED=1 ou '
                'INTELLIGENCE_WORKER_MODE=off).',
            )
            return

        if not _is_leader_process():
            logger.debug(
                'Worker de Inteligência não iniciado neste processo '
                '(mode=single e INTELLIGENCE_WORKER_LEADER!=1).',
            )
            return

        self._start_worker_thread()

    # ------------------------------------------------------------------
    # THREAD
    # ------------------------------------------------------------------

    @staticmethod
    def _start_worker_thread() -> None:
        """
        Inicia o worker em uma thread daemon.

        Idempotente por processo.
        """
        if getattr(
            IntelligenceConfig, '_worker_thread_started', False,
        ):
            return
        IntelligenceConfig._worker_thread_started = True

        thread = threading.Thread(
            target=IntelligenceConfig._run_worker_loop,
            name='intelligence-worker',
            daemon=True,
        )
        thread.start()

        logger.info(
            'Worker de Inteligência iniciado em thread daemon '
            '(mode=%s, interval=%ss, startup_delay=%ss).',
            _worker_mode(),
            _WORKER_INTERVAL_SECONDS,
            _WORKER_STARTUP_DELAY_SECONDS,
        )

    @staticmethod
    def _run_worker_loop() -> None:
        """
        Loop principal do worker.

        Reaproveita `Command.handle()` do management command.
        """
        time.sleep(_WORKER_STARTUP_DELAY_SECONDS)

        try:
            from apps.intelligence.management.commands.intelligence_worker import (
                Command as WorkerCommand,
            )
        except Exception:
            logger.exception(
                'Falha ao importar o management command do worker. '
                'O worker NÃO será iniciado.',
            )
            return

        while True:
            try:
                command = WorkerCommand()
                command.handle(
                    interval=_WORKER_INTERVAL_SECONDS,
                    run_once=False,
                )
                logger.warning(
                    'Worker retornou de handle() inesperadamente. '
                    'Encerrando thread.',
                )
                return
            except KeyboardInterrupt:
                logger.info('Worker interrompido por KeyboardInterrupt.')
                return
            except Exception:
                logger.exception(
                    'Exceção no loop do worker. Reiniciando em %ss.',
                    _WORKER_INTERVAL_SECONDS,
                )
                time.sleep(_WORKER_INTERVAL_SECONDS)

"""
Comando de gerenciamento para processar jobs de inteligência.

CONCORRÊNCIA
------------
Cada job é reivindicado via `JobService.claim_next_job()`, que usa
`select_for_update(skip_locked=True)` E marca o job como `processing`
dentro da mesma transação. Dois workers concorrentes nunca processam
o mesmo job.
"""

import logging
import time

from django.core.management.base import BaseCommand

from apps.intelligence.services import JobService, AnalysisService
from apps.intelligence.models import AnalysisJob, AnalysisRun
from apps.intelligence.exceptions import BudgetExceededError, StaleAnalysisError

logger = logging.getLogger('intelligence')


class Command(BaseCommand):
    help = 'Processa os jobs de análise de inteligência pendentes.'

    def add_arguments(self, parser):
        parser.add_argument('--interval', type=int, default=5)
        parser.add_argument('--run-once', action='store_true')

    def handle(self, *args, **options):
        run_once = options['run_once']
        interval = options['interval']

        self.stdout.write(self.style.SUCCESS('Worker de Inteligência iniciado.'))

        while True:
            try:
                job = JobService.claim_next_job()

                if job is None:
                    if run_once:
                        break
                    time.sleep(interval)
                    continue

                self.process_job(job)

            except KeyboardInterrupt:
                self.stdout.write(self.style.WARNING('\nWorker interrompido.'))
                break
            except Exception:
                logger.exception('Erro no loop principal do worker.')
                if run_once:
                    raise
                time.sleep(interval)

        self.stdout.write(self.style.SUCCESS('Worker finalizado.'))

    def process_job(self, job: AnalysisJob):
        job_id_str = str(job.id)
        self.stdout.write(f'Processando job {job_id_str}...')

        try:
            run = AnalysisService.execute(job)

            if run is None:
                self.stdout.write(self.style.WARNING(
                    f'Job {job_id_str} ignorado (duplicado).'
                ))
                JobService.mark_as_completed(job)
                return

            if run.status == AnalysisRun.STATUS_COMPLETED:
                self.stdout.write(self.style.SUCCESS(
                    f'Job {job_id_str} concluído (Run {run.id}).'
                ))
                JobService.mark_as_completed(job)
            else:
                JobService.mark_as_failed(job, f'Estado inesperado: {run.status}')

        except StaleAnalysisError as exc:
            self.stdout.write(self.style.WARNING(
                f'Job {job_id_str} ignorado (stale): {exc}'
            ))
            JobService.mark_as_completed(job)

        except BudgetExceededError as exc:
            self.stdout.write(self.style.ERROR(
                f'Job {job_id_str} falhou (orçamento): {exc}'
            ))
            JobService.mark_as_failed(job, f'BudgetExceeded: {exc}')

        except Exception as exc:
            self.stdout.write(self.style.ERROR(
                f'Job {job_id_str} falhou: {exc}'
            ))
            logger.exception('Falha ao processar job %s', job_id_str)
            JobService.mark_as_failed(job, str(exc))

"""
Alocação de custo de uma execução de IA aos containers/insights
que ela produziu.

REGRA ARQUITETURAL
------------------
`UsageRecord` (em `console`) registra o custo REAL da chamada ao
provedor. Este modelo registra a PARCELA desse custo atribuída a
cada `InsightItem` produzido, segundo uma regra de rateio explícita.

Separação de responsabilidades:
    - console.UsageRecord   → quanto foi cobrado pelo provedor.
    - InsightCostAllocation → como esse custo se distribui entre
                              os insights gerados.

A regra de rateio é determinística e documentada em
`CostAllocationService.allocate_for_run()`. Este modelo apenas
persiste o resultado.

Semântica dos campos:
    - `allocated_cost = None`   → custo desconhecido (não foi possível
                                  calcular, nem por rateio). O chamador
                                  deve tratar como "indisponível", nunca
                                  como zero.
    - `is_estimated = True`     → o valor é resultado de rateio, não
                                  de uma chamada individual. O campo
                                  `allocation_rule` documenta a regra.
"""

from django.utils.translation import gettext_lazy as _

        Usado por serializers/views para filtrar sem duplicar a verdade.

        mapping = cls.CONTAINER_KINDS.get(container)
        if mapping is None:
            raise ValueError(f'Container desconhecido: {container}')
        result = set()
        for kinds in mapping.values():
            result.update(kinds)
        return sorted(result)

    @classmethod
    def container_slots_for_kind(cls, kind: str) -> dict:
        """
        Retorna, para um `kind`, quais slots de container o exibem.

        Estrutura:
            {
                'mesa': ['01_desejos_prioridades'],
                'relatorio': ['02_desejos_criterios'],

        aparece na Mesa e no Relatório). Este método é o inverso do
        CONTAINER_KINDS e evita duplicação de lógica nas views.

        result = {}
        for container, slots in cls.CONTAINER_KINDS.items():
            matching = [
                slot_name for slot_name, kinds in slots.items()
                if kind in kinds
            ]
            if matching:
                result[container] = matching
        return result

class AnalysisService:

    @staticmethod
    def execute(job: AnalysisJob) -> Optional[AnalysisRun]:
        run = None
        try:
            with transaction.atomic():
                run = AnalysisService._create_analysis_run(job)
                if run is None:
                    return None

            from apps.console import public as console_public
            allowed, reason = console_public.is_ai_execution_allowed(job.workspace)
            if not allowed:
                raise BudgetExceededError(
                    f'Execução bloqueada pela política de quota: {reason}'
                )

            input_data = AnalysisService._prepare_input(run)
            if not input_data:
                raise AnalysisInputError('Falha ao preparar os dados de entrada.')

            ai_response = AnalysisService._call_ai_via_console(run, input_data)

            insights_data = AnalysisService._parse_and_validate_ai_response(
                ai_response,
                target_kinds=AnalysisService._resolve_target_kinds(run),
            )

            with transaction.atomic():
                AnalysisService._apply_ai_results(
                    run=run,
                    ai_response=ai_response,
                    insights_data=insights_data,
                    duration_ms=ai_response.get('duration_ms'),
                )

            try:
                from apps.intelligence.services.cost_allocation_service import (
                    CostAllocationService,
                )
                CostAllocationService.allocate_for_run(run, ai_response)
            except Exception:
                logger.exception(
                    'Falha ao alocar custo (run=%s). Insights persistidos.',
                    run.id,
                )

            try:
                from apps.intelligence.services.staleness_service import (
                    StalenessService,
                )
                StalenessService.mark_previous_runs_stale(run)
            except Exception:
                logger.exception(
                    'Falha ao marcar runs anteriores como stale (run=%s).',
                    run.id,
                )

            return run

        except StaleAnalysisError:
            if run is not None:
                with transaction.atomic():
                    run.status = AnalysisRun.STATUS_SKIPPED_STALE
                    run.is_stale = True
                    run.completed_at = timezone.now()
                    run.save(update_fields=[
                        'status', 'is_stale', 'completed_at', 'updated_at',
                    ])
            raise

        except Exception as exc:
            logger.exception(
                'Falha na execução da análise (Run=%s)',
                run.id if run else '<none>',
            )
            if run is not None:
                with transaction.atomic():
                    run.status = AnalysisRun.STATUS_FAILED
                    run.error_message = str(exc)
                    run.completed_at = timezone.now()
                    run.save(update_fields=[
                        'status', 'error_message', 'completed_at', 'updated_at',
                    ])
            raise

            explicit_model or 'gpt-4o-mini',
            prompt_version,
        )

    @staticmethod
    def _create_analysis_run(job: AnalysisJob) -> Optional[AnalysisRun]:
        """
        Cria um `AnalysisRun` NOVO para esta tentativa do job.

        O `input_hash` inclui `job.attempt_count` (incrementado pelo
        `JobService.claim_next_job` antes de chamar `execute`). Isso
        garante que cada tentativa cria um run distinto:

            - Tentativa 1: hash A → run 1
            - Tentativa 2: hash B → run 2
            - Tentativa 3: hash C → run 3

        Motivos:
            - O `created_at` do run é `auto_now_add=True`. Reaproveitar
              o run mantém o `created_at` ORIGINAL, o que faz o
              `StalenessService` sempre rejeitar como stale quando o
              evento `manual.reanalyze` é posterior ao run original.
            - O `request_id` (=run.id) muda a cada tentativa, então
              o `AICallService` faz uma chamada real ao provider em
              vez de devolver `raw_content=""` do `UsageRecord`
              antigo.

        Se já existe um run `COMPLETED` para ESTA tentativa (mesmo
        attempt_count), retorna `None` — idempotência real (o worker
        reivindicou o mesmo job duas vezes sem incrementar o
        attempt_count não deveria acontecer, mas é defesa em
        profundidade).

        Antes de criar o run novo, DELETA os `UsageRecord` e
        `ProviderLog` de runs anteriores do MESMO job — evita lixo no
        banco e garante que nenhum artefato antigo interfira na nova
        tentativa.
        """
        input_payload = AnalysisService._prepare_input(job)
        if not input_payload:
            raise AnalysisInputError('Falha ao preparar input para hashing.')

        provider, model, prompt_version = (
            AnalysisService._resolve_provider_and_model()
        )

        target_kinds = AnalysisService._resolve_target_kinds_from_job(job)
        payload_hash = hashlib.sha256(
            json.dumps(
                {'input': input_payload, 'kinds': sorted(target_kinds)},
                sort_keys=True, default=str,
            ).encode('utf-8')
        ).hexdigest()

        # `job.id` distingue jobs diferentes com o mesmo input.
        # `job.attempt_count` distingue tentativas do MESMO job.
        # Juntos, garantem hash único por tentativa — e, por
        # consequência, `created_at` sempre atual no run.
        composite = '|'.join([
            str(job.id),
            str(job.attempt_count),
            str(job.workspace_id),
            prompt_version,
            provider,
            model,
            payload_hash,
        ])
        input_hash = hashlib.sha256(composite.encode('utf-8')).hexdigest()

        # ------------------------------------------------------------------
        # 1. Se já existe run COMPLETED para este attempt_count → idempotente
        #
        #    Defesa em profundidade: `claim_next_job` incrementa
        #    `attempt_count` antes de chamar `execute`, então este
        #    cenário não deveria ocorrer. Mas se ocorrer, o run está
        #    completo e a IA já produziu os insights. Não roda de novo.
        # ------------------------------------------------------------------
        completed = AnalysisRun.objects.filter(
            input_hash=input_hash,
            status=AnalysisRun.STATUS_COMPLETED,
        ).first()
        if completed is not None:
            logger.info(
                'Run idêntico já concluído para este attempt (hash=%s). '
                'Ignorando.', input_hash,
            )
            return None

        # ------------------------------------------------------------------
        # 2. Limpa artefatos de runs anteriores do MESMO job.
        #
        #    Como cada tentativa cria um run novo, os `UsageRecord` e
        #    `ProviderLog` das tentativas anteriores ficam órfãos
        #    (apontando para `request_id` de runs antigos). Deletamos
        #    para manter o banco limpo e evitar que o `StalenessService`
        #    ou o `CostAllocationService` tropecem neles.
        # ------------------------------------------------------------------
        previous_runs = AnalysisRun.objects.filter(job=job)
        for prev_run in previous_runs:
            AnalysisService._cleanup_run_artifacts(prev_run)

        # ------------------------------------------------------------------
        # 3. Cria run novo
        # ------------------------------------------------------------------
        return AnalysisRun.objects.create(
            job=job,
            workspace=job.workspace,
            client=job.client,
            briefing=job.briefing,
            input_hash=input_hash,
            data_version=job.data_version,
            provider=provider,
            model=model,
            prompt_version=prompt_version,
            status=AnalysisRun.STATUS_RUNNING,
            started_at=timezone.now(),
        )

    @staticmethod
    def _cleanup_run_artifacts(run: AnalysisRun) -> None:
        """
        Remove `UsageRecord` e `ProviderLog` de um run anterior.

        Chamado antes de criar um run novo para a mesma tentativa
        de job. Não levanta — falhas de limpeza não impedem a nova
        tentativa.

        Não mexe em `InsightItem` — esses são artefatos do run
        anterior, mas o `StalenessService` marca como stale
        apropriadamente quando o run novo conclui. Deletá-los
        aqui quebraria a auditoria histórica.
        """
        try:
            from apps.console.models import UsageRecord, ProviderLog

            deleted_usage, _ = UsageRecord.objects.filter(
                provider__key=run.provider,
                request_id=str(run.id),
            ).delete()
            if deleted_usage:
                logger.info(
                    'UsageRecord antigo do run=%s deletado '
                    '(forçar chamada real).', run.id,
                )

            deleted_log, _ = ProviderLog.objects.filter(
                provider__key=run.provider,
                request_id=str(run.id),
            ).delete()
            if deleted_log:
                logger.info(
                    'ProviderLog antigo do run=%s deletado.', run.id,
                )
        except Exception:
            logger.exception(
                'Falha ao limpar artefatos do run=%s. Prosseguindo.',
                run.id,
            )

    @staticmethod

        valid = {c[0] for c in InsightItem.KIND_CHOICES}
        filtered = [k for k in requested if k in valid]
        if filtered:

        if job.briefing_id:

        return list(DEFAULT_KINDS_WITHOUT_BRIEFING)

    @staticmethod
    def _prepare_input(job_or_run) -> Optional[Dict[str, Any]]:
        from apps.briefings import public as briefings_public
        from apps.clients import public as clients_public

        client_snap = clients_public.get_client_snapshot(job_or_run.client_id)
        if not client_snap:
            raise AnalysisInputError('Cliente não encontrado.')

        payload = {'client': client_snap}

        if job_or_run.briefing_id:
            briefing_snap = briefings_public.get_briefing_snapshot(
                job_or_run.briefing_id
            )
            if not briefing_snap:
                raise AnalysisInputError('Briefing não encontrado.')
            payload['briefing'] = briefing_snap

            images = AnalysisService._collect_images(
                briefings_public, briefing_snap,
            )
            if images:
                payload['images'] = images

        return payload

    @staticmethod
    def _collect_images(briefings_public, briefing_snap: dict) -> List[Dict]:
        getter = getattr(
            briefings_public, 'get_reference_document_payloads', None,
        )
        if getter is None:
            logger.info(
                'briefings.public.get_reference_document_payloads '
                'indisponível. Análise será apenas textual.'
            )
            return []

        doc_ids = set()
        for answer in briefing_snap.get('answers', []):
            doc_ids.update(answer.get('attachments') or [])
            if answer.get('audio_document_id'):
                doc_ids.add(answer['audio_document_id'])

        for section in (briefing_snap.get('structure_snapshot') or {}).get(
            'sections', []
        ):
            for q in section.get('questions', []):
                for rid in q.get('reference_document_ids') or []:
                    doc_ids.add(rid)

        if not doc_ids:
            return []

        try:
            payloads = getter(list(doc_ids)) or []
        except Exception:
            logger.exception(
                'Falha ao obter payloads de documentos. '
                'Análise será apenas textual.'
            )
            return []

        result = []
        for p in payloads:
            if not isinstance(p, dict):
                continue
            url = p.get('url')
            mime = p.get('mime_type') or ''
            if not url or not mime.startswith('image/'):
                continue
            result.append({
                'document_id': str(p.get('id') or ''),
                'url': url,
                'mime_type': mime,
                'source': p.get('source') or 'briefing',
            })
        return result

    @staticmethod
    def _call_ai_via_console(run: AnalysisRun, input_data: dict) -> dict:
        from apps.console import public as console_public

        call_ai = getattr(console_public, 'call_ai', None)
        if call_ai is None:
            raise AIProviderError(
                'A ponte console.public.call_ai não está disponível. '
                'A inteligência não pode executar IA sem passar pelo '
                'Console (governança + custo).'
            )

        system_prompt = get_prompt()

        target_kinds = AnalysisService._resolve_target_kinds(run)
        kinds_list = '\n'.join(f'  - "{k}"' for k in target_kinds)

            "============================================================================\n"
            "KINDS A PRODUZIR NESTA EXECUÇÃO (use SOMENTE estes):\n"
            "============================================================================\n"
            f"{kinds_list}\n\n"
            "Se a análise não tiver evidência suficiente para um kind\n"
            "que exige evidência, NÃO invente — produza apenas os kinds\n"
            "que conseguir sustentar.\n\n"
            "============================================================================\n"
            "DADOS:\n"
            "============================================================================\n"
            f"{json.dumps(input_data, ensure_ascii=False, indent=2, default=str)}"
        )

        started = time.time()
        try:
            response = call_ai(
                workspace=run.workspace,
                operation='intelligence.analysis',
                provider_key=run.provider,
                model=run.model,
                prompt_version=run.prompt_version,
                messages=[
                    {'role': 'system', 'content': system_prompt},
                    {'role': 'user', 'content': user_message},

                images=input_data.get('images') or [],
                response_format='json_object',
                request_id=str(run.id),
            )
        except Exception as exc:
            duration_ms = int((time.time() - started) * 1000)
            logger.exception(
                'Falha na ponte call_ai (run=%s, duration=%sms)',
                run.id, duration_ms,
            )
            raise AIProviderError(f'Falha na chamada da IA: {exc}') from exc

        duration_ms = int((time.time() - started) * 1000)

        return {
            'raw_content': response.get('raw_content', ''),
            'input_tokens': int(response.get('input_tokens') or 0),
            'output_tokens': int(response.get('output_tokens') or 0),
            'duration_ms': response.get('duration_ms', duration_ms),
            'usage_record_id': response.get('usage_record_id'),
            'provider_request_id': response.get('provider_request_id') or '',
        }

    @staticmethod
    def _parse_and_validate_ai_response(
        ai_response: dict,
        target_kinds: Optional[List[str]] = None,
    ) -> List[Dict]:
        raw = ai_response.get('raw_content') or ''

        if not raw.strip():
            logger.error(
                'Resposta da IA veio VAZIA. Isso geralmente significa '
                'que o `AICallService` devolveu `raw_content=""` '
                'porque o `UsageRecord` já existia (idempotência) e '
                'o `metadata["raw_content"]` não foi persistido. '
                'Solução: deletar o UsageRecord antigo com '
                '`request_id=<run.id>` e tentar de novo.'
            )
            raise AIResponseError(
                'Resposta da IA veio vazia. O `UsageRecord` antigo '
                'pode ter sido reaproveitado sem `raw_content`. '
                'Delete o `UsageRecord` do run e tente novamente.'
            )

        try:
            data = json.loads(raw)
        except json.JSONDecodeError as exc:
            logger.error(
                'Resposta da IA não é JSON. raw_content=%r '
                '(primeiros 500 chars)',
                raw[:500],
            )
            raise AIResponseError(f'Resposta da IA não é JSON válido: {exc}')

        if not isinstance(data, dict) or 'insights' not in data:
            logger.error(
                'Resposta da IA não tem chave "insights". '
                'chaves=%r, raw=%r',
                list(data.keys()) if isinstance(data, dict) else type(data).__name__,
                raw[:500],
            )
            raise AIResponseError("Resposta da IA não contém a chave 'insights'.")

        insights = data['insights']
        if not isinstance(insights, list):
            raise AIResponseError("'insights' deve ser uma lista.")

        valid_kinds = {c[0] for c in InsightItem.KIND_CHOICES}
        valid_confidences = {c[0] for c in InsightItem.CONFIDENCE_CHOICES}
        allowed = set(target_kinds) if target_kinds else valid_kinds

        validated = []
        for i, item in enumerate(insights):
            if not isinstance(item, dict):
                logger.error(
                    'Insight #%d não é dict. Recebido: %r', i, item,
                )
                raise AIResponseError(f'Insight #{i} não é um objeto.')

            kind = item.get('kind')
            content = (item.get('content') or '').strip()
            confidence = item.get(
                'confidence', InsightItem.CONFIDENCE_INTERPRETATION,
            )

            if kind not in valid_kinds:
                logger.error(
                    'Insight #%d: kind inválido. '
                    'Recebido=%r. Item completo=%r',
                    i, kind, item,
                )
                raise AIResponseError(f'Insight #{i}: kind inválido ({kind}).')

            if kind not in allowed:
                logger.error(
                    'Insight #%d: kind "%s" fora do target_kinds. '
                    'Permitidos=%r',
                    i, kind, sorted(allowed),
                )
                raise AIResponseError(
                    f'Insight #{i}: kind "{kind}" não solicitado nesta '
                    f'execução (target_kinds).'
                )

            if not content:
                raise AIResponseError(f'Insight #{i}: content vazio.')
            if confidence not in valid_confidences:
                logger.error(
                    'Insight #%d: confidence inválida. '
                    'Recebido=%r, permitidos=%r',
                    i, confidence, sorted(valid_confidences),
                )
                raise AIResponseError(
                    f'Insight #{i}: confidence inválida ({confidence}).'
                )

            evidences = item.get('evidences') or []
            if not isinstance(evidences, list):
                raise AIResponseError(f'Insight #{i}: evidences deve ser lista.')

            if kind in KINDS_REQUIRING_EVIDENCE and not evidences:
                logger.error(
                    'Insight #%d (%s): exige evidência, recebeu [].',
                    i, kind,
                )
                raise AIResponseError(
                    f'Insight #{i} ({kind}): evidência obrigatória ausente.'
                )

            validated.append({
                'kind': kind,
                'content': content,
                'confidence': confidence,
                'metadata': item.get('metadata') or {},
                'evidences': evidences,
            })

        return validated

    @staticmethod
    def _apply_ai_results(run, ai_response, insights_data, duration_ms):
        latest_event = (
            InsightEvent.objects
            .filter(
                workspace=run.workspace,
                client=run.client,
                briefing=run.briefing,
            )
            .order_by('-created_at')
            .first()

            raise StaleAnalysisError(
                'Existe um evento mais recente que o run. Não publicar.'
            )

        run.output_tokens = ai_response.get('output_tokens') or 0
        run.duration_ms = duration_ms
        run.status = AnalysisRun.STATUS_COMPLETED
        run.completed_at = timezone.now()
        run.save()

        for item in insights_data:
            insight = InsightItem.objects.create(
                run=run,
                workspace=run.workspace,
                client=run.client,
                briefing=run.briefing,
                kind=item['kind'],
                content=item['content'],

    @staticmethod
    def _persist_evidences(insight, raw_evidences, workspace):
        from apps.briefings import public as briefings_public
        from apps.clients import public as clients_public

        for ev in raw_evidences:
            if not isinstance(ev, dict):
                continue
            source_type = ev.get('source_type')
            source_ref = (ev.get('source_ref') or '').strip()
            quote = (ev.get('quote') or '').strip()

            if source_type not in dict(InsightEvidence.SOURCE_TYPE_CHOICES):
                logger.warning(
                    'Evidência com source_type inválido ignorada: %s',
                    source_type,
                )
                continue
            if not source_ref:
                logger.warning('Evidência sem source_ref ignorada.')
                continue

            valid = False
            if source_type == InsightEvidence.SOURCE_TYPE_BRIEFING_ANSWER:
                snap = (
                    briefings_public.get_briefing_snapshot(insight.briefing_id)
                    if insight.briefing_id else None
                )
                if snap and any(
                    a['id'] == source_ref for a in snap['answers']
                ):
                    valid = True
            elif source_type == InsightEvidence.SOURCE_TYPE_BRIEFING_NOTE:
                snap = (
                    briefings_public.get_briefing_snapshot(insight.briefing_id)
                    if insight.briefing_id else None
                )
                if snap and any(
                    n['id'] == source_ref for n in snap['notes']
                ):
                    valid = True
            elif source_type == InsightEvidence.SOURCE_TYPE_CLIENT_CONTEXT:
                snap = clients_public.get_client_snapshot(insight.client_id)
                if snap:
                    valid = True

            if not valid:
                logger.warning(
                    'Evidência ignorada (fonte inexistente ou de outro '
                    'workspace): %s:%s',
                    source_type, source_ref,
                )
                continue

            InsightEvidence.objects.get_or_create(
                insight=insight,
                source_type=source_type,
                source_ref=source_ref,
                defaults={'quote': quote},
            )

"""
ContainerService — monta a resposta dos containers fixos.

REGRA ARQUITETURAL
------------------
Os containers são estruturas FIXAS (Mesa, Cliente, Relatório). Este
service é a ÚNICA fonte de verdade para montar o payload de cada um.

NORMALIZAÇÃO DE `content`
-------------------------
O `content` de cada `InsightItem` deve seguir o contrato do
`prompt_loader.py`:

    - desire / attention / demais bullets:
        "afirmação — motivo", máx ~140 chars.
    - summary.briefing:
        texto analítico MÁXIMO 500 caracteres (truncado aqui, no
        service), sempre terminando em ponto final.
    - executive.summary:
        texto corrido, preservado na íntegra.

O truncamento do resumo em 500 caracteres acontece AQUI, para que
o CSS nunca precise cortar por linhas. O texto sempre termina em
ponto final (ou "…" se não houver ponto próximo do limite).
"""

import logging
import re
from typing import Dict, List, Optional

from apps.intelligence.models import InsightItem

logger = logging.getLogger('intelligence')


_SUBJECT_PREFIXES = [
    re.compile(r'^o\s+cliente\s+', re.IGNORECASE),
    re.compile(r'^a\s+cliente\s+', re.IGNORECASE),
    re.compile(r'^os\s+clientes\s+', re.IGNORECASE),
    re.compile(r'^as\s+clientes\s+', re.IGNORECASE),
    re.compile(r'^ele\s+', re.IGNORECASE),
    re.compile(r'^ela\s+', re.IGNORECASE),
    re.compile(r'^eles\s+', re.IGNORECASE),
    re.compile(r'^elas\s+', re.IGNORECASE),
    re.compile(r'^o\s+briefing\s+', re.IGNORECASE),
    re.compile(r'^a\s+briefing\s+', re.IGNORECASE),
    re.compile(r'^o\s+projeto\s+', re.IGNORECASE),
    re.compile(r'^a\s+análise\s+', re.IGNORECASE),
]

_TRAILING_NOISE = [
    re.compile(r'\s*\.\s*$'),
    re.compile(r'\s*\.\.\.\s*$'),
    re.compile(r'\s*…\s*$'),
    re.compile(r',?\s*etc\.?\s*$', re.IGNORECASE),
    re.compile(r',?\s*e\s+outros\.?\s*$', re.IGNORECASE),
    re.compile(r',?\s*entre\s+outros\.?\s*$', re.IGNORECASE),
]

# Tamanho máximo de um bullet curto na saída.
_MAX_BULLET_LENGTH = 140

# Tamanho máximo do resumo. Garante que o texto SEMPRE fecha e caiba
# no container sem corte no frontend.
_RESUMO_MAX_CHARS = 400


def _normalize_bullet(raw: str) -> str:
    """
    Normaliza um bullet curto (desire, attention, etc.).

    Idempotente. Aplica remoção de prefixos/sufixos e truncamento
    em ~140 chars.

    NÃO é usado para summary.briefing nem executive.summary.
    """
    if not raw:
        return ''

    text = raw.strip()

    changed = True
    while changed:
        changed = False
        for pat in _SUBJECT_PREFIXES:
            new_text = pat.sub('', text)
            if new_text != text:
                text = new_text.strip()
                changed = True

    changed = True
    while changed:
        changed = False
        for pat in _TRAILING_NOISE:
            new_text = pat.sub('', text)
            if new_text != text:
                text = new_text.strip()
                changed = True

    if text and text[0].isalpha() and text[0].islower():
        text = text[0].upper() + text[1:]

    if len(text) > _MAX_BULLET_LENGTH:
        text = text[: _MAX_BULLET_LENGTH - 1].rstrip() + '…'

    return text


def _truncate_resumo(raw: str) -> str:
    """
    Trunca o resumo em `_RESUMO_MAX_CHARS` caracteres.

    Regras:
        - Se já cabe → devolve como veio (só strip).
        - Se precisa cortar → tenta cortar no último ponto final
          antes do limite (para o texto terminar de forma natural).
        - Se não houver ponto nos últimos 30% do texto → corta no
          limite e adiciona "…".

    Idempotente: rodar duas vezes dá o mesmo resultado.
    """
    if not raw:
        return ''

    text = raw.strip()

    if len(text) <= _RESUMO_MAX_CHARS:
        return text

    head = text[:_RESUMO_MAX_CHARS]

    # Procura o último ponto final nos últimos 30% do head.
    safe_floor = int(_RESUMO_MAX_CHARS * 0.7)
    last_dot = head.rfind('.')

    if last_dot >= safe_floor:
        return head[: last_dot + 1].rstrip()

    return head[: _RESUMO_MAX_CHARS - 1].rstrip() + '…'


def _normalize_content(raw: str, kind: str) -> str:
    """
    Decide a normalização com base no kind.

    - summary.briefing → trunca em 500 chars, termina em ponto.
    - executive.summary → preserva (texto corrido, sem truncar).
    - demais           → normaliza como bullet curto.
    """
    if not raw:
        return ''

    if kind == InsightItem.KIND_SUMMARY_BRIEFING:
        return _truncate_resumo(raw)

    if kind == InsightItem.KIND_EXECUTIVE_SUMMARY:
        return raw.strip()

    return _normalize_bullet(raw)


def _serialize_item(item: InsightItem) -> Dict:
    return {
        'id': str(item.id),
        'kind': item.kind,
        'kind_display': item.get_kind_display(),
        'confidence': item.confidence,
        'confidence_display': item.get_confidence_display(),
        'content': _normalize_content(item.content, item.kind),
        'metadata': item.metadata or {},
        'review_state': item.review_state,
        'is_stale': item.is_stale,
        'created_at': item.created_at.isoformat(),
    }


class ContainerService:

    @staticmethod
    def build_mesa(
        *,
        workspace,

        briefing=None,

    ) -> Dict:
        return ContainerService._build(

            client=client,
            briefing=briefing,
            include_stale=include_stale,
        )

    def build_cliente(

        include_stale: bool = False,
    ) -> Dict:
        return ContainerService._build(
            container=InsightItem.CONTAINER_CLIENTE,
            workspace=workspace,
            client=client,
            briefing=None,
            include_stale=include_stale,
        )

    @staticmethod
    def build_relatorio(
        *,
        workspace,
        briefing,
        include_stale: bool = False,
    ) -> Dict:
        return ContainerService._build(
            container=InsightItem.CONTAINER_RELATORIO,
            workspace=workspace,
            client=briefing.client,
            briefing=briefing,
            include_stale=include_stale,
        )

    @staticmethod
    def _build(
        *,
        container: str,
        workspace,
        client,
        briefing,
        include_stale: bool,
    ) -> Dict:
        slots = InsightItem.CONTAINER_KINDS[container]
        all_kinds = InsightItem.kinds_for_container(container)

        qs = (
            InsightItem.objects
            .filter(workspace=workspace, kind__in=all_kinds)
            .select_related('client', 'briefing', 'run')
            .order_by('-created_at')
        )
        if client is not None:
            qs = qs.filter(client=client)
        if briefing is not None:
            qs = qs.filter(briefing=briefing)
        if not include_stale:
            qs = qs.filter(is_stale=False)

        by_kind = {}
        for item in qs:
            by_kind.setdefault(item.kind, []).append(item)

        containers = {}
        counts = {}
        for slot_name, slot_kinds in slots.items():
            collected = []
            for kind in slot_kinds:
                collected.extend(by_kind.get(kind, []))
            collected.sort(key=lambda i: i.created_at, reverse=True)
            containers[slot_name] = [_serialize_item(i) for i in collected]
            counts[slot_name] = len(collected)

        return {'containers': containers, 'counts': counts}

"""
CostAllocationService — aloca custo real da execução aos insights
que ela produziu.

REGRA DE RATEIO (documentada e testada)
---------------------------------------
Para uma execução (AnalysisRun) que produziu N insights e teve
custo real C (lido de console.UsageRecord via usage_record_id):

    1. N == 0            → nenhuma alocação (não há a quem alocar).
    2. N == 1            → o único insight recebe C inteiro
                           (rule='single_insight').
    3. N > 1             → cada insight recebe C // N, e o ÚLTIMO
                           recebe C - (C // N) * (N - 1), absorvendo
                           o resto. A soma exata é C.
                           (rule='equal_split').

A absorção do resto no último insight é o que garante que a soma
das parcelas bate com o custo real. Sem isso, `C / 3` × 3 gera
diferença de 1 unidade na última casa decimal.

C é determinístico: vem do UsageRecord (custo real). Se C for
desconhecido (usage_record_id ausente ou UsageRecord sem preço),
a alocação é criada com `allocated_cost=None` e `is_estimated=True`
— sinalizando explicitamente "custo desconhecido", nunca zero.

IDEMPOTÊNCIA
------------
`allocate_for_run` pode ser chamado múltiplas vezes sem duplicar
linhas (constraint única em insight + usage_record_id). Re-execuções
com mesmo usage_record_id são no-op.
"""

import logging
from decimal import Decimal
from typing import Optional

from django.db import transaction

from apps.intelligence.models import (
    AnalysisRun, InsightItem, InsightCostAllocation,
)

logger = logging.getLogger('intelligence')


class CostAllocationService:

    @staticmethod
    def allocate_for_run(
        run: AnalysisRun,
        ai_response: dict,
    ) -> list:
        """
        Aloca o custo da execução aos insights produzidos.

        Deve ser chamado APÓS a persistência dos insights e APÓS o
        console ter criado o UsageRecord (via call_ai).

        Nunca levanta — falha de alocação não desfaz insights.
        """
        usage_record_id = ai_response.get('usage_record_id')
        if not usage_record_id:
            logger.info(
                'Run %s sem usage_record_id. Custo não alocável.', run.id,
            )
            return []

        total_cost, currency = CostAllocationService._fetch_usage_cost(
            usage_record_id,
        )

        insights = list(
            InsightItem.objects.filter(run=run).order_by('created_at')
        )
        if not insights:
            logger.info('Run %s não produziu insights. Nada a alocar.', run.id)
            return []

        allocations = []
        with transaction.atomic():
            if len(insights) == 1:
                allocations.append(
                    CostAllocationService._create(
                        insight=insights[0],
                        usage_record_id=usage_record_id,
                        cost=total_cost,
                        currency=currency,
                        rule=InsightCostAllocation.RULE_SINGLE_INSIGHT,
                        is_estimated=False if total_cost is not None else True,
                    )
                )
            else:
                n = len(insights)
                shares = []
                if total_cost is None:
                    shares = [None] * n
                else:
                    # Divisão com absorção do resto no último.
                    base = (total_cost / Decimal(n)).quantize(
                        Decimal('0.00000001'),
                    )
                    allocated_sum = base * (n - 1)
                    last = total_cost - allocated_sum
                    shares = [base] * (n - 1) + [last]

                for insight, share in zip(insights, shares):
                    allocations.append(
                        CostAllocationService._create(
                            insight=insight,
                            usage_record_id=usage_record_id,
                            cost=share,
                            currency=currency,
                            rule=InsightCostAllocation.RULE_EQUAL_SPLIT,
                            is_estimated=True,
                        )
                    )
        return [a for a in allocations if a is not None]

    # ------------------------------------------------------------------
    # HELPERS
    # ------------------------------------------------------------------

    @staticmethod
    def _fetch_usage_cost(usage_record_id) -> tuple:
        """
        Lê o custo real do UsageRecord.

        Retorna (Decimal | None, str). None significa desconhecido —
        NÃO significa zero.
        """
        try:
            from apps.console.models import UsageRecord
            record = UsageRecord.objects.filter(id=usage_record_id).only(
                'estimated_cost', 'currency',
            ).first()
        except Exception:
            logger.exception(
                'Falha ao consultar UsageRecord %s', usage_record_id,
            )
            return (None, '')

        if record is None:
            return (None, '')

        cost = record.estimated_cost
        currency = record.currency or ''
        if cost is None:
            return (None, currency)
        return (Decimal(cost), currency)

    @staticmethod
    def _create(
        *,
        insight,
        usage_record_id,
        cost: Optional[Decimal],
        currency: str,
        rule: str,
        is_estimated: bool,
    ):
        """
        Cria a alocação ou devolve a existente (idempotência).
        """
        try:
            obj, created = InsightCostAllocation.objects.get_or_create(
                insight=insight,
                usage_record_id=usage_record_id,
                defaults={
                    'allocated_cost': cost,
                    'currency': currency,
                    'allocation_rule': rule,
                    'is_estimated': is_estimated,
                },
            )
            return obj
        except Exception:
            logger.exception(
                'Falha ao criar alocação para insight=%s usage=%s',
                insight.id, usage_record_id,
            )
            return None

"""
IngestService — porta de entrada para novos eventos de inteligência.

FRONTEIRA
---------
Este serviço recebe apenas IDs e metadados — nunca instâncias de
models de outros módulos. A resolução dos objetos é feita aqui,
via imports diretos dos models (mesmo padrão já usado no arquivo
original).

IDEMPOTÊNCIA
------------
A `dedupe_key` inclui `occurrence_id`, que distingue eventos atômicos
legítimos (ex: duas edições na mesma nota) de reentrância do mesmo
evento. Quem chama é responsável por fornecer um `occurrence_id`
estável para o fato (ex: UUID do evento de origem, updated_at).

RETORNO
-------
Passou a devolver `(event, job, created)` em vez de `(event, created)`.
O `job` é o `AnalysisJob` agendado (ou reutilizado) para o evento.
Isso permite que a view de `reanalyze` devolva o `job_id` ao
frontend sem precisar buscar o "job mais recente" por heurística.
"""

import logging

from django.db import transaction

from apps.intelligence.models import InsightEvent
from apps.intelligence.services.job_service import JobService

logger = logging.getLogger('intelligence')


class IngestService:

    @staticmethod
    @transaction.atomic
    def ingest_change_event(
        *,
        workspace_id,
        client_id,
        source_kind: str,
        occurrence_id: str,
        data_version: str,
        briefing_id=None,
        source_ref: str = '',
    ):
        """
        Registra um evento de mudança e agenda uma análise.

        Retorna (event, job, created).

        `job` é o AnalysisJob agendado (novo ou já existente para o
        mesmo job_key). Nunca é None em caso de sucesso.
        """
        from apps.workspaces.models import Workspace
        from apps.clients.models import Client
        from apps.briefings.models import Briefing

        try:
            workspace = Workspace.objects.get(id=workspace_id)
            client = Client.objects.get(id=client_id)
            briefing = Briefing.objects.get(id=briefing_id) if briefing_id else None
        except Exception:
            logger.exception('Falha ao resolver objetos do evento de ingestão')
            return (None, None, False)

        dedupe_key = ':'.join([
            str(workspace.id),
            str(client.id),
            str(briefing.id) if briefing else '',
            source_kind,
            source_ref or '',
            occurrence_id,
        ])

        event, created = InsightEvent.objects.get_or_create(
            dedupe_key=dedupe_key,
            defaults={
                'workspace': workspace,
                'client': client,
                'briefing': briefing,
                'source_kind': source_kind,
                'source_ref': source_ref,
                'occurrence_id': occurrence_id,
            },
        )

        if not created:
            logger.info(
                'Evento duplicado ignorado (dedupe_key=%s)', dedupe_key,
            )
            # Mesmo quando o evento é duplicado, o job correspondente
            # já existe. Recupera-o para devolver ao chamador.
            job = JobService.find_job_for_event(
                workspace_id=workspace.id,
                client_id=client.id,
                briefing_id=briefing.id if briefing else None,
                data_version=data_version,
            )
            return (event, job, False)

        logger.info('Novo evento registrado: %s', source_kind)

        job = JobService.schedule_analysis_job(
            workspace_id=workspace.id,
            client_id=client.id,
            briefing_id=briefing.id if briefing else None,
            event=event,
            data_version=data_version,
        )

        return (event, job, True)

"""
JobService — agenda e reivindica jobs de análise de forma atômica.

CONCORRÊNCIA
------------
`claim_next_job()` abre uma transação, seleciona UM job pendente com
`select_for_update(skip_locked=True)` e o marca como `processing` ANTES
de liberar o lock. Dois workers concorrentes nunca reivindicam o mesmo
job.
"""

import hashlib
import logging
from typing import Optional

from django.db import transaction
from django.utils import timezone

from apps.intelligence.models import AnalysisJob, InsightEvent, InsightItem

logger = logging.getLogger('intelligence')


class JobService:

    @staticmethod
    @transaction.atomic
    def schedule_analysis_job(
        *,
        workspace_id,
        client_id,
        event: InsightEvent,
        data_version: str,
        briefing_id=None,
    ) -> AnalysisJob:
        from apps.workspaces.models import Workspace
        from apps.clients.models import Client
        from apps.briefings.models import Briefing

        workspace = Workspace.objects.get(id=workspace_id)
        client = Client.objects.get(id=client_id)
        briefing = Briefing.objects.get(id=briefing_id) if briefing_id else None

        if briefing:
            target_kinds = [
                InsightItem.KIND_SUMMARY_BRIEFING,
                InsightItem.KIND_DESIRE,
                InsightItem.KIND_ATTENTION,
                InsightItem.KIND_TAG,
                InsightItem.KIND_RECOMMENDED_QUESTION,
                InsightItem.KIND_GAP,
                InsightItem.KIND_INCONSISTENCY,
                InsightItem.KIND_CROSS_INSIGHT,
                InsightItem.KIND_SUCCESS_CRITERIA,
                InsightItem.KIND_REFERENCE_ANALYSIS,
                InsightItem.KIND_RESTRICTION,
                InsightItem.KIND_READINESS_ASSESSMENT,
            ]
        else:
            # Ingestão sem briefing: analisa o cliente como um todo.
            # `KIND_SUMMARY_CLIENT` não existe em InsightItem. O
            # equivalente semântico é `KIND_CLIENT_PROFILE`, que já
            # está na lista abaixo. Não duplicar.
            target_kinds = [
                InsightItem.KIND_KNOWLEDGE_ACCUMULATED,
                InsightItem.KIND_PREFERENCE_PROFILE,
                InsightItem.KIND_HOW_TO_CONDUCT,
                InsightItem.KIND_CLIENT_PROFILE,
            ]

        kinds_str = ','.join(sorted(target_kinds))
        kinds_hash = hashlib.sha256(kinds_str.encode('utf-8')).hexdigest()

        job_key = ':'.join([
            str(workspace.id),
            str(client.id),
            str(briefing.id) if briefing else '',
            kinds_hash,
            data_version,
        ])

        job, created = AnalysisJob.objects.get_or_create(
            job_key=job_key,
            defaults={
                'workspace': workspace,
                'client': client,
                'briefing': briefing,
                'target_kinds': target_kinds,
                'data_version': data_version,
            },
        )

        if created:
            logger.info("Novo job agendado (ID=%s)", job.id)
        else:
            logger.info("Job existente reutilizado (ID=%s)", job.id)
        return job

    @staticmethod
    def find_job_for_event(
        *,
        workspace_id,
        client_id,
        briefing_id,
        data_version: str,
    ) -> Optional[AnalysisJob]:
        """
        Recupera o job correspondente a um evento que já foi ingerido.

        Usado quando `ingest_change_event` encontra um dedupe_key
        existente — o evento é duplicado, mas o chamador ainda precisa
        do `job_id`. A busca é feita por (workspace, client, briefing,
        data_version) no job mais recente.
        """
        qs = AnalysisJob.objects.filter(
            workspace_id=workspace_id,
            client_id=client_id,
            briefing_id=briefing_id,
            data_version=data_version,
        ).order_by('-created_at')
        return qs.first()

    @staticmethod
    def claim_next_job() -> Optional[AnalysisJob]:
        """
        Reivindica atomicamente um job pendente.

        O `select_for_update(skip_locked=True)` dentro da transação
        garante que dois workers não peguem o mesmo job. A atualização
        para `processing` acontece ANTES da transação fechar — o lock
        cobre a reivindicação.
        """
        with transaction.atomic():
            job = (
                AnalysisJob.objects
                .select_for_update(skip_locked=True)
                .filter(status=AnalysisJob.STATUS_PENDING)
                .order_by('created_at')
                .first()
            )
            if job is None:
                return None

            job.status = AnalysisJob.STATUS_PROCESSING
            job.attempt_count += 1
            job.processing_started_at = timezone.now()
            job.save(update_fields=[
                'status', 'attempt_count', 'processing_started_at', 'updated_at',
            ])
            return job

    @staticmethod
    def mark_as_completed(job: AnalysisJob):
        job.status = AnalysisJob.STATUS_COMPLETED
        job.completed_at = timezone.now()
        job.save(update_fields=['status', 'completed_at', 'updated_at'])

    @staticmethod
    def mark_as_failed(job: AnalysisJob, error_message: str):
        job.status = AnalysisJob.STATUS_FAILED
        job.failed_at = timezone.now()
        job.last_error_message = error_message
        job.save(update_fields=['status', 'failed_at', 'last_error_message', 'updated_at'])

"""
StalenessService — marca insights anteriores como stale quando uma
execução nova os substitui.

REGRA
-----
Um run novo torna stale os insights de runs anteriores do MESMO
workspace e MESMO cliente cujos `kind` estão cobertos pelo run novo.
O escopo de comparação cobre:

    - briefing=X   (quando o run novo tem briefing)
    - briefing=None (para kinds de cliente que o run também produz)

Por quê os dois escopos: um run com briefing gera kinds de briefing
E kinds de cliente. Os kinds de cliente (preference.profile,
client.profile, how_to_conduct, knowledge.accumulated) têm
`briefing=None` no InsightItem, mesmo tendo sido produzidos por um
run com briefing. Comparar só pelo `briefing` do run novo deixaria
esses insights antigos "vivos" — o que mostraria duas versões
concorrentes nos containers do Cliente.

NÃO apaga nada. O insight antigo continua consultável, mas sinalizado
como desatualizado. A interface pode exibi-lo com aviso ou ocultá-lo
por padrão.
"""

import logging

from django.db import transaction
from django.db.models import Q

from apps.intelligence.models import AnalysisRun, InsightItem

logger = logging.getLogger('intelligence')


class StalenessService:

    @staticmethod
    def mark_previous_runs_stale(new_run: AnalysisRun) -> int:
        """
        Marca como stale os insights de runs anteriores compatíveis.

        Compatibilidade:
            - mesmo workspace
            - mesmo cliente
            - mesmo `briefing` do run (para kinds de briefing)
              OU `briefing=None` (para kinds de cliente que o run
              também produz)

        Retorna a quantidade de insights marcados.
        """
        new_kinds = list(
            InsightItem.objects
            .filter(run=new_run)
            .values_list('kind', flat=True)
            .distinct()
        )
        if not new_kinds:
            return 0

        # Escopo do briefing: o `briefing` do run novo, se houver.
        # Insights de cliente têm `briefing=None` mesmo quando
        # produzidos por um run com briefing.
        scope_q = Q(briefing=new_run.briefing)
        if new_run.briefing_id is not None:
            scope_q |= Q(briefing__isnull=True)

        qs = (
            InsightItem.objects
            .filter(
                scope_q,
                workspace=new_run.workspace,
                client=new_run.client,
                kind__in=new_kinds,
                is_stale=False,
            )
            .exclude(run=new_run)
        )
        count = qs.count()
        if count == 0:
            return 0

        with transaction.atomic():
            updated = qs.update(is_stale=True)

        logger.info(
            'StalenessService: %d insights marcados como stale '
            '(run=%s, kinds=%s, briefing=%s)',
            updated, new_run.id, new_kinds, new_run.briefing_id,
        )
        return updated

"""
TagService — transição de estado de tags com histórico imutável.

REGRA ARQUITETURAL
------------------
Toda mudança em `InsightItem.review_state` para itens de kind='tag'
DEVE passar por este service. Ele grava uma linha em
`InsightTagHistory` a cada transição, preservando quem, quando e
qual era o estado anterior.

Insights de outros kinds podem usar `review_state` diretamente —
mas a diretriz de histórico só se aplica a tags.
"""

import logging
from typing import Optional

from django.db import transaction

from apps.intelligence.models import InsightItem, InsightTagHistory
from apps.intelligence.exceptions import IntelligenceError

logger = logging.getLogger('intelligence')


class InvalidTagTransitionError(IntelligenceError):
    """Transição de estado de tag inválida."""
    pass


class TagService:

    # Transições permitidas. A tag nasce em 'new'; pode ser confirmada
    # ou descartada; uma decisão pode ser revertida (revisão humana),
    # mas só com histórico explícito.
    ALLOWED = {
        (InsightTagHistory.STATE_NEW, InsightTagHistory.STATE_CONFIRMED),
        (InsightTagHistory.STATE_NEW, InsightTagHistory.STATE_DISCARDED),
        (InsightTagHistory.STATE_CONFIRMED, InsightTagHistory.STATE_DISCARDED),
        (InsightTagHistory.STATE_DISCARDED, InsightTagHistory.STATE_CONFIRMED),
    }

    @staticmethod
    @transaction.atomic
    def transition(
        *,
        insight: InsightItem,
        new_state: str,
        changed_by=None,
        reason: str = '',
    ) -> InsightItem:
        """
        Aplica uma transição de estado com histórico.

        Rejeita:
            - insights que não sejam kind='tag';
            - new_state fora de STATE_CHOICES;
            - transição não permitida (ex: confirmed → new);
            - no-op (mesmo estado).
        """
        if insight.kind != InsightItem.KIND_TAG:
            raise InvalidTagTransitionError(
                'Apenas itens de kind="tag" possuem histórico de estado.'
            )

        valid_states = {c[0] for c in InsightTagHistory.STATE_CHOICES}
        if new_state not in valid_states:
            raise InvalidTagTransitionError(
                f'Estado inválido: {new_state}.'
            )

        previous = insight.review_state
        if previous == new_state:
            raise InvalidTagTransitionError(
                f'A tag já está no estado "{new_state}".'
            )

        if (previous, new_state) not in TagService.ALLOWED:
            raise InvalidTagTransitionError(
                f'Transição não permitida: {previous} → {new_state}.'
            )

        InsightTagHistory.objects.create(
            insight=insight,
            previous_state=previous,
            new_state=new_state,
            changed_by=(
                changed_by
                if (changed_by and getattr(changed_by, 'is_authenticated', False))
                else None
            ),
            reason=reason or '',
        )

        insight.review_state = new_state
        insight.save(update_fields=['review_state', 'updated_at'])

        logger.info(
            'Tag transicionada: insight=%s %s→%s by=%s',
            insight.id, previous, new_state,
            getattr(changed_by, 'email', None) or 'system',
        )
        return insight

    # ------------------------------------------------------------------
    # CONSULTAS
    # ------------------------------------------------------------------

    @staticmethod
    def history_for(insight: InsightItem):
        """Histórico completo da tag, mais recente primeiro."""
        if insight.kind != InsightItem.KIND_TAG:
            return InsightTagHistory.objects.none()
        return (
            InsightTagHistory.objects
            .filter(insight=insight)
            .select_related('changed_by')
            .order_by('-created_at')
        )

    @staticmethod
    def list_tags(
        *,
        workspace,
        client=None,
        state: Optional[str] = None,
    ):
        """
        Lista tags do workspace (opcionalmente filtradas por cliente
        e/ou estado atual).
        """
        qs = (
            InsightItem.objects
            .filter(workspace=workspace, kind=InsightItem.KIND_TAG)
            .select_related('client', 'briefing')
            .order_by('-created_at')
        )
        if client is not None:
            qs = qs.filter(client=client)
        if state is not None:
            qs = qs.filter(review_state=state)
        return qs

def test_claim_next_job_is_exclusive(analysis_job):

    """

@pytest.mark.skipif(
    not _running_on_postgres(),
    reason=(
        'Teste real de concorrência requer Postgres. '
        'SQLite em memória não compartilha o banco entre threads. '
        'Rode no CI com Postgres habilitado.'
    ),
)
def test_claim_skips_locked_rows(analysis_job, django_db_blocker):
    """
    Simula um worker que segura o lock. Um segundo worker rodando
    em outra conexão (thread) deve usar `skip_locked` e retornar
    None em vez de bloquear.
    """
    import threading
    import time
    results = []

    def hold_lock():
        from django.db import transaction
        with django_db_blocker.unblock():
            with transaction.atomic():
                AnalysisJob.objects.select_for_update().filter(
                    id=analysis_job.id
                ).first()
                time.sleep(1.5)

    def try_claim():
        from django.db import connection
        with django_db_blocker.unblock():
            try:
                j = JobService.claim_next_job()
                results.append(j)
            finally:
                connection.close()

    t1 = threading.Thread(target=hold_lock)
    t1.start()
    time.sleep(0.3)
    t2 = threading.Thread(target=try_claim)
    t2.start()
    t1.join()
    t2.join()

    # O segundo worker deve ter retornado None (skip_locked)
    assert len(results) == 1
    assert results[0] is None
