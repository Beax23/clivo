"""
Integrations — ferramentas externas conectadas pelo escritório.

============================================================================
FRONTEIRA ARQUITETURAL — console × integrations
============================================================================

    console      → fornecedores de infraestrutura da Clivo.
                   Quanto a Clivo gasta, com quem, em quê.
                   Dono de: Provider, ProviderConnection,
                            ProviderContract, ProviderPricing,
                            UsageRecord, ProviderLog.

    integrations → ferramentas externas que o escritório conecta
                   para operar.
                   Dono de: IntegrationProvider, IntegrationResource,
                            WorkspaceIntegration, ProviderConnector,
                            ExternalFile, ExternalForm.

============================================================================
REGRAS DE FRONTEIRA (CONGELADAS)
============================================================================

    1. console NUNCA importa integrations.
    2. integrations NUNCA importa console.models.Provider
       (nem qualquer outro model de fornecedor do console).
    3. Utilitários técnicos genéricos (cifra, logging, correlation,
       envelope de erro) podem viver em core/ e ser consumidos por
       ambos, sem criar dependência entre os domínios.
    4. Modelos e regras de negócio NUNCA são compartilhados entre os
       dois eixos.

    Protegido por teste automatizado em
    apps/integrations/tests/test_architectural_boundaries.py.

O QUE O APP POSSUI
============================================================================

    IntegrationProvider     → catálogo de serviços externos
    IntegrationResource     → recursos que um provider oferece

    WorkspaceIntegrationService

    ProviderRegistry        → registro em memória dos connectors

    SAÍDA (preview):

    SAÍDA (import):

from rest_framework import serializers


# =========================================================================
# ENTRADA
# =========================================================================

class FormsSearchQuerySerializer(serializers.Serializer):
    """
    Query params de GET /api/integrations/google/forms/search/
    """
    q = serializers.CharField(
        required=False,
        allow_blank=True,
        max_length=200,
        default='',
    )


class FormPreviewSerializer(serializers.Serializer):
    """
    Body de POST /api/integrations/google/forms/preview/

    Aceita URL completa ou ID cru. A validação fina é feita pelo
    service — aqui garantimos que é uma string não vazia.
    """
    form = serializers.CharField(
        required=True,
        allow_blank=False,
        max_length=2000,
        help_text=(
            'URL do Google Forms ou o ID do formulário.'
        ),
    )


class FormImportSerializer(serializers.Serializer):
    """
    Body de POST /api/integrations/google/forms/import/

    `template_name` é OPCIONAL. Se ausente, o backend usa o nome
    do formulário do Google como nome do template criado.
    """
    form = serializers.CharField(
        required=True,
        allow_blank=False,
        max_length=2000,
        help_text=(
            'URL do Google Forms ou o ID do formulário.'
        ),
    )
    template_name = serializers.CharField(
        required=False,
        allow_blank=True,
        max_length=200,
        default='',
        help_text=(
            'Nome do novo modelo. Se vazio, usa o nome do formulário '
            'do Google.'
        ),
    )

    def validate_template_name(self, value):
        if value is None:
            return ''
        return value.strip()

# =========================================================================

class FormSummarySerializer(serializers.Serializer):
    """
    Item da lista de formulários retornado por /search/.
    """

    description = serializers.CharField(read_only=True, allow_blank=True)

    template_id = serializers.UUIDField(read_only=True)
    template_name = serializers.CharField(read_only=True)
    sections_created = serializers.IntegerField(read_only=True)
    questions_created = serializers.IntegerField(read_only=True)

"""
Views DRF para a API de Google Forms.

Endpoints:

    GET  /api/integrations/google/forms/search/
        Lista formulários da conta conectada.
        Se não houver integração ativa → 428 Precondition Required.

    POST /api/integrations/google/forms/preview/
        Busca o resumo de um formulário SEM persistir nada.
        Usado pelo frontend para montar a tela de confirmação
        antes de criar o template.

    POST /api/integrations/google/forms/import/
        Cria um NOVO BriefingTemplate com as sessões e perguntas
        do formulário. Devolve o `template_id` criado.
        Se não houver integração ativa → 428 Precondition Required.

SOBRE 428:
    Usamos HTTP 428 (Precondition Required) para sinalizar que
    falta uma condição prévia (integração conectada). O frontend
    trata 428 como "ofereça conectar o Google".

SOBRE O WORKSPACE:
    O workspace é resolvido a partir da sessão
    (`current_workspace_id`). NÃO aceitamos `workspace_id` no body
    — o tenant é sempre contextual.
"""

import logging

class _GoogleFormsBaseView(APIView):
    """
    Comportamento comum: resolve workspace da sessão, traduz
    exceções para respostas HTTP.
    """

    permission_classes = [IsAuthenticated]

    def _current_workspace(self):
        ws_id = self.request.session.get('current_workspace_id')
        if not ws_id:
            return None
        return WorkspaceQueries.get_for_user(ws_id, self.request.user)

    def _require_workspace(self):
        """
        Retorna (workspace, None) ou (None, Response de erro).
        """
        ws = self._current_workspace()
        if ws is None:
            return None, Response(
                {'error': 'Nenhum workspace ativo.'},
                status=status.HTTP_400_BAD_REQUEST,
            )
        return ws, None

    def _handle_domain_error(self, exc) -> Response:
        """
        Traduz exceções do serviço em respostas HTTP.

        Retorna sempre uma Response — nunca None.
        """
        if isinstance(exc, ConnectionNotEstablishedError):
            return Response(
                {
                    'error': (
                        'Nenhuma conta Google conectada a este '
                        'workspace. Conecte uma para continuar.'
                    ),
                    'code': 'google_not_connected',
                },
                status=status.HTTP_428_PRECONDITION_REQUIRED,
            )

        if isinstance(exc, FormsScopeMissingError):
            return Response(
                {
                    'error': (
                        'A conta Google conectada não autorizou '
                        'acesso ao Google Forms. Reconecte autorizando '
                        'esta permissão.'
                    ),
                    'code': 'google_forms_scope_missing',
                },
                status=status.HTTP_428_PRECONDITION_REQUIRED,
            )

        if isinstance(exc, ConnectionExpiredError):
            return Response(
                {
                    'error': (
                        'A credencial do Google expirou. Reconecte '
                        'a conta para continuar.'
                    ),
                    'code': 'google_expired',
                },
                status=status.HTTP_428_PRECONDITION_REQUIRED,
            )

        if isinstance(exc, ConnectionRevokedError):
            return Response(
                {
                    'error': (
                        'A conta Google foi revogada. Reconecte '
                        'para continuar.'
                    ),
                    'code': 'google_revoked',
                },
                status=status.HTTP_428_PRECONDITION_REQUIRED,
            )

        if isinstance(exc, AmbiguousConnectionError):
            return Response(
                {
                    'error': (
                        'Há múltiplas contas Google conectadas sem '
                        'uma preferida. Escolha uma para continuar.'
                    ),
                    'code': 'google_ambiguous',
                },
                status=status.HTTP_409_CONFLICT,
            )

        if isinstance(exc, IntegrationInactiveError):
            return Response(
                {
                    'error': 'A integração Google não está ativa.',
                    'code': 'google_inactive',
                },
                status=status.HTTP_428_PRECONDITION_REQUIRED,
            )

        if isinstance(exc, RateLimitError):
            return Response(
                {
                    'error': (
                        'A API do Google está limitando requisições. '
                        'Tente novamente em alguns instantes.'
                    ),
                    'code': 'google_rate_limited',
                },
                status=status.HTTP_429_TOO_MANY_REQUESTS,
            )

        if isinstance(exc, ProviderTimeoutError):
            return Response(
                {
                    'error': (
                        'A API do Google não respondeu a tempo. '
                        'Tente novamente.'
                    ),
                    'code': 'google_timeout',
                },
                status=status.HTTP_504_GATEWAY_TIMEOUT,
            )

        if isinstance(exc, ProviderServerError):
            return Response(
                {
                    'error': (
                        'Erro temporário na API do Google. '
                        'Tente novamente em alguns instantes.'
                    ),
                    'code': 'google_server_error',
                },
                status=status.HTTP_502_BAD_GATEWAY,
            )

        if isinstance(exc, DjangoValidationError):
            messages = getattr(exc, 'messages', None) or [str(exc)]
            return Response(
                {'error': messages[0], 'code': 'validation_error'},
                status=status.HTTP_400_BAD_REQUEST,
            )

        if isinstance(exc, NotImplementedError):
            return Response(
                {
                    'error': (
                        'A leitura de formulários ainda não está '
                        'disponível. Estamos finalizando esta '
                        'integração.'
                    ),
                    'code': 'not_implemented_yet',
                },
                status=status.HTTP_501_NOT_IMPLEMENTED,
            )

        logger.exception('Erro inesperado em GoogleFormsView')
        return Response(
            {'error': 'Erro interno. Tente novamente.', 'code': 'internal'},
            status=status.HTTP_500_INTERNAL_SERVER_ERROR,
        )

class GoogleFormsSearchView(_GoogleFormsBaseView):
    """
    GET /api/integrations/google/forms/search/?q=<query>

    Lista formulários da conta Google conectada, filtrados por
    nome se `q` for fornecido.

    Resposta 200:
        {
            "results": [
                {"id", "name", "description", "url"},
                ...
            ],
            "count": N
        }

    Resposta 428:
        Sem integração conectada / escopo faltando / credencial
        expirada.
    """

    def get(self, request):
        workspace, err = self._require_workspace()
        if err is not None:
            return err

        query_serializer = FormsSearchQuerySerializer(
            data=request.query_params,
        )
        query_serializer.is_valid(raise_exception=True)
        q = query_serializer.validated_data.get('q') or ''

        try:
            results = GoogleFormsService.search_forms(
                workspace=workspace,
                query=q,
            )
        except Exception as exc:
            return self._handle_domain_error(exc)

        return Response({
            'results': FormSummarySerializer(results, many=True).data,
            'count': len(results),
        })

class GoogleFormsPreviewView(_GoogleFormsBaseView):
    """
    POST /api/integrations/google/forms/preview/

    Body:
        { "form": "<URL ou ID do formulário>" }

    Resposta 200:
        {
            "form_id": "...",
            "form_name": "...",          # nome sugerido
            "form_description": "...",
            "sections_count": N,
            "questions_count": M
        }

    Este endpoint NÃO cria nada. Serve para o frontend montar a
    tela de confirmação antes de criar o template novo.
    """

    def post(self, request):
        workspace, err = self._require_workspace()
        if err is not None:
            return err

        serializer = FormPreviewSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        form_value = serializer.validated_data['form']

        try:
            result = GoogleFormsService.preview_form(
                workspace=workspace,
                form_url_or_id=form_value,
            )
        except Exception as exc:
            return self._handle_domain_error(exc)

        return Response(FormPreviewResultSerializer(result).data)

class GoogleFormsImportView(_GoogleFormsBaseView):
    """
    POST /api/integrations/google/forms/import/

    Body:
        {
            "form": "<URL ou ID do formulário>",
            "template_name": "<nome escolhido pelo usuário>"  # opcional
        }

    Resposta 200:
        {
            "imported": true,
            "form_id": "...",
            "template_id": "<uuid do NOVO template criado>",
            "template_name": "...",
            "sections_created": N,
            "questions_created": M
        }

    Resposta 400:
        URL/ID inválido.

    Resposta 428:
        Sem integração conectada / escopo faltando / credencial
        expirada.

    Resposta 501:
        Chamada real ao Google ainda não implementada (V1).
    """

    def post(self, request):
        workspace, err = self._require_workspace()
        if err is not None:
            return err

        serializer = FormImportSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        form_value = serializer.validated_data['form']
        template_name = serializer.validated_data.get('template_name', '')

        try:
            result = GoogleFormsService.import_form(
                workspace=workspace,
                form_url_or_id=form_value,
                template_name=template_name,
                actor=request.user,
            )
        except Exception as exc:
            return self._handle_domain_error(exc)

        return Response(FormImportResultSerializer(result).data)

"""
EncryptionService — cifra de credenciais em repouso.

ALGORITMO: Fernet (cryptography). Simétrico, autenticado, com envelope.
CHAVE: settings.INTEGRATIONS_ENCRYPTION_KEY (str base64-url de 32 bytes).

FORMATO DO PAYLOAD CIFRADO:
    O EncryptionService cifra um blob opaco. O formato do conteúdo
    é responsabilidade do chamador (WorkspaceIntegrationService).
    O blob inclui prefixo de versão de chave para permitir rotação.

SEGURANÇA:
    - Nunca loga o plaintext.
    - Nunca inclui plaintext em exceção.
    - Rotation: se a chave rotaciona, payloads antigos continuam
      decifráveis via chaves anteriores (INTEGRATIONS_ENCRYPTION_KEY_OLD_*).
"""

import logging

from cryptography.fernet import Fernet, InvalidToken
from django.conf import settings

from apps.integrations.exceptions import (
    EncryptionDecryptError,
    EncryptionKeyMissingError,
)

logger = logging.getLogger('integrations')


# Chave primária das settings. Formato esperado: string base64-url de 32 bytes.
SETTINGS_KEY_PRIMARY = 'INTEGRATIONS_ENCRYPTION_KEY'


class EncryptionService:
    """
    Serviço de cifra simétrica com envelope e rotação de chaves.
    """

    # =====================================================================
    # CHAVES
    # =====================================================================

    @staticmethod
    def _load_keys() -> dict[str, Fernet]:
        """
        Carrega todas as chaves configuradas.

        Chave primária: INTEGRATIONS_ENCRYPTION_KEY (usada para cifrar).
        Chaves antigas: INTEGRATIONS_ENCRYPTION_KEY_OLD_1, _2, ... (usadas
        para decifrar dados legados durante rotação).

        Retorna dict {key_id: Fernet}. `key_id` é a string do atributo
        settings, sem o prefixo.
        """
        keys: dict[str, Fernet] = {}

        primary_raw = getattr(settings, SETTINGS_KEY_PRIMARY, None)
        if not primary_raw:
            raise EncryptionKeyMissingError(
                f'{SETTINGS_KEY_PRIMARY} não definida nas settings.'
            )

        keys['primary'] = EncryptionService._build_fernet(primary_raw)

        # Chaves antigas (rotação)
        i = 1
        while True:
            name = f'{SETTINGS_KEY_PRIMARY}_OLD_{i}'
            raw = getattr(settings, name, None)
            if not raw:
                break
            keys[f'old_{i}'] = EncryptionService._build_fernet(raw)
            i += 1

        return keys

    @staticmethod
    def _build_fernet(key: str) -> Fernet:
        """
        Constrói uma instância Fernet a partir da chave bruta.
        """
        try:
            return Fernet(key.encode('utf-8'))
        except Exception as exc:
            # A chave em si nunca aparece na mensagem.
            raise EncryptionKeyMissingError(
                'Formato de chave de criptografia inválido.'
            ) from exc

    # =====================================================================
    # CIFRA
    # =====================================================================

    @staticmethod
    def encrypt(plaintext: bytes) -> bytes:
        """
        Cifra um blob. Retorna o token Fernet (bytes).

        O token já é autenticado (Fernet) e contém timestamp.
        """
        keys = EncryptionService._load_keys()
        fernet = keys['primary']
        return fernet.encrypt(plaintext)

    @staticmethod
    def decrypt(ciphertext: bytes) -> bytes:
        """
        Decifra um blob. Tenta a chave primária, depois as antigas.

        Levanta EncryptionDecryptError se nenhuma chave funcionar.
        """
        keys = EncryptionService._load_keys()
        token = bytes(ciphertext)

        # Tenta primária
        try:
            return keys['primary'].decrypt(token)
        except InvalidToken:
            pass
        except Exception as exc:
            logger.warning(
                'Falha inesperada ao decifrar com chave primária: %s',
                type(exc).__name__,
            )

        # Tenta antigas
        for key_id, fernet in keys.items():
            if key_id == 'primary':
                continue
            try:
                return fernet.decrypt(token)
            except InvalidToken:
                continue
            except Exception as exc:
                logger.warning(
                    'Falha inesperada ao decifrar com %s: %s',
                    key_id, type(exc).__name__,
                )

        raise EncryptionDecryptError(
            'Não foi possível decifrar o payload com nenhuma chave configurada.'
        )

"""
Hierarquia de exceções do módulo integrations.

    IntegrationError (base)
    ├── IntegrationConfigurationError
    ├── ConnectorNotFoundError
    ├── ConnectionNotFoundError
    ├── ConnectionNotEstablishedError
    ├── AmbiguousConnectionError
    ├── ConnectionExpiredError
    ├── ConnectionRevokedError
    ├── EncryptionError
    │   ├── EncryptionDecryptError
    │   └── EncryptionKeyMissingError
    ├── ProviderError
    │   ├── UnsupportedResourceError
    │   ├── OAuthExchangeError
    │   ├── InsufficientScopesError
    │   ├── RefreshTokenInvalidError
    │   ├── RefreshTransientError
    │   ├── RateLimitError
    │   ├── ProviderTimeoutError
    │   ├── ProviderServerError
    │   └── ProviderClientError
    ├── NormalizationError
    │   └── UnexpectedProviderPayloadError
    └── WorkspaceIntegrationError
        ├── IntegrationInactiveError
        ├── IntegrationAlreadyExistsError
        └── CannotRevokePreferredError

REGRA DE SEGURANÇA:
    `message` e `context` NUNCA contêm tokens, refresh_tokens,
    authorization codes ou client_secrets. Se um contexto precisar
    de tais dados, eles são hash/mascarados ANTES de serem anexados.

    A regra é verificada por testes de regressão em
    apps/integrations/tests/test_exception_sanitization.py.

NOMENCLATURA:
    "Connector" (não "Adapter") é o nome do componente que encapsula
    a comunicação com um provider externo. Isso evita colisão visual
    com `CustomSocialAccountAdapter` de apps/accounts/ (adapter do
    django-allauth para login de usuário via Google).
"""


class IntegrationError(Exception):
    """
    Base de todas as exceções de integrations.

    Não deve ser levantada diretamente. Use uma subclasse.

    `context` é um dict com dados diagnósticos seguros. Nunca deve
    conter tokens, refresh_tokens, authorization codes ou secrets.
    """

    def __init__(self, message: str, *, context: dict | None = None):
        super().__init__(message)
        self.context = context or {}


# =========================================================================

# =========================================================================

class IntegrationConfigurationError(IntegrationError):
    """
    Configuração obrigatória do módulo integrations está ausente
    ou inválida.

    Causa: variável de ambiente faltando, Client OAuth não configurado,
    settings incompletos.

    Diferente de UnsupportedResourceError (que é sobre um recurso
    pedido pelo chamador), esta é sobre o estado do próprio módulo.

    Contexto esperado:
        {"provider_key": str | None, "missing": str}
    """


class ConnectorNotFoundError(IntegrationError):
    """
    Provider sem connector registrado.

    Causa: bug de configuração (provider no catálogo sem connector
    correspondente registrado no ProviderRegistry).

    Contexto esperado: {"provider_key": str}
    """


class ConnectionNotFoundError(IntegrationError):
    """
    Integração referenciada não existe no workspace.

    Causa: chamador passou `integration_id` de outro workspace,
    ou um id que não existe.

    Contexto esperado: {"workspace_id": str, "integration_id": str}
    """


class ConnectionNotEstablishedError(IntegrationError):
    """
    Nenhuma integração ativa do provider no workspace.

    Causa: usuário ainda não conectou nenhuma conta, ou todas foram
    revogadas/expiradas.

    Contexto esperado: {"workspace_id": str, "provider_key": str}
    """


class AmbiguousConnectionError(IntegrationError):
    """
    Múltiplas contas ativas sem preferida. Exige escolha explícita.

    Causa: usuário conectou mais de uma conta do mesmo provider, e
    nenhuma foi marcada como preferida. O sistema NÃO escolhe por
    conta própria — o chamador deve passar `integration_id`.

    Contexto esperado:
        {"workspace_id": str, "provider_key": str, "count": int}
    """


class ConnectionExpiredError(IntegrationError):
    """
    Conexão expirada. Precisa reconectar.

    Causa: refresh_token inválido, ou revogação remota detectada.
    O estado persiste como 'expired' até o usuário reconectar.

    Contexto esperado: {"integration_id": str}
    """


class ConnectionRevokedError(IntegrationError):
    """
    Conexão revogada (local ou remotamente).

    Causa: revogação explícita pelo usuário ou pela Clivo, ou
    detecção de revogação remota no provider.

    Contexto esperado: {"integration_id": str, "revoked_remotely": bool}
    """

class ProviderError(IntegrationError):
    """
    Erro originado na comunicação com um provider externo.

    Base para todas as exceções que representam algo deu errado
    do lado do provider (rede, HTTP, autenticação remota).
    """


class UnsupportedResourceError(ProviderError):
    """
    Recurso pedido não é suportado pelo connector.

    Causa: chamador passou um slug de recurso que o connector
    não sabe traduzir para escopo OAuth.

    Contexto esperado:
        {"provider_key": str, "resource_slug": str}
    """


class OAuthExchangeError(ProviderError):
    """
    Troca de code por tokens falhou.

    Causa: code inválido, expirado, ou redirect_uri incompatível.
    A mensagem NUNCA inclui o code nem os tokens.

    Contexto esperado: {"provider_key": str}
    """


class InsufficientScopesError(ProviderError):
    """
    Faltam recursos obrigatórios para a operação.

    Causa: o usuário concedeu menos recursos do que a operação
    exige. Verificado contra os recursos obrigatórios, não contra
    todos os solicitados.

    Contexto esperado:
        {"operation": str, "missing_resources": list[str]}
    """


class RefreshTokenInvalidError(ProviderError):
    """
    Refresh token inválido ou revogado remotamente.

    Causa: usuário revogou acesso no provider, ou o refresh_token
    expirou por política do provider.

    Contexto esperado: {"provider_key": str}
    """


class RefreshTransientError(ProviderError):
    """
    Falha transitória no refresh (rede/5xx).

    Causa: erro de rede, timeout, ou resposta 5xx durante o refresh.

    Contexto esperado:
        {"provider_key": str, "retry_after_s": int | None}
    """


class RateLimitError(ProviderError):
    """
    Provider sinalizou rate limit (HTTP 429).

    Causa: muitas chamadas em curto intervalo.

    Contexto esperado: {"retry_after_s": int | None}
    """


class ProviderTimeoutError(ProviderError):
    """
    Timeout na comunicação com o provider.

    Causa: operação excedeu o timeout configurado.

    Contexto esperado: {"timeout_s": int}
    """


class ProviderServerError(ProviderError):
    """
    Erro 5xx do provider.

    Causa: falha no lado do provider.

    Contexto esperado: {"status_code": int}
    """


class ProviderClientError(ProviderError):
    """
    Erro 4xx genérico do provider.

    Causa: 4xx que não se encaixa em causas específicas.
    Preferir exceções específicas (NotFound, Forbidden, etc.) quando
    possível. Esta é a "fallback" para casos não mapeados.

    Contexto esperado: {"status_code": int}
    """

class NormalizationError(IntegrationError):

    Falha ao converter payload do provider para contrato interno.

    Base para erros de normalização.

class UnexpectedProviderPayloadError(NormalizationError):
    """
    Payload do provider não corresponde ao formato esperado.

    Causa: mudança no provider OU bug no normalizador. Indica

    Contexto esperado: {"provider_key": str, "field": str}
    """

class IntegrationAlreadyExistsError(WorkspaceIntegrationError):

"""
Refresh proativo dos access_tokens do Google.

Uso:
    python manage.py refresh_google_tokens
    python manage.py refresh_google_tokens --dry-run
    python manage.py refresh_google_tokens --limit 50

Objetivo:
    Renovar access_tokens ANTES que expirem, para que o usuário
    não experimente latência em operações que dependem da API do
    Google.

    O refresh lazy (`IntegrationService.get_valid_access_token`)
    já cobre o caso de uso. Este comando é OTIMIZAÇÃO — reduz o
    tempo de resposta percebido pelo usuário.

Agendamento recomendado (cron):
    */30 * * * * python manage.py refresh_google_tokens --limit 100

IDEMPOTÊNCIA:
    Se chamado repetidamente, cada chamada só renova integrações
    que precisam. Não é destrutivo. Não causa efeito colateral no
    provider além do próprio refresh (que é o desejado).

COMPORTAMENTO EM ERRO:
    Se uma integração falhar com RefreshTokenInvalidError, ela
    vira `expired`. Isso é o comportamento correto — o cron
    "descobre" que a integração está quebrada antes que o
    usuário tente usá-la.
"""

import logging

from django.core.management.base import BaseCommand
from django.utils import timezone

from apps.integrations.exceptions import (
    ConnectionExpiredError,
    IntegrationError,
    RefreshTransientError,
)
from apps.integrations.models import WorkspaceIntegration
from apps.integrations.services.integration_service import (
    IntegrationService,
)

logger = logging.getLogger('integrations')


class Command(BaseCommand):
    help = (
        'Renova access_tokens do Google que estão próximos de expirar.'
    )

    def add_arguments(self, parser):
        parser.add_argument(
            '--limit',
            type=int,
            default=100,
            help='Máximo de integrações a processar. Default: 100.',
        )
        parser.add_argument(
            '--dry-run',
            action='store_true',
            help='Lista as integrações que precisariam de refresh, '
                 'sem renovar.',
        )

    def handle(self, *args, **options):
        limit = max(1, options['limit'])
        dry_run = options['dry_run']

        # Só processa integrações Google ativas.
        qs = (
            WorkspaceIntegration.objects
            .filter(
                provider__key='google',
                status=WorkspaceIntegration.STATUS_ACTIVE,
            )
            .select_related('provider', 'workspace')
            .order_by('last_refresh_at')
            [:limit]
        )

        refreshed = 0
        skipped = 0
        failed = 0
        expired = 0

        for integration in qs:
            # Checa se precisa de refresh (mesma lógica do lazy).
            if not IntegrationService._integration_needs_refresh(
                integration,
            ):
                skipped += 1
                continue

            if dry_run:
                self.stdout.write(
                    f'  [DRY] {integration.id} '
                    f'({integration.workspace.name}) '
                    f'precisa de refresh'
                )
                refreshed += 1
                continue

            try:
                IntegrationService.refresh_credentials(integration)
                refreshed += 1
            except ConnectionExpiredError:
                expired += 1
                self.stdout.write(self.style.WARNING(
                    f'  ⚠️  {integration.id} — credencial revogada, '
                    f'marcada como expired'
                ))
            except RefreshTransientError:
                failed += 1
                self.stdout.write(self.style.WARNING(
                    f'  ⚠️  {integration.id} — falha transitória'
                ))
            except IntegrationError as exc:
                failed += 1
                self.stdout.write(self.style.ERROR(
                    f'  ❌ {integration.id} — {exc}'
                ))

        self.stdout.write('')
        if dry_run:
            self.stdout.write(self.style.SUCCESS(
                f'✅ Dry-run: {refreshed} precisariam de refresh, '
                f'{skipped} já estão válidos.'
            ))
        else:
            self.stdout.write(self.style.SUCCESS(
                f'✅ Refresh concluído: '
                f'{refreshed} renovadas, '
                f'{skipped} já válidas, '
                f'{expired} expiradas, '
                f'{failed} falharam.'
            ))

"""
Seed idempotente do catálogo de providers e resources de integrations.

Popula:
    - IntegrationProvider(key='google') com recursos de Drive e Forms.
    - IntegrationResource para cada recurso suportado pelo GoogleConnector.

IDEMPOTENTE:
    - Usa `update_or_create` por `key` / `(provider, slug)`.
    - NÃO sobrescreve `is_active` se o operador alterou.
    - NÃO apaga providers que não estão no seed.

Uso:
    python manage.py seed_integration_providers
"""

from django.core.management.base import BaseCommand
from django.db import transaction

from apps.integrations.models import (
    IntegrationProvider,
    IntegrationResource,
)


PROVIDERS = [
    {
        'key': 'google',
        'name': 'Google',
        'category': 'productivity',
        'description': (
            'Ferramentas Google: Drive e Forms. '
            'Permite consultar arquivos e formulários do escritório.'
        ),
        'resources': [
            {
                'slug': 'drive.file.read',
                'name': 'Ler arquivos do Drive',
                'description': (
                    'Ler arquivos e metadados do Google Drive.'
                ),
            },
            {
                'slug': 'forms.body.read',
                'name': 'Ler formulários do Forms',
                'description': (
                    'Ler formulários e respostas do Google Forms.'
                ),
            },
        ],
    },
]


class Command(BaseCommand):
    help = (
        'Seed idempotente de IntegrationProvider + IntegrationResource.'
    )

    @transaction.atomic
    def handle(self, *args, **options):
        self.stdout.write('🌱 Seed de IntegrationProvider...')
        self.stdout.write('')

        for spec in PROVIDERS:
            provider, created = (
                IntegrationProvider.objects.update_or_create(
                    key=spec['key'],
                    defaults={
                        'name': spec['name'],
                        'category': spec['category'],
                        'description': spec['description'],
                        # `is_active` deliberadamente NÃO incluso:
                        # preserva decisão operacional.
                    },
                )
            )
            if created:
                self.stdout.write(self.style.SUCCESS(
                    f'   🆕 provider {provider.key} — {provider.name}'
                ))
            else:
                self.stdout.write(
                    f'   🔄 provider {provider.key} — {provider.name}'
                )

            for res_spec in spec['resources']:
                resource, res_created = (
                    IntegrationResource.objects.update_or_create(
                        provider=provider,
                        slug=res_spec['slug'],
                        defaults={
                            'name': res_spec['name'],
                            'description': res_spec['description'],
                        },
                    )
                )
                if res_created:
                    self.stdout.write(
                        f'      🆕 resource {resource.slug}'
                    )
                else:
                    self.stdout.write(
                        f'      🔄 resource {resource.slug}'
                    )

        self.stdout.write('')
        self.stdout.write(self.style.SUCCESS('✅ Seed concluído.'))

from django.db import models
from django.utils.translation import gettext_lazy as _

        - drive.file.read
        - drive.file.write
        - forms.body.read
        - forms.body.write
    """

            models.Index(fields=['provider', 'slug']),

                'slug': _(
                    'O slug deve ser em minúsculas, sem espaços '
                    '(ex: "drive.file.read").'
                )
            })

        self.full_clean()
        super().save(*args, **kwargs)

"""
GoogleDriveNormalizer — converte payloads do Google Drive em
ExternalFile.

Puro: sem I/O. Testável sem mock de rede.
"""

from datetime import datetime

from apps.integrations.normalization.contracts import ExternalFile


class GoogleDriveNormalizer:

    @staticmethod
    def to_external_file(raw: dict) -> ExternalFile:
        """
        Converte um dict de arquivo do Google Drive em ExternalFile.
        """
        return ExternalFile(
            external_id=str(raw.get('id', '')),
            name=raw.get('name', ''),
            mime_type=raw.get('mimeType'),
            size_bytes=_parse_int(raw.get('size')),
            parent_ids=tuple(raw.get('parents') or ()),
            path=None,  # Drive não expõe path direto.
            url=raw.get('webViewLink'),
            created_at=_parse_datetime(raw.get('createdTime')),
            modified_at=_parse_datetime(raw.get('modifiedTime')),
            provider_metadata={
                'owners': raw.get('owners', []),
                'shared': raw.get('shared', False),
                'trashed': raw.get('trashed', False),
            },
        )


def _parse_int(value):
    if value is None:
        return None
    try:
        return int(value)
    except (TypeError, ValueError):
        return None


def _parse_datetime(value):
    if not value:
        return None
    try:
        return datetime.fromisoformat(value.replace('Z', '+00:00'))
    except (TypeError, ValueError):
        return None

"""
ProviderConnector — contrato que todo connector de provider cumpre.

O connector encapsula OAuth, tradução recurso→escopo, chamadas HTTP,
paginação, rate limit e erros específicos do provider.

O connector NÃO faz:
    - autorização de usuário
    - isolamento de workspace
    - decisão de qual conta usar

NOTA DE NOMENCLATURA:
    NÃO confundir com `CustomSocialAccountAdapter` de `apps.accounts`,
    que é adapter do django-allauth para login de usuário via Google.
    São coisas diferentes:
        - allauth Adapter → identidade de usuário (login)
        - ProviderConnector → operações com API externa (Drive, Forms)
"""

from abc import ABC, abstractmethod

from apps.integrations.providers.types import OAuthResult


class ProviderConnector(ABC):

    # Identificador canônico do provider. Corresponde a
    # IntegrationProvider.key.
    key: str = ''

    # =====================================================================
    # OAUTH
    # =====================================================================

    @abstractmethod
    def build_auth_url(
        self,
        *,
        redirect_uri: str,
        resources: list,
        state: str,
    ) -> str:
        """
        Gera a URL de autorização do provider.

        `resources` é uma lista de slugs de IntegrationResource.
        O connector traduz cada slug para o escopo OAuth correspondente.

        Levanta UnsupportedResourceError se algum recurso não é
        suportado pelo connector.
        """

    @abstractmethod
    def exchange_code(
        self,
        *,
        code: str,
        redirect_uri: str,
    ) -> OAuthResult:
        """
        Troca code por tokens.

        Levanta:
            - OAuthExchangeError se o provider recusa o code.
            - InsufficientScopesError se faltar recurso obrigatório.
        """

    @abstractmethod
    def refresh(
        self,
        *,
        refresh_token: str,
    ) -> OAuthResult:
        """
        Renova o access_token.

        Levanta:
            - RefreshTokenInvalidError se o refresh_token é inválido.
            - RefreshTransientError se foi erro transitório.
        """

    @abstractmethod
    def revoke(
        self,
        *,
        access_token: str,
        refresh_token: str | None,
    ) -> None:
        """
        Revoga tokens no provider.

        Best-effort: se o provider não responder, o connector registra
        o erro mas não levanta (a revogação local já foi feita).
        """

    # =====================================================================

"""
GoogleConnector — conector para as APIs do Google (Drive, Forms).

Escopo desta versão:
    - build_auth_url  → IMPLEMENTADO (real)
    - exchange_code   → IMPLEMENTADO (real, via HTTP direto)
    - refresh         → IMPLEMENTADO (real)
    - revoke          → IMPLEMENTADO (real)

O connector é registrado em IntegrationsConfig.ready() e resolvido
pelo ProviderRegistry.

NOTA DE NOMENCLATURA:
    NÃO confundir com `apps.accounts.adapters.social.CustomSocialAccountAdapter`,
    que é o adapter do django-allauth para login de usuário via Google.
    São coisas diferentes:

        - allauth Adapter     → identidade (login via Google)
        - GoogleConnector     → operações com Drive/Forms (ferramentas)

    Usam OAuth Clients DIFERENTES (GOOGLE_CLIENT_ID vs.
    INTEGRATIONS_GOOGLE_CLIENT_ID), com escopos diferentes.

DIVISÃO DE RESPONSABILIDADES:
    O connector TRADUZ fielmente a resposta do Google. NÃO aplica
    políticas de negócio. Especificamente:

        1. Se o Google não devolve refresh_token, o connector reporta
           `None`. A camada de serviço decide se a conexão pode
           prosseguir ou se precisa de nova autorização.

        2. Se o usuário concede menos escopos do que os solicitados,
           o connector reporta o que veio em `granted_scopes`. A
           camada de serviço específica (ex: GoogleDriveService)
           valida se o escopo necessário está presente antes de
           chamar a API.

        3. O connector NÃO valida se `granted_scopes` cobre
           `REQUIRED_RESOURCES`. Essa validação é feita por operação.

        4. `refresh` mapeia erros HTTP do Google para exceções
           específicas (RefreshTokenInvalidError vs. RefreshTransientError).
           A decisão de o que fazer com cada uma é do serviço.
"""

    def build_auth_url(
        self,
        *,
        redirect_uri: str,
        resources: list,
        state: str,
        login_hint: str | None = None,
    ) -> str:
        """
        Gera a URL de consentimento do Google.

        Parâmetros:
            redirect_uri — DEVE estar cadastrado no OAuth Client.
            resources    — slugs de IntegrationResource
                           (ex: ["drive.file.read"]).
                           Traduzidos para escopos OAuth via
                           RESOURCE_TO_SCOPE.
            state        — nonce anti-CSRF gerado pelo chamador.
                           Devolvido intacto pelo Google no callback.
            login_hint   — email do usuário logado na Clivo. Quando
                           informado, o Google pré-seleciona a
                           conta correspondente no consentimento,
                           evitando que outras contas logadas no
                           navegador sejam exibidas por padrão.

                           NÃO há acoplamento com `accounts`: o
                           valor vem de `request.user.email`, que
                           é campo padrão do Django.

        Retorna: URL completa de consentimento.

        Levanta:
            UnsupportedResourceError — resource slug não suportado.
            IntegrationConfigurationError — Client OAuth ausente.

        NOTA sobre `prompt='consent'`:
            Forçamos `prompt='consent'` para GARANTIR que o Google
            envie um `refresh_token` novo a cada autorização.

            Sem isso, o Google pode omitir o `refresh_token` quando
            o usuário já autorizou o app anteriormente — o que
            inviabiliza persistir uma integração durável.

            Trade-off: o usuário vê a tela de consentimento sempre.
            Aceitável em desenvolvimento/testes. Em produção, com o
            app verificado e política de reconexão bem definida,
            pode-se considerar remover este parâmetro.

        NOTA sobre `login_hint`:
            O Google trata `login_hint` como SUGESTÃO. O usuário
            ainda pode trocar para outra conta se quiser. Mas o
            caminho padrão passa a ser a conta correta — que é
            justamente a conta com a qual o usuário faz login na
            Clivo.
        """
        scopes = [self.resource_to_scope(r) for r in resources]
        client_config = self._build_client_config()

        flow = Flow.from_client_config(
            client_config=client_config,
            scopes=scopes,
            redirect_uri=redirect_uri,
            state=state,
        )

        # Parâmetros extras enviados ao Google.
        #
        # `access_type='offline'` é necessário para obter refresh_token.
        # `include_granted_scopes='true'` permite reaproveitar
        # consentimentos anteriores (útil para re-autorizações).
        # `prompt='consent'` FORÇA a tela de consentimento a cada
        # autorização — e, com isso, garante o envio de um novo
        # `refresh_token`.
        auth_params = {
            'access_type': 'offline',
            'include_granted_scopes': 'true',
            'prompt': 'consent',
        }

        # Se temos o email do usuário logado na Clivo, usa como
        # `login_hint`. O Google pré-seleciona essa conta.
        if login_hint:
            auth_params['login_hint'] = login_hint

        auth_url, _ = flow.authorization_url(**auth_params)

        logger.debug(
            'Google auth URL gerada | scopes=%s | redirect=%s | '
            'login_hint=%s | prompt=consent',
            scopes, redirect_uri, login_hint or '—',
        )

        return auth_url

    def exchange_code(
        self,
        *,
        code: str,
        redirect_uri: str,
    ) -> OAuthResult:
        """
        Troca um `code` de autorização por tokens OAuth.

        Faz a chamada HTTP DIRETAMENTE ao endpoint de token do
        Google, em vez de usar `Flow.fetch_token`.

        MOTIVO DA MUDANÇA:
            O `Flow.fetch_token` do `google-auth-oauthlib` faz uma
            validação de escopos que dispara `Warning` em vários
            cenários legítimos (ex: o Google devolve `openid`,
            `email`, `profile` além dos escopos solicitados).
            Como o Clivo só se importa com os escopos que
            efetivamente consegue traduzir em recursos, essa
            validação agressiva é desnecessária e quebra o fluxo.

            A chamada HTTP direta devolve exatamente o mesmo
            payload, sem a camada de validação.

        Parâmetros:
            code         — `code` devolvido pelo Google no callback.
            redirect_uri — o MESMO redirect_uri usado em `build_auth_url`.

        Retorna: `OAuthResult` com:
            - access_token
            - refresh_token (pode ser None — ver nota abaixo)
            - expires_at
            - granted_scopes (o que o Google efetivamente concedeu)
            - external_account_id, email, display_name, avatar_url

        Levanta:
            OAuthExchangeError — se o Google recusa a troca.
            IntegrationConfigurationError — Client OAuth ausente.

        NOTA sobre `refresh_token=None`:
            O Google pode não devolver `refresh_token` quando o
            usuário já consentiu antes. Nesse caso, devolvemos `None`.
            A camada de serviço (IntegrationService.upsert_from_oauth)
            é quem decide se preserva um refresh_token anterior ou
            se recusa ativar a conexão.
        """
        client_config = self._build_client_config()
        web_config = client_config['web']

        # Chamada HTTP direta ao endpoint de token do Google.
        # O contrato é idêntico ao do `Flow`, mas sem validação
        # de escopos.
        try:
            response = httpx.post(
                web_config['token_uri'],
                data={
                    'grant_type': 'authorization_code',
                    'code': code,
                    'redirect_uri': redirect_uri,
                    'client_id': web_config['client_id'],
                    'client_secret': web_config['client_secret'],
                },
                headers={
                    'Content-Type': 'application/x-www-form-urlencoded',
                    'Accept': 'application/json',
                },
                timeout=15.0,
            )
        except httpx.HTTPError as exc:
            logger.warning(
                'Falha de rede ao trocar code por tokens (Google): %s',
                type(exc).__name__,
            )
            raise OAuthExchangeError(
                'Não foi possível trocar o código de autorização '
                'por tokens.',
                context={
                    'provider_key': self.key,
                    'error_type': type(exc).__name__,
                },
            ) from exc

        # Se o Google recusou, extrai o motivo para o contexto
        # (sem vazar o code nem os tokens).
        if response.status_code != 200:
            error_code = None
            error_description = None
            try:
                payload = response.json()
                error_code = payload.get('error')
                error_description = payload.get('error_description')
            except Exception:
                pass

            logger.warning(
                'Google recusou exchange_code | status=%s | error=%s',
                response.status_code, error_code,
            )
            raise OAuthExchangeError(
                'O Google recusou a troca do código de autorização '
                'por tokens.',
                context={
                    'provider_key': self.key,
                    'status_code': response.status_code,
                    'error': error_code,
                    'error_description': error_description,
                },
            )

        # Parse do payload JSON.
        try:
            data = response.json()
        except ValueError as exc:
            logger.warning(
                'Google devolveu payload não-JSON em exchange_code',
            )
            raise OAuthExchangeError(
                'O Google devolveu uma resposta inválida.',
                context={
                    'provider_key': self.key,
                    'error_type': 'invalid_json',
                },
            ) from exc

        access_token = data.get('access_token')
        refresh_token = data.get('refresh_token')
        expires_in = data.get('expires_in')
        granted_scopes = data.get('scope', '').split()

        if not access_token:
            logger.warning(
                'Google não devolveu access_token em exchange_code',
            )
            raise OAuthExchangeError(
                'O Google não devolveu o token de acesso.',
                context={'provider_key': self.key},
            )

        # Normaliza expires_at a partir de `expires_in`.
        expires_at = None
        if expires_in:
            try:
                expires_at = datetime.now(timezone.utc) + timedelta(
                    seconds=int(expires_in),
                )
            except (TypeError, ValueError):
                expires_at = None

        # Busca userinfo usando o access_token.
        userinfo = self._fetch_userinfo(access_token)

        # Traduz escopos → recursos Clivo.
        granted_resources = self._scopes_to_resources(granted_scopes)

        logger.info(
            'exchange_code OK | account=%s | has_refresh=%s | scopes=%s',
            userinfo.get('email', '?'),
            bool(refresh_token),
            granted_scopes,
        )

        return OAuthResult(
            access_token=access_token,
            refresh_token=refresh_token,
            expires_at=expires_at,
            granted_scopes=list(granted_scopes),
            external_account_id=userinfo.get('sub', ''),
            external_account_email=userinfo.get('email', ''),
            external_account_display_name=userinfo.get('name', ''),
            external_account_avatar_url=userinfo.get('picture'),
            provider_metadata={
                'granted_resources': granted_resources,
                'userinfo': userinfo,
            },
        )

    def refresh(
        self,
        *,
        refresh_token: str,
    ) -> OAuthResult:
        """
        Renova o access_token usando um refresh_token.

        Parâmetros:
            refresh_token — refresh_token previamente obtido via
                            `exchange_code`.

        Retorna: `OAuthResult` com:
            - access_token (novo)
            - refresh_token (o MESMO que foi passado — o Google
              geralmente não devolve um novo em refresh)
            - expires_at (novo)
            - granted_scopes (o Google pode devolver os mesmos
              escopos; usamos o que vier)
            - external_account_* (o Google NÃO devolve userinfo em
              refresh; esses campos vêm vazios)

        Levanta:
            RefreshTokenInvalidError — refresh_token revogado,
                expirado ou inválido. O usuário precisa reconectar.
            RefreshTransientError — erro transitório (rede, 5xx,
                rate limit). Retry é possível.
            IntegrationConfigurationError — Client OAuth ausente.

        NOTA sobre os campos de identidade:
            Um refresh NÃO devolve `sub`, `email`, etc. Esses campos
            vêm vazios no `OAuthResult` de refresh. O chamador
            (IntegrationService) NÃO usa esse resultado para atualizar
            a identidade da integração — só os tokens.

        NOTA sobre rotação de refresh_token:
            O Google geralmente NÃO rotaciona o refresh_token. Mas
            se ele devolver um novo (o que é raro, mas acontece em
            algumas políticas de segurança), usamos o novo. O
            chamador deve sempre persistir o refresh_token devolvido,
            mesmo que seja igual ao anterior.
        """
        client_config = self._build_client_config()

        # Método 1: usar google-auth (mais robusto).
        # Tentamos primeiro este caminho porque ele mapeia erros
        # de forma mais estruturada.
        try:
            return self._refresh_via_google_auth(
                client_config=client_config,
                refresh_token=refresh_token,
            )
        except RefreshError as exc:
            # google-auth devolve RefreshError. Precisamos distinguir
            # "inválido permanentemente" de "transitório".
            message = str(exc).lower()

            # O Google devolve "invalid_grant" para refresh_token
            # inválido/expirado/revogado.
            if 'invalid_grant' in message or 'invalid_client' in message:
                logger.warning(
                    'Refresh token inválido (Google): %s',
                    type(exc).__name__,
                )
                raise RefreshTokenInvalidError(
                    'A credencial de longa duração foi revogada ou '
                    'expirou. O usuário precisa reconectar a integração.',
                    context={'provider_key': self.key},
                ) from exc

            # Qualquer outra coisa é transitória (rede, 5xx, rate
            # limit).
            logger.warning(
                'Falha transitória no refresh (Google): %s',
                type(exc).__name__,
            )
            raise RefreshTransientError(
                'Falha temporária ao renovar a credencial. '
                'Tente novamente.',
                context={'provider_key': self.key},
            ) from exc

    def revoke(
        self,
        *,
        access_token: str,
        refresh_token: str | None,
    ) -> None:
        """
        Revoga tokens no Google.

        Estratégia:
            - Revoga AMBOS os tokens (access e refresh) via endpoint
              /revoke do Google.
            - Best-effort: se o Google não responder, loga mas NÃO
              levanta. A revogação LOCAL é responsabilidade do
              IntegrationService — o Google ser notificado é o ideal,
              mas não bloqueia o fluxo do usuário.

        Parâmetros:
            access_token  — access_token atual.
            refresh_token — refresh_token (pode ser None).

        NOTA sobre idempotência:
            Se o token já foi revogado no Google, o endpoint devolve
            200 (idempotente). Não é erro.
        """
        tokens_to_revoke = []
        if refresh_token:
            tokens_to_revoke.append(refresh_token)
        if access_token:
            tokens_to_revoke.append(access_token)

        if not tokens_to_revoke:
            return

        for token in tokens_to_revoke:
            try:
                response = httpx.post(
                    self.GOOGLE_REVOKE_URI,
                    data={'token': token},
                    headers={
                        'Content-Type': 'application/x-www-form-urlencoded',
                    },
                    timeout=10.0,
                )
            except httpx.HTTPError as exc:
                logger.warning(
                    'Falha de rede ao revogar token no Google: %s',
                    type(exc).__name__,
                )
                continue

            if response.status_code >= 500:
                logger.warning(
                    'Google devolveu %s ao revogar token',
                    response.status_code,
                )
                continue

            # 200 = revogado (ou já revogado).
            # 400 = token inválido (já revogado ou nunca existiu).
            # Ambos são "efeito desejado alcançado".
            if response.status_code not in (200, 400):
                logger.warning(
                    'Google devolveu %s ao revogar token',
                    response.status_code,
                )

        logger.info(
            'Revogação no Google concluída (best-effort) | tokens=%d',
            len(tokens_to_revoke),
        )

    # MAPEAMENTO
    # =====================================================================

    def resource_to_scope(self, resource_slug: str) -> str:
        scope = self.RESOURCE_TO_SCOPE.get(resource_slug)
        if scope is None:
            raise UnsupportedResourceError(
                f'Recurso "{resource_slug}" não suportado pelo GoogleConnector.',
                context={
                    'provider_key': self.key,
                    'resource_slug': resource_slug,
                },
            )
        return scope

    def required_resources_for_operation(self, operation: str) -> list:
        return list(self.REQUIRED_RESOURCES.get(operation, []))

    def optional_resources_for_operation(self, operation: str) -> list:
        return list(self.OPTIONAL_RESOURCES.get(operation, []))

    # HELPERS INTERNOS
    # =====================================================================

    def _refresh_via_google_auth(
        self,
        *,
        client_config: dict,
        refresh_token: str,
    ) -> OAuthResult:
        """
        Renova o token usando `google.oauth2.credentials.Credentials`.

        Este caminho é preferível ao refresh manual via HTTP porque
        o `google-auth` mapeia os erros do Google de forma estruturada
        (RefreshError), permitindo distinguir "inválido" de
        "transitório" com precisão.
        """
        web_config = client_config['web']

        credentials = GoogleCredentials(
            token=None,
            refresh_token=refresh_token,
            token_uri=web_config['token_uri'],
            client_id=web_config['client_id'],
            client_secret=web_config['client_secret'],
            scopes=[],
        )

        # Este é o ponto onde o RefreshError pode ser levantado.
        # Não capturamos aqui — deixamos subir para `refresh()`, que
        # traduz para as exceções do domínio.
        credentials.refresh(GoogleAuthRequest())

        return OAuthResult(
            access_token=credentials.token,
            refresh_token=credentials.refresh_token or refresh_token,
            expires_at=self._normalize_expiry(credentials.expiry),
            granted_scopes=list(credentials.scopes or []),
            # Campos de identidade NÃO vêm em refresh.
            external_account_id='',
            external_account_email='',
            external_account_display_name='',
            external_account_avatar_url=None,
            provider_metadata={
                'granted_resources': self._scopes_to_resources(
                    credentials.scopes or []
                ),
                'source': 'refresh',
            },
        )

    def _build_client_config(self) -> dict:
        """
        Monta o dict no formato que `Flow.from_client_config` espera.

        Lê de INTEGRATIONS_GOOGLE_CLIENT_ID (dedicado ao integrations),
        NÃO de GOOGLE_CLIENT_ID (allauth).
        """
        client_id = getattr(
            settings, 'INTEGRATIONS_GOOGLE_CLIENT_ID', None,
        )
        client_secret = getattr(
            settings, 'INTEGRATIONS_GOOGLE_CLIENT_SECRET', None,
        )

        if not client_id:
            raise IntegrationConfigurationError(
                'INTEGRATIONS_GOOGLE_CLIENT_ID não está configurado.',
                context={
                    'provider_key': self.key,
                    'missing': 'client_id',
                },
            )
        if not client_secret:
            raise IntegrationConfigurationError(
                'INTEGRATIONS_GOOGLE_CLIENT_SECRET não está configurado.',
                context={
                    'provider_key': self.key,
                    'missing': 'client_secret',
                },
            )

        return {
            'web': {
                'client_id': client_id,
                'client_secret': client_secret,
                'auth_uri': self.GOOGLE_AUTH_URI,
                'token_uri': self.GOOGLE_TOKEN_URI,
            }
        }

    def _fetch_userinfo(self, access_token: str) -> dict:
        """
        Busca informações da conta externa via Google UserInfo API.

        Não usa `google-api-python-client` — usa HTTP direto.
        Isso mantém a superfície de dependências pequena.

        Levanta OAuthExchangeError se o Google recusar.
        """
        try:
            response = httpx.get(
                self.GOOGLE_USERINFO_URI,
                headers={'Authorization': f'Bearer {access_token}'},
                timeout=10.0,
            )
        except httpx.HTTPError as exc:
            logger.warning(
                'Falha ao buscar userinfo (Google): %s',
                type(exc).__name__,
            )
            raise OAuthExchangeError(
                'Não foi possível obter dados da conta Google.',
                context={
                    'provider_key': self.key,
                    'error_type': type(exc).__name__,
                },
            ) from exc

        if response.status_code != 200:
            logger.warning(
                'Google userinfo retornou %s',
                response.status_code,
            )
            raise OAuthExchangeError(
                'O Google recusou a consulta de userinfo.',
                context={
                    'provider_key': self.key,
                    'status_code': response.status_code,
                },
            )

        return response.json()

    def _scopes_to_resources(self, scopes: list) -> list:
        """
        Traduz uma lista de escopos OAuth do Google para slugs de
        recursos Clivo.

        Escopos que não estão no mapa são ignorados silenciosamente.
        Isso é esperado: `include_granted_scopes=true` pode devolver
        escopos concedidos em consentimentos anteriores.
        """
        resources = []
        for scope in scopes:
            resource = self.SCOPE_TO_RESOURCE.get(scope)
            if resource is not None:
                resources.append(resource)
        return resources

    @staticmethod
    def _normalize_expiry(expiry) -> datetime | None:
        """
        Normaliza o `expiry` do `google.auth.credentials.Credentials`.

        O Google devolve um `datetime` naive (sem tzinfo). Convertemos
        para UTC, para não gerar ambiguidade quando comparado com
        `timezone.now()`.
        """
        if expiry is None:
            return None
        if expiry.tzinfo is None:
            return expiry.replace(tzinfo=timezone.utc)
        return expiry

class ProviderRegistry:
    """
    Registro em memória. Não é thread-safe por design; registro
    acontece em ready(), antes de qualquer request.
    """

    _connectors: dict = {}

    @classmethod
    def register(cls, connector, *, override: bool = False) -> None:
        """
        Registra um connector.

        Se a chave já existe e override=False, levanta DuplicateConnectorError.
        Se override=True, sobrescreve silenciosamente (uso em testes).
        """
        key = connector.key
        if key in cls._connectors and not override:
            raise DuplicateConnectorError(
                f'Connector "{key}" já registrado. '
                f'Use override=True apenas em testes.',
                context={'provider_key': key},
            )
        cls._connectors[key] = connector

    @classmethod
    def get(cls, key: str):
        from apps.integrations.exceptions import ConnectorNotFoundError
        connector = cls._connectors.get(key)
        if connector is None:
            raise ConnectorNotFoundError(
                f'Connector "{key}" não registrado.',
                context={'provider_key': key},
            )
        return connector

    @classmethod
    def all_keys(cls) -> list:
        return list(cls._connectors.keys())

    @classmethod
    def clear(cls) -> None:
        """Uso em testes."""
        cls._connectors.clear()

"""
WorkspaceIntegrationService — API interna para apps consumidores.

NÃO é uma API REST. É uma API Python que os apps do Clivo usam
diretamente (ex: `from apps.integrations.services.access import
WorkspaceIntegrationService`).

SEM NOÇÃO DE consumer_app:
    Não há registro de "qual app pode usar qual recurso".
    Se o workspace tem uma integração ativa, qualquer app com acesso
    ao workspace pode usá-la.

SELEÇÃO DE CONTA:
    Regra determinística, sem `.first()` arbitrário.

CONCORRÊNCIA:
    A auto-marcação de `is_preferred` acontece dentro de
    `transaction.atomic()` + `select_for_update()`.
"""

import logging

from django.db import IntegrityError, transaction

from apps.integrations.exceptions import (
    AmbiguousConnectionError,
    ConnectionNotFoundError,
    ConnectionNotEstablishedError,
    IntegrationInactiveError,
)
from apps.integrations.models import WorkspaceIntegration
from apps.integrations.providers.registry import ProviderRegistry

logger = logging.getLogger('integrations')


class WorkspaceIntegrationService:
    """
    Resolve a integração ativa do workspace com o provider.
    """

    @staticmethod
    def get_for_workspace(
        *,
        workspace,
        provider_key: str,
        integration_id=None,
    ) -> WorkspaceIntegration:
        """
        Retorna a WorkspaceIntegration ativa do workspace com o provider.

        Regras de seleção (aplicadas nesta ordem):

            1. Se `integration_id` explícito:
               - Valida que pertence ao workspace.
               - Valida que `status='active'`.
               - Retorna.

            2. Se não foi passado `integration_id`:
               2a. Se existe uma conta com `is_preferred=True,
                   status='active'` → retorna.
               2b. Se existe exatamente uma conta ativa → retorna e
                   marca como preferida.
               2c. Se existem múltiplas ativas sem preferida →
                   AmbiguousConnectionError.
               2d. Se não existe nenhuma ativa →
                   ConnectionNotEstablishedError.

        Levanta:
            ConnectionNotFoundError
            IntegrationInactiveError
            ConnectionNotEstablishedError
            AmbiguousConnectionError
        """
        if integration_id is not None:
            return WorkspaceIntegrationService._resolve_explicit(
                workspace=workspace,
                integration_id=integration_id,
            )
        return WorkspaceIntegrationService._resolve_by_preference(
            workspace=workspace,
            provider_key=provider_key,
        )

    @staticmethod
    def _resolve_explicit(
        *,
        workspace,
        integration_id,
    ) -> WorkspaceIntegration:
        try:
            integration = WorkspaceIntegration.objects.get(
                id=integration_id,
                workspace=workspace,
            )
        except (WorkspaceIntegration.DoesNotExist, ValueError):
            raise ConnectionNotFoundError(
                'Integração não encontrada neste workspace.',
                context={
                    'workspace_id': str(workspace.pk),
                    'integration_id': str(integration_id),
                },
            )

        if not integration.is_usable():
            raise IntegrationInactiveError(
                f'Integração está no estado "{integration.status}" '
                f'e não pode ser usada.',
                context={
                    'integration_id': str(integration.pk),
                    'status': integration.status,
                },
            )

        return integration

    @staticmethod
    def _resolve_by_preference(
        *,
        workspace,
        provider_key: str,
    ) -> WorkspaceIntegration:
        with transaction.atomic():
            candidates = list(
                WorkspaceIntegration.objects
                .select_for_update()
                .filter(
                    workspace=workspace,
                    provider__key=provider_key,
                    status=WorkspaceIntegration.STATUS_ACTIVE,
                )
                .order_by('-created_at')
            )

            if not candidates:
                raise ConnectionNotEstablishedError(
                    f'Nenhuma integração ativa do provider '
                    f'"{provider_key}" neste workspace.',
                    context={
                        'workspace_id': str(workspace.pk),
                        'provider_key': provider_key,
                    },
                )

            preferred = [c for c in candidates if c.is_preferred]

            if len(preferred) == 1:
                return preferred[0]

            if len(preferred) > 1:
                logger.warning(
                    'Estado inconsistente: múltiplas integrações '
                    'preferidas para workspace=%s provider=%s',
                    workspace.pk, provider_key,
                )
                raise AmbiguousConnectionError(
                    f'Múltiplas integrações preferidas para '
                    f'"{provider_key}" (estado inconsistente).',
                    context={
                        'workspace_id': str(workspace.pk),
                        'provider_key': provider_key,
                        'count': len(preferred),
                    },
                )

            if len(candidates) == 1:
                integration = candidates[0]
                if not integration.is_preferred:
                    try:
                        integration.is_preferred = True
                        integration.save(
                            update_fields=['is_preferred', 'updated_at'],
                        )
                    except IntegrityError:
                        logger.warning(
                            'Race na marcação de preferida para '
                            'integration=%s. Revalidando estado.',
                            integration.pk,
                        )
                        integration.refresh_from_db()
                        if not integration.is_preferred:
                            raise AmbiguousConnectionError(
                                'Estado inconsistente após race na '
                                'marcação de preferida.',
                                context={
                                    'workspace_id': str(workspace.pk),
                                    'provider_key': provider_key,
                                    'integration_id': str(integration.pk),
                                },
                            )
                return integration

            raise AmbiguousConnectionError(
                f'Múltiplas integrações ativas do provider '
                f'"{provider_key}" sem preferida. '
                f'Informe `integration_id` explicitamente.',
                context={
                    'workspace_id': str(workspace.pk),
                    'provider_key': provider_key,
                    'count': len(candidates),
                },
            )

    @staticmethod
    def get_connector(provider_key: str):
        """Atalho para o ProviderRegistry."""
        return ProviderRegistry.get(provider_key)

    @staticmethod
    def list_active_for_workspace(*, workspace, provider_key: str = None):
        """
        Lista integrações ativas do workspace.

        Se `provider_key` é passado, filtra por provider.
        """
        qs = (
            WorkspaceIntegration.objects
            .filter(
                workspace=workspace,
                status=WorkspaceIntegration.STATUS_ACTIVE,
            )
            .select_related('provider')
            .order_by('-is_preferred', '-created_at')
        )
        if provider_key is not None:
            qs = qs.filter(provider__key=provider_key)
        return list(qs)

        from django.core.exceptions import ValidationError

        form_id = GoogleFormsService._extract_form_id(form_url_or_id)
        if not form_id:
            raise ValidationError(
                'URL do Google Forms inválida. Cole o link do formulário.'

    # =====================================================================
    # HELPERS DE URL

    @staticmethod
    def _extract_form_id(url_or_id: str) -> Optional[str]:
        if not url_or_id:
            return None
        value = url_or_id.strip()

        match = _FORMS_ID_PATTERN.search(value)
        if match:
            return match.group(1)

        if _FORMS_SHORT_URL_PATTERN.match(value):
            logger.info(
                'URL encurtada do Forms (não suportada no V1): %s', value,
            )
            return None

        if re.match(r'^[A-Za-z0-9_-]{20,}$', value):
            return value

        return None

    # =====================================================================
    # CHAMADAS À API DO GOOGLE
    # =====================================================================

    @staticmethod
    def _fetch_forms_list(access_token: str) -> list[dict]:
        """Lista os formulários do usuário via Drive API."""
        params = {
            'q': (
                f"mimeType='{GOOGLE_FORMS_MIME_TYPE}' and trashed=false"
            ),
            'fields': (
                'files(id,name,description,webViewLink,'
                'createdTime,modifiedTime),nextPageToken'
            ),
            'pageSize': 100,
            'orderBy': 'modifiedTime desc',
        }

        all_files: list = []
        page_token = None

        while True:
            if page_token:
                params['pageToken'] = page_token
            else:
                params.pop('pageToken', None)

            try:
                response = httpx.get(
                    DRIVE_FILES_URL,
                    headers={'Authorization': f'Bearer {access_token}'},
                    params=params,
                    timeout=HTTP_TIMEOUT_SECONDS,
                )
            except httpx.TimeoutException as exc:
                raise ProviderTimeoutError(
                    'A API do Google Drive não respondeu a tempo.',
                    context={
                        'provider_key': PROVIDER_KEY,
                        'timeout_s': int(HTTP_TIMEOUT_SECONDS),
                    },
                ) from exc
            except httpx.HTTPError as exc:
                raise ProviderServerError(
                    'Falha de rede ao consultar o Google Drive.',
                    context={'provider_key': PROVIDER_KEY},
                ) from exc

            GoogleFormsService._raise_for_status(response)

            try:
                data = response.json()
            except ValueError:
                logger.warning(
                    'Google Drive retornou payload não-JSON em /files'
                )
                return all_files

            files = data.get('files') or []
            if isinstance(files, list):
                all_files.extend(files)

            page_token = data.get('nextPageToken')
            if not page_token:
                break

        normalized = []
        for f in all_files:
            normalized.append({
                'formId': f.get('id', ''),
                'info': {
                    'title': f.get('name', ''),
                    'description': f.get('description', ''),
                },
                'responderUri': f.get('webViewLink', ''),
            })

        logger.info(
            'Drive API retornou %d formulários', len(normalized),
        )
        return normalized

    @staticmethod
    def _fetch_form_payload(*, access_token: str, form_id: str) -> dict:
        """Busca o payload completo de um formulário (Forms API get)."""
        url = FORMS_GET_URL_TEMPLATE.format(form_id=form_id)

        try:
            response = httpx.get(
                url,
                headers={'Authorization': f'Bearer {access_token}'},
                timeout=HTTP_TIMEOUT_SECONDS,
            )
        except httpx.TimeoutException as exc:
            raise ProviderTimeoutError(
                'A API do Google Forms não respondeu a tempo.',
                context={
                    'provider_key': PROVIDER_KEY,
                    'form_id': form_id,
                    'timeout_s': int(HTTP_TIMEOUT_SECONDS),
                },
            ) from exc
        except httpx.HTTPError as exc:
            raise ProviderServerError(
                'Falha de rede ao consultar o Google Forms.',
                context={'provider_key': PROVIDER_KEY, 'form_id': form_id},
            ) from exc

        GoogleFormsService._raise_for_status(response)

        try:
            data = response.json()
        except ValueError as exc:
            raise IntegrationError(
                'O Google Forms devolveu uma resposta inválida.',
                context={
                    'provider_key': PROVIDER_KEY,
                    'form_id': form_id,
                    'error_type': 'invalid_json',
                },
            ) from exc

        if not isinstance(data, dict):
            raise IntegrationError(
                'O Google Forms devolveu um payload inesperado.',
                context={
                    'provider_key': PROVIDER_KEY,
                    'form_id': form_id,
                    'error_type': 'unexpected_payload_type',
                },
            )

        items = data.get('items') or []
        info = data.get('info') or {}
        logger.info(
            'Form payload obtido | form_id=%s | title=%r | items=%d',
            form_id,
            info.get('title', ''),
            len(items) if isinstance(items, list) else 0,
        )
        return data

    @staticmethod
    def _download_image_bytes(
        *,
        access_token: str,
        image_payload: dict,
    ) -> Optional[tuple]:
        """
        Baixa os bytes de uma imagem do Google Forms.

        Aceita `contentUri` (link temporário baixável) ou
        `sourceUri` (referência interna). Retorna (bytes, mime_type)
        ou None se falhar.
        """
        if not isinstance(image_payload, dict):
            return None

        url = (
            image_payload.get('contentUri')
            or image_payload.get('sourceUri')
        )
        if not url:
            return None

        try:
            response = httpx.get(
                url,
                headers={'Authorization': f'Bearer {access_token}'},
                timeout=IMAGE_DOWNLOAD_TIMEOUT_SECONDS,
                follow_redirects=True,
            )
        except httpx.HTTPError as exc:
            logger.warning(
                'Falha de rede ao baixar imagem do Forms: %s',
                type(exc).__name__,
            )
            return None

        if response.status_code != 200:
            logger.warning(
                'Google devolveu %s ao baixar imagem do Forms',
                response.status_code,
            )
            return None

        content_type = (
            response.headers.get('Content-Type')
            or 'image/png'
        )
        content_type = content_type.split(';')[0].strip()

        return response.content, content_type

    @staticmethod
    def _raise_for_status(response: httpx.Response) -> None:
        status = response.status_code

        if 200 <= status < 300:
            return

        if status == 401:
            raise ConnectionExpiredError(
                'A credencial do Google foi recusada. Reconecte.',
                context={'provider_key': PROVIDER_KEY, 'status_code': 401},
            )

        if status == 403:
            raise FormsScopeMissingError(
                'A conta Google conectada não tem permissão para '
                'acessar os formulários. Reconecte autorizando o '
                'acesso ao Google Forms.',
                context={'provider_key': PROVIDER_KEY, 'status_code': 403},
            )

        if status == 404:
            raise IntegrationError(
                'Recurso não encontrado no Google.',
                context={'provider_key': PROVIDER_KEY, 'status_code': 404},
            )

        if status == 429:
            retry_after = response.headers.get('Retry-After')
            try:
                retry_after_s = int(retry_after) if retry_after else None
            except (TypeError, ValueError):
                retry_after_s = None
            raise RateLimitError(
                'A API do Google está limitando requisições.',
                context={
                    'provider_key': PROVIDER_KEY,
                    'retry_after_s': retry_after_s,
                },
            )

        if 500 <= status < 600:
            raise ProviderServerError(
                f'Erro {status} na API do Google.',
                context={'provider_key': PROVIDER_KEY, 'status_code': status},
            )

        raise IntegrationError(
            f'Erro {status} na API do Google.',
            context={'provider_key': PROVIDER_KEY, 'status_code': status},
        )

    @staticmethod
    def _normalize_form(raw: dict) -> dict:
        """
        Converte o payload completo de um formulário no contrato
        interno do Clivo.

        Preserva as imagens das opções no campo `_raw_image` (dict
        cru vindo do Google). O download real acontece depois, na
        criação do template.
        """
        info = raw.get('info') or {}
        items = raw.get('items') or []

        sections = []
        current = {
            'title': (info.get('title') or '').strip() or 'Formulário',
            'description': (info.get('description') or '').strip(),
            'questions': [],
        }

        for item in items:
            title = (item.get('title') or '').strip()

            if 'pageBreakItem' in item:
                if current['questions'] or sections:
                    sections.append(current)
                current = {
                    'title': title or f'Seção {len(sections) + 1}',
                    'description': (item.get('description') or '').strip(),
                    'questions': [],
                }
                continue

            if 'questionGroupItem' in item:
                grid_questions = (
                    GoogleFormsService._normalize_question_group(item)
                )
                current['questions'].extend(grid_questions)
                continue

            question = GoogleFormsService._normalize_question(item)
            if question is not None:
                current['questions'].append(question)

        if current['questions'] or not sections:
            sections.append(current)

        return {
            'id': str(raw.get('formId') or ''),
            'name': (
                (info.get('title') or '').strip()
                or 'Formulário importado'
            ),
            'description': (info.get('description') or '').strip(),
            'sections': sections,
        }

    @staticmethod
    def _normalize_question(item: dict) -> Optional[dict]:
        """Converte um item de pergunta em uma pergunta do Clivo."""
        qi = item.get('questionItem')
        if not qi:
            return None

        question = qi.get('question') or {}
        title = (item.get('title') or '').strip()
        description = (item.get('description') or '').strip()

        if not title:
            return None

        is_required = bool(question.get('required', False))
        question_type, options = GoogleFormsService._map_question_type(
            question,
        )

        raw_q_image = (
            qi.get('image') if isinstance(qi.get('image'), dict)
            else None
        )

        return {
            'text': title,
            'instructions': description,
            'question_type': question_type,
            'allowed_response_formats': (
                GoogleFormsService._formats_for_type(question_type)
            ),
            'options': options,
            'is_required': is_required,
            '_raw_question_image': raw_q_image,
            '_is_grid': False,
        }

    @staticmethod
    def _normalize_question_group(item: dict) -> list:
        """
        Converte um QuestionGroupItem em perguntas do Clivo.

        Caso A — GRID: gera UMA pergunta `grid_single` ou
                 `grid_multiple` com options {rows, columns}.
        Caso B — agrupamento visual: processa cada pergunta
                 interna individualmente.
        """
        qgi = item.get('questionGroupItem') or {}
        group_title = (item.get('title') or '').strip()
        group_description = (item.get('description') or '').strip()

        # -------- CASO A: GRID --------
        if 'grid' in qgi:
            grid = qgi['grid'] or {}
            columns = grid.get('columns') or {}
            col_type = columns.get('type', 'RADIO')
            raw_col_options = columns.get('options') or []

            clean_columns = []
            for opt in raw_col_options:
                text = (opt.get('value') or '').strip()
                if not text:
                    continue
                clean_columns.append({
                    'text': text,
                    '_raw_image': (
                        opt.get('image')
                        if isinstance(opt.get('image'), dict)
                        else None
                    ),
                })

            clean_rows = []
            for row in (qgi.get('questions') or []):
                row_title = (
                    (row.get('rowQuestion') or {}).get('title') or ''
                ).strip()
                if row_title:
                    clean_rows.append(row_title)

            clivo_type = (
                'grid_multiple' if col_type == 'CHECKBOX'
                else 'grid_single'
            )

            return [{
                'text': group_title or 'Grade',
                'instructions': group_description,
                'question_type': clivo_type,
                'allowed_response_formats': ['text'],
                'options': {
                    'rows': clean_rows,
                    'columns': clean_columns,
                },
                'is_required': False,
                '_raw_question_image': None,
                '_is_grid': True,
            }]

        # -------- CASO B: agrupamento visual --------
        result = []
        for question in (qgi.get('questions') or []):
            q_type, q_opts = GoogleFormsService._map_question_type(question)
            title = (
                (question.get('rowQuestion') or {}).get('title')
                or group_title
            )
            if not title:
                continue
            result.append({
                'text': title,
                'instructions': group_description,
                'question_type': q_type,
                'allowed_response_formats': (
                    GoogleFormsService._formats_for_type(q_type)
                ),
                'options': q_opts,
                'is_required': bool(question.get('required', False)),
                '_raw_question_image': None,
                '_is_grid': False,
            })
        return result

    @staticmethod
    def _map_question_type(question: dict) -> tuple:
        """
        Mapeia o tipo do Google Forms para (question_type, options).

        Referência:
            textQuestion (paragraph=false) → short_text
            textQuestion (paragraph=true)  → long_text
            choiceQuestion RADIO           → single_choice
            choiceQuestion CHECKBOX        → multiple_choice
            choiceQuestion DROP_DOWN       → dropdown
            scaleQuestion                  → scale
            ratingQuestion                 → rating
            dateQuestion                   → date
            timeQuestion                   → time
            fileUploadQuestion             → file_upload
            rowQuestion                    → text (pergunta interna)
        """
        if 'scaleQuestion' in question:
            return 'scale', []

        if 'dateQuestion' in question:
            return 'date', []

        if 'timeQuestion' in question:
            return 'time', []

        if 'fileUploadQuestion' in question:
            return 'file_upload', []

        if 'ratingQuestion' in question:
            return 'rating', []

        if 'choiceQuestion' in question:
            cq = question['choiceQuestion']
            g_type = cq.get('type', 'RADIO')
            options = []
            for opt in (cq.get('options') or []):
                text = (opt.get('value') or '').strip()
                if not text:
                    continue
                options.append({
                    'text': text,
                    '_raw_image': (
                        opt.get('image')
                        if isinstance(opt.get('image'), dict)
                        else None
                    ),
                })

            if g_type == 'CHECKBOX':
                return 'multiple_choice', options
            if g_type == 'DROP_DOWN':
                return 'dropdown', options
            return 'single_choice', options

        if 'textQuestion' in question:
            tq = question['textQuestion']
            if tq.get('paragraph'):
                return 'long_text', []
            return 'short_text', []

        if 'rowQuestion' in question:
            return 'text', []

        return 'text', []

    @staticmethod
    def _formats_for_type(question_type: str) -> list:
        """
        Formatos de resposta aceitos por tipo.

        Tipos ESTRUTURADOS (escolha, escala, classificação, data,
        horário, número, sim/não, URL, upload, grid) só aceitam
        `text`. Tipos ABERTOS (texto curto, texto longo, texto
        livre) aceitam texto, áudio e imagem.
        """
        structured = {
            'single_choice',
            'multiple_choice',
            'dropdown',
            'choice',
            'scale',
            'rating',
            'number',
            'grid_single',
            'grid_multiple',
            'date',
            'time',
            'yes_no',
            'url',
            'file_upload',
        }
        if question_type in structured:
            return ['text']
        return ['text', 'audio', 'image']

    @staticmethod
    def _import_image_as_document(
        *,
        access_token: str,
        image_payload: dict,
        workspace,
        actor=None,
        fallback_name: str = 'Imagem importada do Forms',
    ) -> Optional[str]:
        """
        Baixa uma imagem do Google Forms e a salva como Document
        no workspace. Retorna o ID (str) do Document, ou None se
        falhar. Nunca levanta exceção — uma imagem que falha não
        deve interromper a importação do template.
        """
        from apps.documents.models import Document
        from django.core.files.base import ContentFile

        downloaded = GoogleFormsService._download_image_bytes(
            access_token=access_token,
            image_payload=image_payload,
        )
        if downloaded is None:
            return None

        image_bytes, mime_type = downloaded

        ext_map = {
            'image/png': 'png',
            'image/jpeg': 'jpg',
            'image/jpg': 'jpg',
            'image/gif': 'gif',
            'image/webp': 'webp',
            'image/bmp': 'bmp',
        }
        extension = ext_map.get(mime_type, 'png')

        safe_name = re.sub(r'[^A-Za-z0-9._-]+', '-', fallback_name)[:80]
        file_name = f'{safe_name}.{extension}'

        try:
            document = Document.objects.create(

                name=fallback_name[:200],
                file=ContentFile(image_bytes, name=file_name),
                extension=extension,
                mime_type=mime_type,

                created_by=actor if (
                    actor and getattr(actor, 'is_authenticated', False)

            )
        except Exception:

                'Falha ao salvar imagem como Document (name=%s)',
                fallback_name,
            )
            return None

        return str(document.id)

"""
IntegrationService — ciclo de vida da WorkspaceIntegration.

Este serviço aplica as políticas de negócio que o GoogleConnector
deliberadamente NÃO aplica:

    1. Preservação de refresh_token anterior válido.
    2. Não ativar conexão sem refresh_token utilizável.
    3. Validação de escopos concedidos antes de habilitar operações.
    4. Renovação preguiçosa de access_token (on-demand).
    5. Revogação local + best-effort no provider.
    6. Marcação de estado (expired) quando refresh falha.

Também cuida da cifra dos tokens antes de persistir.

MODELO DE CONCORRÊNCIA (refresh):
    Várias partes do produto podem precisar do access_token válido
    ao mesmo tempo (ex: várias chamadas à API do Drive em paralelo).
    Se cada uma disparasse um refresh, teríamos tempestade de
    requisições ao Google.

    Estratégia: `refresh_credentials` roda dentro de uma transação
    com `select_for_update` na integração. A primeira chamada renova;
    as concorrentes esperam, veem o token novo e saem sem renovar.

    Em SQLite o `select_for_update` não tem efeito — o que é
    aceitável, porque SQLite é usado apenas em dev/test. Em
    PostgreSQL (produção) a serialização é real.

MODELO DE "RAISE OUTSIDE ATOMIC":
    Marcações de estado de erro (`_mark_expired`, `_mark_error`,
    `_record_transient_error`) NÃO podem acontecer dentro do
    `transaction.atomic()` de `refresh_credentials`, porque o
    rollback da exceção desfaria a marcação.

    Por isso, o padrão adotado é:

        1. Capturar a exceção dentro do atomic.
        2. Guardar `pending_state_update` e `pending_exception`.
        3. Sair do atomic (commit acontece).
        4. Marcar o estado FORA do atomic (persiste).
        5. Re-levantar a exceção guardada.

    Isso garante que a integração vai para `expired` mesmo com o
    refresh falhando, sem que o rollback apague essa transição.
"""

    @staticmethod
    @transaction.atomic
    def upsert_from_oauth(
        *,
        workspace,
        provider: IntegrationProvider,
        oauth_result: OAuthResult,
        requested_resources: list,
        actor,
    ) -> WorkspaceIntegration:
        """
        Cria ou atualiza uma WorkspaceIntegration a partir de um
        OAuthResult.

        Políticas aplicadas:

            1. Se `oauth_result.refresh_token` é None, preserva o
               refresh_token da integração existente (se houver).

            2. Se, depois da preservação, ainda não há refresh_token
               utilizável, levanta `WorkspaceIntegrationError`.
               A conexão NÃO é ativada.

            3. Valida que os escopos concedidos cobrem os recursos
               solicitados. Se faltar algum, levanta
               `InsufficientScopesError`.

        Levanta:
            WorkspaceIntegrationError — sem refresh_token utilizável.
            InsufficientScopesError — faltam recursos solicitados.
        """

        # -----------------------------------------------------------------
        # 1. Localiza integração existente (mesma conta externa).
        # -----------------------------------------------------------------
        existing = WorkspaceIntegration.objects.filter(
            workspace=workspace,
            provider=provider,
            external_account_id=oauth_result.external_account_id,
        ).first()

        # -----------------------------------------------------------------
        # 2. Preservação de refresh_token.
        # -----------------------------------------------------------------
        refresh_token = oauth_result.refresh_token

        if refresh_token is None and existing is not None:
            refresh_token = IntegrationService._extract_refresh_token(
                existing
            )
            if refresh_token:
                logger.info(
                    'Preservando refresh_token existente para '
                    'workspace=%s provider=%s account=%s',
                    workspace.pk, provider.key,
                    oauth_result.external_account_id,
                )

        # -----------------------------------------------------------------
        # 3. Se ainda não há refresh_token utilizável, NÃO ativa.
        # -----------------------------------------------------------------
        if not refresh_token:
            raise WorkspaceIntegrationError(
                'A conexão não pode ser ativada: o Google não '
                'retornou credencial de longa duração e não há uma '
                'anterior válida. Reconecte e autorize novamente.',
                context={
                    'workspace_id': str(workspace.pk),
                    'provider_key': provider.key,
                    'reason': 'no_refresh_token',
                },
            )

        # -----------------------------------------------------------------
        # 4. Validar escopos concedidos.
        # -----------------------------------------------------------------
        granted_resources = set(
            oauth_result.provider_metadata.get(
                'granted_resources', []
            )
        )
        requested = set(requested_resources)
        missing = requested - granted_resources

        if missing:
            raise InsufficientScopesError(
                'A conexão não pode ser ativada: alguns recursos '
                'solicitados não foram concedidos.',
                context={
                    'workspace_id': str(workspace.pk),
                    'provider_key': provider.key,
                    'missing_resources': sorted(missing),
                },
            )

        # -----------------------------------------------------------------
        # 5. Cifrar payload.
        # -----------------------------------------------------------------
        encrypted = IntegrationService._encrypt_payload(

        )

        # 6. Persistir (cria ou atualiza).
        # -----------------------------------------------------------------
        now = timezone.now()

        if existing is None:
            integration = WorkspaceIntegration.objects.create(
                workspace=workspace,
                provider=provider,
                external_account_id=oauth_result.external_account_id,
                external_account_email=oauth_result.external_account_email,

                encrypted_payload=encrypted,
                granted_scopes=list(oauth_result.granted_scopes),
                status=WorkspaceIntegration.STATUS_ACTIVE,
                last_refresh_at=now,
                is_preferred=False,  # será resolvido pelo service de acesso
                created_by=actor if actor and actor.is_authenticated else None,
            )
            logger.info(
                'WorkspaceIntegration criada | workspace=%s '
                'provider=%s account=%s',
                workspace.pk, provider.key,
                oauth_result.external_account_id,

            existing.granted_scopes = list(oauth_result.granted_scopes)

    @staticmethod
    def get_valid_access_token(
        integration: WorkspaceIntegration,
    ) -> str:
        """
        Retorna um access_token VÁLIDO, renovando se necessário.

        Esta é a PORTA PADRÃO para qualquer código que precise
        chamar a API externa em nome do usuário.

        Comportamento:
            1. Se a integração não está `active`, levanta
               `IntegrationInactiveError` / `ConnectionExpiredError`
               / `ConnectionRevokedError` (conforme o estado).
            2. Decifra o payload.
            3. Se o access_token expira em mais de REFRESH_MARGIN,
               devolve-o sem tocar no Google.
            4. Se falta menos que a margem (ou já expirou), dispara
               `refresh_credentials` e devolve o novo token.

        Levanta:
            ConnectionRevokedError — integração revogada.
            ConnectionExpiredError — refresh_token inválido (Google
                rejeitou).
            IntegrationInactiveError — estado inesperado.
            RefreshTransientError — falha transitória de rede.
        """
        IntegrationService._assert_usable(integration)

        creds = IntegrationService.get_credentials(integration)

        if not IntegrationService._needs_refresh(creds):
            return creds['access_token']

        # Precisamos renovar. Delegamos para o método que tem lock.
        refreshed = IntegrationService.refresh_credentials(integration)
        return refreshed['access_token']

    @staticmethod
    def refresh_credentials(
        integration: WorkspaceIntegration,
    ) -> dict:
        """
        Força a renovação dos tokens da integração.

        Serializado por `select_for_update`. Se duas chamadas
        concorrentes pedirem refresh, apenas a primeira renova;
        a segunda vê o token novo e não chama o Google.

        Retorna o payload decifrado novo (dict).

        Levanta:
            ConnectionExpiredError — refresh_token inválido. A
                integração é marcada como `expired` ANTES da exceção.
            RefreshTransientError — falha transitória. A integração
                NÃO é marcada como expired — apenas `last_error_*`.
            IntegrationInactiveError — integração não está `active`
                quando o refresh foi pedido.
            ConnectionRevokedError — integração já está revogada.

        NOTA SOBRE TRANSAÇÕES:
            A marcação de estado de erro (`_mark_expired`,
            `_record_transient_error`, `_mark_error`) NÃO pode
            acontecer dentro do `transaction.atomic()`, porque o
            rollback da exceção desfaria a marcação. Por isso,
            usamos um padrão de "raise outside atomic": capturamos
            a exceção dentro do bloco, guardamos para re-levantar
            depois, e marcamos o estado FORA do bloco.
        """
        IntegrationService._assert_usable(integration)

        pending_exception = None
        pending_state_update = None
        result = None

        with transaction.atomic():
            # Relê a integração com lock. Se outra thread renovou
            # enquanto esperávamos, os dados novos estarão aqui.
            locked = (
                WorkspaceIntegration.objects
                .select_for_update()
                .get(pk=integration.pk)
            )

            # Se outra thread renovou, o `last_refresh_at` mudou e
            # o payload é novo. Não renovamos de novo.
            if not IntegrationService._integration_needs_refresh(locked):
                return IntegrationService.get_credentials(locked)

            creds = IntegrationService.get_credentials(locked)
            refresh_token = creds.get('refresh_token')

            if not refresh_token:
                # Estado inconsistente — sem refresh_token, a
                # integração não deveria estar `active`.
                pending_state_update = ('expired', 'missing_refresh_token')
                pending_exception = ConnectionExpiredError(
                    'A integração não possui credencial de longa '
                    'duração. Reconecte.',
                    context={'integration_id': str(locked.pk)},
                )
            else:
                connector = ProviderRegistry.get(locked.provider.key)

                try:
                    oauth_result = connector.refresh(
                        refresh_token=refresh_token,
                    )
                except RefreshTokenInvalidError:
                    # O Google disse: essa credencial não vale mais.
                    pending_state_update = ('expired', 'refresh_invalid')
                    pending_exception = ConnectionExpiredError(
                        'A credencial de longa duração foi revogada ou '
                        'expirou. Reconecte a integração.',
                        context={'integration_id': str(locked.pk)},
                    )
                except RefreshTransientError as exc:
                    # Falha de rede / 5xx. NÃO marca como expired —
                    # pode ser só um blip.
                    pending_state_update = (
                        'transient', 'refresh_transient',
                    )
                    pending_exception = exc
                except Exception as exc:
                    # Qualquer erro inesperado. Marcamos como erro
                    # mas não sabemos se é permanente.
                    logger.exception(
                        'Erro inesperado no refresh (integration=%s)',
                        locked.pk,
                    )
                    pending_state_update = ('error', 'refresh_unknown')
                    pending_exception = WorkspaceIntegrationError(
                        'Erro inesperado ao renovar credenciais.',
                        context={'integration_id': str(locked.pk)},
                    )

                if pending_exception is None:
                    # Persiste os tokens novos.
                    encrypted = IntegrationService._encrypt_payload(
                        access_token=oauth_result.access_token,
                        refresh_token=(
                            oauth_result.refresh_token or refresh_token
                        ),
                        expires_at=oauth_result.expires_at,
                        scopes=(
                            oauth_result.granted_scopes
                            or creds.get('scopes', [])
                        ),
                    )

                    locked.encrypted_payload = encrypted
                    locked.last_refresh_at = timezone.now()
                    locked.last_error_at = None
                    locked.last_error_code = ''
                    locked.save(update_fields=[
                        'encrypted_payload',
                        'last_refresh_at',
                        'last_error_at',
                        'last_error_code',
                        'updated_at',
                    ])

                    logger.info(
                        'Refresh concluído | integration=%s',
                        locked.pk,
                    )

                    # Atualiza o objeto em memória do chamador para
                    # que ele veja o estado novo caso faça mais
                    # alguma coisa com ele.
                    integration.encrypted_payload = encrypted
                    integration.last_refresh_at = locked.last_refresh_at

                    result = IntegrationService.get_credentials(locked)

        # FORA do atomic — o commit já aconteceu. Agora:
        #
        # 1. Se houve erro, marcamos o estado (fora do rollback).
        # 2. Re-levantamos a exceção para o chamador.
        if pending_state_update is not None:
            state, error_code = pending_state_update

            # Relê do banco para pegar o estado atual.
            integration.refresh_from_db()

            if state == 'expired':
                IntegrationService._mark_expired(
                    integration, error_code=error_code,
                )
            elif state == 'transient':
                IntegrationService._record_transient_error(
                    integration, error_code=error_code,
                )
            elif state == 'error':
                IntegrationService._mark_error(
                    integration, error_code=error_code,
                )

            raise pending_exception

        return result

    @staticmethod
    @transaction.atomic
    def revoke_integration(
        *,
        integration: WorkspaceIntegration,
        actor=None,
        reason: str = 'user_revoked',
    ) -> WorkspaceIntegration:
        """
        Revoga a integração.

        Comportamento:
            1. Chama o connector para revogar os tokens no provider
               (best-effort).
            2. Marca a integração local como `revoked` — SEMPRE,
               mesmo se o provider falhar.
            3. Preserva o histórico (não deleta).

        Idempotente: se a integração já está `revoked`, não faz nada.

        Levanta:
            IntegrationInactiveError — se a integração está em estado
                inconsistente (ex: `error`) e não pode ser revogada.

        NOTA sobre a revogação no provider:
            A ordem é: revogar no provider PRIMEIRO, marcar local
            DEPOIS. Se o provider estiver offline, não travamos o
            usuário — a integração é marcada como `revoked` mesmo
            assim. Um cron futuro pode re-tentar a revogação remota
            (não implementado na V1).
        """
        if integration.status == WorkspaceIntegration.STATUS_REVOKED:
            return integration

        # Lê as credenciais para ter os tokens em mãos.
        try:
            creds = IntegrationService.get_credentials(integration)
        except Exception:
            # Se as credenciais estão corrompidas, não conseguimos
            # revogar no provider. Mas ainda assim marcamos como
            # revogada localmente — o acesso do usuário não depende
            # do provider.
            logger.warning(
                'Falha ao decifrar credenciais ao revogar '
                '(integration=%s). Marcando como revoked localmente.',
                integration.pk,
            )
            creds = {}

        connector = ProviderRegistry.get(integration.provider.key)

        # Best-effort no provider.
        try:
            connector.revoke(
                access_token=creds.get('access_token', ''),
                refresh_token=creds.get('refresh_token'),
            )
        except Exception:
            logger.exception(
                'Falha ao revogar no provider (integration=%s). '
                'Marcando como revoked localmente mesmo assim.',
                integration.pk,
            )

        integration.status = WorkspaceIntegration.STATUS_REVOKED
        integration.is_preferred = False  # revogada nunca é preferida
        integration.last_error_at = timezone.now()
        integration.last_error_code = reason
        integration.save(update_fields=[
            'status',
            'is_preferred',
            'last_error_at',
            'last_error_code',
            'updated_at',
        ])

        logger.info(
            'WorkspaceIntegration revogada | integration=%s '
            'reason=%s actor=%s',
            integration.pk,
            reason,
            getattr(actor, 'email', None) or 'system',
        )

        return integration

    @staticmethod
    def get_credentials(integration: WorkspaceIntegration) -> dict:
        """
        Decifra e devolve o payload da integração.

        Uso interno. Não expõe em APIs públicas.

        Retorna dict com:
            {
                "access_token": str,
                "refresh_token": str,
                "expires_at": str | None,  # ISO 8601
                "scopes": list[str],

                integration.pk,
            )
            raise WorkspaceIntegrationError(
                'Credenciais corrompidas. Reconecte a integração.',
                context={
                    'integration_id': str(integration.pk),
                    'reason': 'invalid_payload',
                },
            ) from exc

    @staticmethod
    def _assert_usable(integration: WorkspaceIntegration) -> None:
        """
        Levanta a exceção apropriada se a integração não pode ser
        usada no estado atual.
        """
        if integration.status == WorkspaceIntegration.STATUS_ACTIVE:
            return

        if integration.status == WorkspaceIntegration.STATUS_REVOKED:
            raise ConnectionRevokedError(
                'A integração foi revogada.',
                context={'integration_id': str(integration.pk)},
            )

        if integration.status == WorkspaceIntegration.STATUS_EXPIRED:
            raise ConnectionExpiredError(
                'A integração expirou. Reconecte.',
                context={'integration_id': str(integration.pk)},
            )

        raise IntegrationInactiveError(
            f'A integração está no estado "{integration.status}".',
            context={
                'integration_id': str(integration.pk),
                'status': integration.status,
            },
        )

    @staticmethod
    def _needs_refresh(creds: dict) -> bool:
        """
        Determina se o access_token do payload precisa ser renovado.

        Regra: renova se falta menos de REFRESH_MARGIN_SECONDS para
        expirar (incluindo já expirado).

        Se `expires_at` for None (não informado), assume que NÃO
        precisa — quem decide é o provider, que vai devolver 401 se
        o token estiver ruim.
        """
        expires_at_str = creds.get('expires_at')
        if not expires_at_str:
            return False

        try:
            expires_at = datetime.fromisoformat(expires_at_str)
        except (TypeError, ValueError):
            # Não conseguimos interpretar — melhor renovar.
            return True

        # Garantir timezone-aware.
        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(tzinfo=dt_timezone.utc)

        now = datetime.now(tz=dt_timezone.utc)
        margin = timedelta(seconds=REFRESH_MARGIN_SECONDS)

        return expires_at <= (now + margin)

    @staticmethod
    def _integration_needs_refresh(
        integration: WorkspaceIntegration,
    ) -> bool:
        """
        Versão que opera sobre a integração — lê o payload e chama
        `_needs_refresh`.

        Usada dentro do bloco com lock para decidir se ainda vale
        renovar ou se outra thread já fez.
        """
        try:
            creds = IntegrationService.get_credentials(integration)
        except Exception:
            return True  # se não consegue ler, assume que precisa
        return IntegrationService._needs_refresh(creds)

        integration.save(update_fields=[

            'last_error_code',
            'updated_at',

    @staticmethod
    def _encrypt_payload(
        *,
        access_token: str,
        refresh_token: str,
        expires_at,
        scopes: list,
    ) -> bytes:
        """
        Serializa e cifra o payload de credenciais.
        """
        payload = {
            'schema_version': PAYLOAD_SCHEMA_VERSION,
            'access_token': access_token,
            'refresh_token': refresh_token,
            'expires_at': (
                expires_at.isoformat() if expires_at else None
            ),
            'scopes': list(scopes or []),
        }
        blob = json.dumps(payload).encode('utf-8')
        return EncryptionService.encrypt(blob)

    @staticmethod
    def _extract_refresh_token(
        integration: WorkspaceIntegration,
    ) -> Optional[str]:
        """
        Tenta decifrar o refresh_token da integração existente.

            - a decifra falhar,
            - o payload não tiver refresh_token,
            - o refresh_token for string vazia.

        except Exception as exc:
            logger.warning(
                'Falha ao decifrar integração existente (%s): %s. '
                'Tratando como ausência de refresh_token.',
                integration.pk, type(exc).__name__,
            )
            return None

        rt = data.get('refresh_token')
        return rt if rt else None

"""
OAuthStateService — armazenamento e consumo do `state` OAuth.

O `state` do OAuth é uma string anti-CSRF. Deve ser:
    1. Imprevisível (nonce criptograficamente seguro).
    2. Vinculado ao usuário e ao workspace que iniciaram a conexão.
    3. Consumido UMA ÚNICA VEZ (DELETE atômico).
    4. Com expiração curta (TTL 5 minutos).

O `state` em si é o nonce. O contexto vive no Redis, indexado por ele.

Armazenamento:
    Redis, chave: integrations:oauth_state:<nonce>
    Valor: JSON com o contexto.
    TTL: 300 segundos.

SOBRE `next_path`
-----------------
`next_path` é o caminho INTERNO para onde o callback deve redirecionar
o usuário depois de conectar com sucesso.

    final_redirect  → onde redirecionar quando o callback FALHA
                      (erro do Google, state inválido, escopo recusado).
                      Tipicamente é a página onde o usuário começou.

    next_path       → onde redirecionar quando o callback TIVER SUCESSO.
                      Mesmo lugar, mas o frontend recebe um sinal
                      (`?google_connected=1`) para reabrir o que estava
                      fazendo.

Ambos passam por `safe_redirect_or_fallback` no callback.
"""

import json
import logging
import secrets
from dataclasses import asdict, dataclass
from datetime import datetime, timezone
from typing import Optional

from django.core.cache import cache

logger = logging.getLogger('integrations')


# Prefixo das chaves no cache. Nunca colidir com outras chaves do Redis.
CACHE_KEY_PREFIX = 'integrations:oauth_state:'

# TTL do state. Após isso, o state expira.
STATE_TTL_SECONDS = 300


@dataclass(frozen=True)
class OAuthStateContext:
    """
    Contexto associado a um state OAuth em andamento.
    """
    nonce: str
    user_id: str
    workspace_id: str
    provider_key: str
    resources: list
    redirect_uri: str
    final_redirect: str
    next_path: str
    created_at: str  # ISO 8601


class OAuthStateService:
    """
    Gerencia o ciclo de vida do `state` OAuth.
    """

    @staticmethod
    def issue(
        *,
        user_id: str,
        workspace_id: str,
        provider_key: str,
        resources: list,
        redirect_uri: str,
        final_redirect: str,
        next_path: str = '',
    ) -> str:
        """
        Cria um novo state, persiste o contexto no Redis, e devolve
        o nonce que deve ser enviado ao Google.

        Retorna: o nonce (string imprevisível).
        """
        nonce = secrets.token_urlsafe(32)

        context = OAuthStateContext(
            nonce=nonce,
            user_id=str(user_id),
            workspace_id=str(workspace_id),
            provider_key=provider_key,
            resources=list(resources),
            redirect_uri=redirect_uri,
            final_redirect=final_redirect,
            next_path=next_path or '',
            created_at=datetime.now(timezone.utc).isoformat(),
        )

        cache.set(
            f'{CACHE_KEY_PREFIX}{nonce}',
            json.dumps(asdict(context)),
            timeout=STATE_TTL_SECONDS,
        )

        logger.debug(
            'OAuth state emitido | user=%s workspace=%s provider=%s '
            'next=%s',
            user_id, workspace_id, provider_key, next_path or '—',
        )

        return nonce

    @staticmethod
    def consume(nonce: str) -> Optional[OAuthStateContext]:
        """
        Consome (lê e apaga) um state.

        Retorna o contexto, ou None se:
            - o nonce é vazio,
            - o state não existe (expirado ou nunca existiu).

        Após esta chamada, o nonce é inválido para sempre.
        """
        if not nonce:
            return None

        key = f'{CACHE_KEY_PREFIX}{nonce}'
        raw = cache.get(key)

        if raw is None:
            logger.warning(
                'OAuth state não encontrado (expirado ou forjado): %s',
                nonce[:8] if nonce else '',
            )
            return None

        # Apaga ANTES de decodificar, para garantir consumo único
        # mesmo se o decode falhar.
        cache.delete(key)

        try:
            data = json.loads(raw)
        except (TypeError, ValueError):
            logger.warning(
                'OAuth state malformado no Redis: %s', nonce[:8],
            )
            return None

        # Compatibilidade retroativa: states antigos não têm `next_path`.
        data.setdefault('next_path', '')

        try:
            return OAuthStateContext(**data)
        except TypeError as exc:
            logger.warning(
                'OAuth state com schema inválido: %s', exc,
            )
            return None

"""
Validação de destinos de redirecionamento pós-OAuth.

Todo redirecionamento após o callback DEVE apontar para um destino
interno conhecido. Nunca para uma URL arbitrária passada pelo usuário.

Regras:
    - Deve começar com `/`.
    - Não pode começar com `//` (protocol-relative URL).
    - Não pode começar com `/\` (bypass histórico).
    - Não pode conter `\` (backslash como separador alternativo).
    - Não pode conter `..` (path traversal).
    - Não pode conter `%2e` (encoding URL de `.`, usado para bypass).
    - Deve casar com um dos padrões internos permitidos.

Padrões permitidos (regex):
    - /w/<public_id>/...          → qualquer página dentro de um workspace
    - /fluxo/                     → painel de fluxo do arquiteto
    - /clientes/                  → painel de clientes
    - /configuracoes/             → configurações do workspace
    - /mesa/                      → página da Mesa
    - /workspaces/entrar/         → entrada de workspaces (fallback)

Query strings são aceitas (ex: `/fluxo/?foo=bar`).
"""

import logging
import re
import urllib.parse

logger = logging.getLogger('integrations')


_ALLOWED_PATTERNS = [
    re.compile(r'^/w/[A-Za-z0-9_-]+(/[^\\]*)?$'),
    re.compile(r'^/fluxo(/[^\\]*)?$'),
    re.compile(r'^/clientes(/[^\\]*)?$'),
    re.compile(r'^/configuracoes(/[^\\]*)?$'),
    re.compile(r'^/mesa(/[^\\]*)?$'),
    re.compile(r'^/workspaces/entrar(/[^\\]*)?$'),
]


DEFAULT_FALLBACK = '/workspaces/entrar/'


def _has_path_traversal(target: str) -> bool:
    """
    Detecta tentativas de path traversal.

    Estratégia:
        1. Decodifica a URL repetidamente (até estabilizar) para
           pegar encodings duplos (`%252e` → `%2e` → `.`).
        2. Checa se o path decodificado contém `..` como segmento.
        3. Compara o path original com o normalizado.
    """
    # Decodifica repetidamente para pegar encodings múltiplos.
    decoded = target
    for _ in range(3):
        new_decoded = urllib.parse.unquote(decoded)
        if new_decoded == decoded:
            break
        decoded = new_decoded

    # Checa `..` como segmento de path (evita falso positivo com
    # nomes legítimos como "my..folder").
    # Extrai só o path (antes do `?`).
    path_part = decoded.split('?', 1)[0]
    segments = path_part.split('/')
    if '..' in segments:
        return True

    # Checa a versão original também, caso `unquote` não pegue.
    orig_path = target.split('?', 1)[0]
    if '..' in orig_path.split('/'):
        return True

    return False


def is_safe_redirect(target: str) -> bool:
    """
    Retorna True se `target` é um destino interno permitido.
    """
    if not target or not isinstance(target, str):
        return False

    if not target.startswith('/'):
        return False

    if target.startswith('//'):
        return False

    if target.startswith('/\\') or '\\' in target:
        return False

    if _has_path_traversal(target):
        return False

    return any(p.match(target) for p in _ALLOWED_PATTERNS)


def safe_redirect_or_fallback(target: str) -> str:
    """
    Se `target` é seguro, devolve-o. Caso contrário, devolve o
    destino de fallback.
    """
    if is_safe_redirect(target):
        return target
    if target:
        logger.warning(
            'Redirect bloqueado (não corresponde a nenhum padrão '
            'interno permitido): %r',
            target[:80],
        )
    return DEFAULT_FALLBACK

def _walk_python_files(root: Path):
    """Devolve todos os .py sob `root`, exceto testes e migrations."""
    for path in root.rglob('*.py'):
        rel = path.relative_to(root)
        parts = rel.parts
        if 'migrations' in parts:
            continue
        if 'tests' in parts:
            continue
        yield path


def _iter_imports(tree: ast.AST):
    """
    Devolve tuplas (module_name, is_from) para cada import no AST.
    """
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                yield (alias.name, False)
        elif isinstance(node, ast.ImportFrom):
            if node.module:
                yield (node.module, True)


def _has_dynamic_import(tree: ast.AST):
    """
    Detecta __import__() e importlib.import_module().
    """
    for node in ast.walk(tree):
        if isinstance(node, ast.Call):
            fn = node.func
            if isinstance(fn, ast.Name) and fn.id == '__import__':
                return True
            if isinstance(fn, ast.Attribute) and fn.attr == 'import_module':
                return True
    return False

def _make_integration(
    workspace, provider, actor,
    access_token='valid-access',
    refresh_token='valid-refresh',
    expires_in_seconds=3600,
    status=WorkspaceIntegration.STATUS_ACTIVE,
    is_preferred=False,
):
    """
    Cria uma WorkspaceIntegration com payload cifrado real.
    """
    expires_at = (
        datetime.now(timezone.utc) + timedelta(seconds=expires_in_seconds)
    )
    payload = {
        'schema_version': 1,
        'access_token': access_token,
        'refresh_token': refresh_token,
        'expires_at': expires_at.isoformat(),
        'scopes': ['https://www.googleapis.com/auth/drive.readonly'],
    }

    return WorkspaceIntegration.objects.create(
        workspace=workspace,
        provider=provider,
        external_account_id='ext-user-1',
        external_account_email='user@google.com',
        encrypted_payload=encrypted,
        granted_scopes=payload['scopes'],
        status=status,
        is_preferred=is_preferred,
        created_by=actor,
    )

class TestWorkspaceIntegrationConcurrency(TransactionTestCase):
    """
    Verifica que duas chamadas concorrentes a `get_for_workspace`,
    com apenas uma conta ativa, marcam essa conta como preferida
    APENAS UMA VEZ e retornam a mesma integração.
    """

    reset_sequences = False

    def setUp(self):
        from apps.workspaces.models import Workspace

        self.provider = IntegrationProvider.objects.create(
            key='google',
            name='Google',
            category='productivity',
        )
        self.workspace = Workspace.objects.create(
            name='WS Conc',
            slug='ws-conc-int',
        )
        self.integration = WorkspaceIntegration.objects.create(
            workspace=self.workspace,
            provider=self.provider,
            external_account_id='ext-1',
            external_account_email='a@b.com',
            encrypted_payload=b'x',
            granted_scopes=[],
            is_preferred=False,
            status=WorkspaceIntegration.STATUS_ACTIVE,
        )

    def tearDown(self):
        connection.close()

    def test_concurrent_get_marks_only_one_preferred(self):
        errors = []
        results = []
        lock = threading.Lock()

        def worker():
            try:
                integration = (
                    WorkspaceIntegrationService.get_for_workspace(
                        workspace=self.workspace,
                        provider_key='google',
                    )
                )
                with lock:
                    results.append(integration.id)
            except Exception as exc:
                with lock:
                    errors.append(exc)
            finally:
                connection.close()

        t1 = threading.Thread(target=worker)
        t2 = threading.Thread(target=worker)
        t1.start()
        t2.start()
        t1.join()
        t2.join()

        # Nenhum erro esperado
        assert not errors, f'Erros durante concorrência: {errors}'

        # Ambas devem retornar a mesma integração
        assert len(results) == 2
        assert results[0] == results[1]
        assert results[0] == self.integration.id

        # E apenas uma preferida no banco
        preferred_count = WorkspaceIntegration.objects.filter(
            workspace=self.workspace,
            is_preferred=True,
            status=WorkspaceIntegration.STATUS_ACTIVE,
        ).count()
        assert preferred_count == 1

    def test_concurrent_ambiguity_is_consistent(self):
        """
        Duas contas ativas, nenhuma preferida. Duas chamadas
        concorrentes devem ambas levantar AmbiguousConnectionError.

        WorkspaceIntegration.objects.create(

            status=WorkspaceIntegration.STATUS_ACTIVE,

        lock = threading.Lock()

        def worker():
            try:
                integration = (
                    WorkspaceIntegrationService.get_for_workspace(
                        workspace=self.workspace,
                        provider_key='google',
                    )
                )
                with lock:
                    results.append(integration.id)
            except AmbiguousConnectionError as exc:
                with lock:
                    errors.append(exc)
            except Exception as exc:
                with lock:
                    errors.append(exc)
            finally:
                connection.close()

        t1 = threading.Thread(target=worker)
        t2 = threading.Thread(target=worker)
        t1.start()
        t2.start()
        t1.join()
        t2.join()

        assert not results, (

        assert len(errors) == 2
        assert all(
            isinstance(e, AmbiguousConnectionError) for e in errors

"""
Views HTML do módulo integrations.

Rotas:
    GET /integrations/google/start/?next=<path>
        Autenticado. Inicia o fluxo OAuth para o workspace ativo
        da sessão.

    GET /integrations/google/callback/?code=...&state=...
        Recebe o retorno do Google. Aplica políticas, persiste a
        WorkspaceIntegration e redireciona para o `next_path` do
        state (com sinal `?google_connected=1`), ou para o
        `final_redirect` em caso de falha.

FRONTEIRA:
    Este módulo NÃO importa apps.console.

SOBRE `next`
------------
O parâmetro `next` (query string) é o caminho para onde o usuário
deve voltar DEPOIS de conectar com sucesso. É validado por
`safe_redirect_or_fallback` e persistido no state OAuth.

No retorno, o backend acrescenta `?google_connected=1` na URL.
O frontend usa esse sinal para reabrir a ação que estava em
andamento (por exemplo, o modal de Importar Forms).

SOBRE `login_hint`
------------------
Passamos `request.user.email` como `login_hint` para o Google.
Isso pré-seleciona a conta correta na tela de consentimento,
evitando mostrar outras contas logadas no navegador.

Isso NÃO cria acoplamento com `accounts`: apenas lemos
`request.user.email`, que é campo padrão do Django.
"""

import logging

from django.contrib import messages
from django.http import HttpResponseRedirect
from django.shortcuts import redirect
from django.views import View

from apps.integrations.exceptions import (
    IntegrationConfigurationError,
    IntegrationError,
    InsufficientScopesError,
    OAuthExchangeError,
    UnsupportedResourceError,
    WorkspaceIntegrationError,
)
from apps.integrations.models import IntegrationProvider
from apps.integrations.providers.registry import ProviderRegistry
from apps.integrations.services.integration_service import IntegrationService
from apps.integrations.services.oauth_state import OAuthStateService
from apps.integrations.services.redirects import safe_redirect_or_fallback
from apps.workspaces.queries import WorkspaceQueries

logger = logging.getLogger('integrations')


# Recursos padrão solicitados ao Google.
# Hoje: Drive (leitura) + Forms (leitura).
DEFAULT_GOOGLE_RESOURCES = [
    'drive.file.read',
    'forms.body.read',
]


# Destino padrão quando não há `next` no query string.
DEFAULT_NEXT_PATH = '/workspaces/entrar/'


# Sinal para o frontend saber que o OAuth foi concluído com sucesso.
GOOGLE_CONNECTED_PARAM = 'google_connected'


# =========================================================================
# HELPERS
# =========================================================================

def _resolve_current_workspace(request):
    """
    Resolve o workspace ativo a partir da sessão.
    Retorna None se não houver.
    """
    ws_id = request.session.get('current_workspace_id')
    if not ws_id:
        return None
    return WorkspaceQueries.get_for_user(ws_id, request.user)


def _callback_redirect_uri(request) -> str:
    """
    Monta a redirect_uri absoluta do callback.
    """
    return request.build_absolute_uri('/integrations/google/callback/')


def _with_google_connected(path: str) -> str:
    """
    Acrescenta `?google_connected=1` a uma URL interna.

    Preserva query strings existentes.
    """
    safe = safe_redirect_or_fallback(path)
    sep = '&' if '?' in safe else '?'
    return f'{safe}{sep}{GOOGLE_CONNECTED_PARAM}=1'


# =========================================================================
# START — INICIA O FLUXO OAUTH
# =========================================================================

class GoogleOAuthStartView(View):
    """
    GET /integrations/google/start/?next=<path>

    Exige:
        - usuário autenticado;
        - workspace ativo na sessão.

    Fluxo:
        1. Valida pré-requisitos.
        2. Lê `next` da query string e valida.
        3. Gera state no Redis via OAuthStateService (com `next_path`).
        4. Chama GoogleConnector.build_auth_url, passando o email do
           usuário como `login_hint`.
        5. Redireciona (302) para o Google.

    NOTA sobre capability:
        NÃO exigimos `integrations.connection.manage`. Conectar uma
        conta Google é parte de operar o workspace — qualquer membro
        pode. O controle de escopo real acontece nos consumidores
        (ex: `briefings.template.manage` para importar Forms).
    """

    def get(self, request):
        if not request.user.is_authenticated:
            return redirect('/login/')

        workspace = _resolve_current_workspace(request)
        if workspace is None:
            messages.error(
                request,
                'Nenhum workspace ativo. Escolha um workspace antes.',
            )
            return redirect('/workspaces/entrar/')

        # Lê e valida o `next`.
        raw_next = request.GET.get('next', '')
        next_path = safe_redirect_or_fallback(raw_next or DEFAULT_NEXT_PATH)

        # `final_redirect` é o destino em caso de FALHA no callback.
        # `next_path` é o destino em caso de SUCESSO.
        # Ambos são o mesmo caminho aqui — o sinal `?google_connected=1`
        # é acrescentado apenas no sucesso.
        final_redirect = next_path

        try:
            provider = IntegrationProvider.objects.get(
                key='google', is_active=True,
            )
        except IntegrationProvider.DoesNotExist:
            logger.warning(
                'Provider Google não encontrado ou inativo | user=%s',
                request.user.pk,
            )
            messages.error(
                request,
                'O provider Google não está disponível neste '
                'momento. Contate o suporte.',
            )
            return redirect(final_redirect)

        redirect_uri = _callback_redirect_uri(request)

        state = OAuthStateService.issue(
            user_id=str(request.user.pk),
            workspace_id=str(workspace.pk),
            provider_key='google',
            resources=DEFAULT_GOOGLE_RESOURCES,
            redirect_uri=redirect_uri,
            final_redirect=final_redirect,
            next_path=next_path,
        )

        try:
            connector = ProviderRegistry.get('google')
            auth_url = connector.build_auth_url(
                redirect_uri=redirect_uri,
                resources=DEFAULT_GOOGLE_RESOURCES,
                state=state,
                # ─────────────────────────────────────────────────
                # login_hint: usa o email do usuário logado na Clivo
                # para pré-selecionar a conta correta no Google.
                #
                # O `accounts` continua sendo a única autoridade
                # sobre "quem é o usuário". Aqui apenas LEMOS
                # `request.user.email`, um campo padrão do Django —
                # sem importar nada de `apps.accounts`.
                # ─────────────────────────────────────────────────
                login_hint=request.user.email,
            )
        except IntegrationConfigurationError:
            logger.exception(
                'Configuração do Google OAuth ausente em /start/',
            )
            messages.error(
                request,
                'O sistema não está configurado para conectar ao '
                'Google. Contate o suporte.',
            )
            return redirect(final_redirect)
        except UnsupportedResourceError:
            logger.exception(
                'Recurso não suportado em /start/: %s',
                DEFAULT_GOOGLE_RESOURCES,
            )
            messages.error(
                request,
                'A configuração de recursos do Google está '
                'inconsistente. Contate o suporte.',
            )
            return redirect(final_redirect)

        logger.info(
            'Iniciando OAuth Google | user=%s | workspace=%s | login_hint=%s',
            request.user.email,
            workspace.pk,
            request.user.email,
        )

        return HttpResponseRedirect(auth_url)

# CALLBACK — RECEBE O RETORNO DO GOOGLE
# =========================================================================

class GoogleOAuthCallbackView(View):
    """
    GET /integrations/google/callback/?code=...&state=...

    Fluxo:
        1. Valida state (consome do Redis).
        2. Confirma que `request.user` bate com o contexto do state.
        3. Confirma que o workspace do contexto ainda é acessível.
        4. Troca code por tokens (exchange_code).
        5. Aplica políticas (preservar refresh_token, validar
           escopos, exigir refresh_token durável).
        6. Cifra e persiste a WorkspaceIntegration.
        7. Redireciona:
           - sucesso  → ctx.next_path + `?google_connected=1`
           - falha    → ctx.final_redirect
    """

    def get(self, request):
        if not request.user.is_authenticated:
            return redirect('/login/')

        error = request.GET.get('error')
        state_nonce = request.GET.get('state') or ''

        # Consome o state ANTES de checar erro, porque precisamos
        # do `final_redirect`/`next_path` do contexto.
        ctx = OAuthStateService.consume(state_nonce)

        if ctx is None:
            logger.warning(
                'Callback com state inválido/expirado | user=%s',
                request.user.pk,
            )
            messages.error(
                request,
                'Sua sessão de conexão expirou ou é inválida. '
                'Tente novamente.',
            )
            return redirect('/workspaces/entrar/')

        # Confirma vínculo: o usuário que clicou em "Conectar"
        # precisa ser o MESMO que está recebendo o callback.
        if str(request.user.pk) != ctx.user_id:
            logger.warning(
                'Callback state pertence a outro usuário | '
                'esperado=%s atual=%s',
                ctx.user_id, request.user.pk,
            )
            messages.error(
                request,
                'Esta conexão não pertence à sua sessão. '
                'Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        # Se o usuário cancelou, não é erro técnico.
        if error == 'access_denied':
            logger.info(
                'Usuário cancelou OAuth Google | user=%s',
                request.user.pk,
            )
            messages.warning(
                request,
                'Você cancelou a conexão com o Google.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        if error:
            logger.warning(
                'Callback com erro do Google: %s', error,
            )
            messages.error(
                request,
                'O Google retornou um erro ao conectar. '
                'Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        code = request.GET.get('code')
        if not code:
            messages.error(
                request,
                'Código de autorização ausente. Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        # Confirma que o workspace ainda é acessível.
        workspace = WorkspaceQueries.get_for_user(
            ctx.workspace_id, request.user,
        )
        if workspace is None:
            messages.error(
                request,
                'Você não tem mais acesso a este workspace.',
            )
            return redirect('/workspaces/entrar/')

        # Provider.
        try:
            provider = IntegrationProvider.objects.get(
                key=ctx.provider_key, is_active=True,
            )
        except IntegrationProvider.DoesNotExist:
            messages.error(
                request,
                'O provider desta conexão não está mais disponível.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        # Troca code por tokens.
        try:
            connector = ProviderRegistry.get(ctx.provider_key)
            oauth_result = connector.exchange_code(
                code=code,
                redirect_uri=ctx.redirect_uri,
            )
        except OAuthExchangeError:
            logger.warning(
                'Falha ao trocar code por tokens | user=%s',
                request.user.pk,
            )
            messages.error(
                request,
                'Não foi possível concluir a conexão com o Google. '
                'Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))
        except IntegrationError as exc:
            logger.exception(
                'Erro de integração no callback: %s', exc,
            )
            messages.error(
                request,
                'Erro ao processar a conexão. Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        # Aplica políticas e persiste.
        try:
            IntegrationService.upsert_from_oauth(
                workspace=workspace,
                provider=provider,
                oauth_result=oauth_result,
                requested_resources=ctx.resources,
                actor=request.user,
            )
        except InsufficientScopesError as exc:
            missing = exc.context.get('missing_resources', [])
            logger.info(
                'Conexão rejeitada por escopos insuficientes | '
                'user=%s missing=%s',
                request.user.pk, missing,
            )
            messages.error(
                request,
                'Faltam permissões para concluir a conexão. '
                'Reconecte autorizando todas as opções solicitadas.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))
        except WorkspaceIntegrationError as exc:
            reason = exc.context.get('reason')
            if reason == 'no_refresh_token':
                messages.error(
                    request,
                    'O Google não retornou credencial de longa '
                    'duração. Reconecte e autorize o acesso '
                    'permanente.',
                )
            else:
                messages.error(
                    request,
                    'Não foi possível ativar a conexão. '
                    'Tente novamente.',
                )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))
        except Exception:
            logger.exception(
                'Falha inesperada ao persistir integração | user=%s',
                request.user.pk,
            )
            messages.error(
                request,
                'Erro inesperado ao salvar a conexão. '
                'Tente novamente.',
            )
            return redirect(safe_redirect_or_fallback(ctx.final_redirect))

        messages.success(
            request,
            'Google conectado com sucesso.',
        )

        logger.info(
            'Google conectado com sucesso | user=%s | workspace=%s',
            request.user.email,
            workspace.pk,
        )

        # Redireciona para `next_path` com o sinal `?google_connected=1`.
        success_url = _with_google_connected(
            ctx.next_path or ctx.final_redirect
        )
        return redirect(success_url)
